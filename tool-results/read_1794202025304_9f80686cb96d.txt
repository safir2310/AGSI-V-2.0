     1→"use client";
     2→
     3→import { useCallback, useEffect, useRef, useState } from "react";
     4→import { createPortal } from "react-dom";
     5→import { AnimatePresence, motion, useReducedMotion } from "framer-motion";
     6→import { QRCodeSVG } from "qrcode.react";
     7→import {
     8→  BadgeCheck,
     9→  FlaskConical,
    10→  Loader2,
    11→  Printer,
    12→  RefreshCw,
    13→  ScanLine,
    14→  ShieldCheck,
    15→  Timer,
    16→  TriangleAlert,
    17→  X,
    18→  XCircle,
    19→} from "lucide-react";
    20→import { apiFetch } from "@/lib/api";
    21→import { toast } from "@/hooks/use-toast";
    22→import { formatIDR } from "@/lib/types";
    23→import { cn } from "@/lib/utils";
    24→
    25→/** Pesanan yang akan dibayar via QRIS */
    26→export type QrisPayOrder = { id: string; code: string; total: number };
    27→
    28→type Phase = "scan" | "paid" | "cancelled";
    29→
    30→type QrisInfo = {
    31→  payload: string | null;
    32→  imageUrl: string | null;
    33→  mode: "GATEWAY" | "LOCAL";
    34→  autoVerify: boolean;
    35→  /** gateway QRIS bawaan (sandbox) aktif — tombol simulasi dana ditampilkan */
    36→  sandbox: boolean;
    37→  /** penyedia verifikasi: "casaku" | "midtrans" | "midtrans-sandbox" | "internal" | null */
    38→  provider: string | null;
    39→  /** Alasan penyedia utama (mis. Casaku/Midtrans) ditolak sehingga QR berasal dari fallback uji lokal */
    40→  fallbackReason: string | null;
    41→  /** Galat gateway saat mode LOCAL — ditampilkan jujur (kanal belum aktif, dll.) */
    42→  gatewayError: string | null;
    43→  /** Sandbox Midtrans: tautan simulator uji pembayaran (QR berisi URL uji) */
    44→  sandboxSimulatorUrl: string | null;
    45→  /** Nominal FINAL yang harus dibayar (total + kode unik Casaku bila aktif) */
    46→  payAmount: number | null;
    47→  /** Kode unik yang disematkan Casaku ke nominal (bila aktif) */
    48→  uniqueCode: number | null;
    49→};
    50→
    51→/** Interval polling verifikasi otomatis */
    52→const POLL_MS = 4000;
    53→/** Batas waktu pembayaran QRIS — 5 MENIT sejak pesanan dibuat (permintaan
    54→ *  pemilik). Lewat dari itu pesanan batal otomatis (server membatalkan
    55→ *  saat sweep/poll) dan kode bayar hangus. */
    56→const DURATION_SEC = 5 * 60; // 5 menit
    57→
    58→function mmss(sec: number): string {
    59→  const m = Math.floor(sec / 60);
    60→  const s = sec % 60;
    61→  return `${String(m).padStart(2, "0")}:${String(s).padStart(2, "0")}`;
    62→}
    63→
    64→/**
    65→ * Layar pembayaran QRIS — pelanggan maupun kasir memakai layar yang sama.
    66→ *
    67→ * VERIFIKASI OTOMATIS: setelah QR tampil, layar ini memantau status ke server
    68→ * tiap beberapa detik; server bertanya ke penyedia QRIS "sudah masuk belum
    69→ * dananya?" dan melunaskan pesanan begitu dana masuk. TIDAK ADA tombol
    70→ * "Saya Sudah Bayar" / "Uang Sudah Masuk" — kasir memang tidak boleh
    71→ * (dan tidak bisa) melunaskan pesanan QRIS secara manual.
    72→ */
    73→export function QrisPaySheet({
    74→  order,
    75→  variant,
    76→  onClose,
    77→  onPaid,
    78→  onPrintReceipt,
    79→}: {
    80→  order: QrisPayOrder;
    81→  variant: "customer" | "pos";
    82→  /** Dipanggil saat layar ditutup. result.paid memberi tahu apakah pembayaran
    83→   *  SUDAH lunas saat ditutup (false = belum dibayar / sudah batal) —
    84→   *  pemanggil TIDAK boleh menampilkan pesan sukses bila paid=false. */
    85→  onClose: (result: { paid: boolean }) => void;
    86→  /** dipanggil saat status pembayaran berubah menjadi lunas (mis. refresh daftar) */
    87→  onPaid?: () => void | Promise<void>;
    88→  /** POS: cetak struk setelah lunas (bila disediakan halaman pemanggil) */
    89→  onPrintReceipt?: () => void | Promise<void>;
    90→}) {
    91→  const prefersReduced = useReducedMotion();
    92→  const isPos = variant === "pos";
    93→
    94→  const [info, setInfo] = useState<QrisInfo | null>(null);
    95→  const [loadError, setLoadError] = useState<string | null>(null);
    96→  const [phase, setPhase] = useState<Phase>("scan");
    97→  const [secondsLeft, setSecondsLeft] = useState(DURATION_SEC);
    98→  const [checkingNow, setCheckingNow] = useState(false);
    99→  const [regenerating, setRegenerating] = useState(false);
   100→  /** status terakhir dari penyedia (mis. "EXPIRED") */
   101→  const [gatewayStatus, setGatewayStatus] = useState<string | null>(null);
   102→  /** gangguan sementara saat memantau (diciutkan agar tidak bikin cemas) */
   103→  const [monitorNote, setMonitorNote] = useState<string | null>(null);
   104→  /** batas waktu pesanan (createdAt + 5 menit) dalam ms epoch */
   105→  const [orderDeadline, setOrderDeadline] = useState<number | null>(null);
   106→  const paidNotified = useRef(false);
   107→  /** onPaid disimpan di ref: identitas prop baru dari re-render parent
   108→   *  (mis. jam kasir yang berdetik tiap detik) TIDAK boleh memicu ulang
   109→   *  efek fetch/polling di bawah — itulah penyebab layar QR berkedip. */
   110→  const onPaidRef = useRef(onPaid);
   111→  useEffect(() => {
   112→    onPaidRef.current = onPaid;
   113→  }, [onPaid]);
   114→
   115→  /** Stabil antar-render — aman dipakai sebagai dependensi efek */
   116→  const notifyPaid = useCallback(() => {
   117→    if (paidNotified.current) return;
   118→    paidNotified.current = true;
   119→    void onPaidRef.current?.();
   120→  }, []);
   121→
   122→  /** Penomoran permintaan: respons yang datang terlambat dari fetch lama
   123→   *  tidak boleh menimpa data fetch terbaru (cegah QR "maju-mundur"). */
   124→  const fetchSeq = useRef(0);
   125→  /** Ambil kode QRIS dari server (gateway bila aktif, fallback lokal).
   126→   *  soft=true → QR lama TETAP tampil selagi data baru diambil (kode
   127→   *  langsung tertukar saat respons tiba) sehingga layar tak berkedip. */
   128→  const fetchPayload = useCallback(async (soft = false) => {
   129→    const seq = ++fetchSeq.current;
   130→    setLoadError(null);
   131→    setGatewayStatus(null);
   132→    if (!soft) setInfo(null);
   133→    try {
   134→      const res = await apiFetch(`/api/orders/${order.id}/qris`, { cache: "no-store" });
   135→      const data = await res.json();
   136→      if (seq !== fetchSeq.current) return; // respons kedaluwarsa — abaikan
   137→      if (!res.ok) {
   138→        // Pesanan ternyata sudah lunas → langsung tampilkan sukses
   139→        if (typeof data.error === "string" && data.error.toLowerCase().includes("lunas")) {
   140→          setPhase("paid");
   141→          notifyPaid();
   142→          return;
   143→        }
   144→        // Pesanan dibatalkan otomatis (lewat batas 5 menit) → tampilan batal
   145→        if (data.cancelled === true) {
   146→          setPhase("cancelled");
   147→          return;
   148→        }
   149→        setLoadError(data.error ?? "Gagal memuat kode QRIS.");
   150→        return;
   151→      }
   152→      if (data.paid === true) {
   153→        setPhase("paid");
   154→        notifyPaid();
   155→        return;
   156→      }
   157→      const expiresAt = data.expiresAt ? new Date(data.expiresAt as string) : null;
   158→      const cancelMs = data.cancelAt ? new Date(data.cancelAt as string).getTime() : null;
   159→      if (cancelMs) setOrderDeadline(cancelMs);
   160→      setInfo({
   161→        payload: (data.payload as string) ?? null,
   162→        imageUrl: (data.imageUrl as string) ?? null,
   163→        mode: data.mode === "GATEWAY" ? "GATEWAY" : "LOCAL",
   164→        autoVerify: !!data.autoVerify,
   165→        sandbox: data.sandbox === true,
   166→        provider: typeof data.provider === "string" ? data.provider : null,
   167→        fallbackReason:
   168→          typeof data.fallbackReason === "string" ? data.fallbackReason : null,
   169→        gatewayError:
   170→          typeof data.gatewayError === "string" ? data.gatewayError : null,
   171→        sandboxSimulatorUrl:
   172→          typeof data.sandboxSimulatorUrl === "string" ? data.sandboxSimulatorUrl : null,
   173→        payAmount: typeof data.amount === "number" ? data.amount : null,
   174→        uniqueCode: typeof data.uniqueCode === "number" ? data.uniqueCode : null,
   175→      });
   176→      // Hitung mundur pakai batas TERKETAT: umur kode gateway vs batas pesanan
   177→      const nowMs = Date.now();
   178→      const gatewaySec = expiresAt
   179→        ? Math.max(0, Math.floor((expiresAt.getTime() - nowMs) / 1000))
   180→        : DURATION_SEC;
   181→      const orderSec = cancelMs
   182→        ? Math.max(0, Math.floor((cancelMs - nowMs) / 1000))
   183→        : DURATION_SEC;
   184→      setSecondsLeft(Math.max(0, Math.min(DURATION_SEC, gatewaySec, orderSec)));
   185→    } catch {
   186→      if (seq !== fetchSeq.current) return; // respons kedaluwarsa — abaikan
   187→      setLoadError("Tidak dapat menghubungi server. Periksa koneksi lalu coba lagi.");
   188→    }
   189→    // notifyPaid stabil ([]) — dependensi ini tidak berubah antar-render
   190→    // parent, sehingga QR hanya diambil SEKALI per pesanan (bukan tiap detik).
   191→  }, [order.id, notifyPaid]);
   192→
   193→  useEffect(() => {
   194→    void fetchPayload();
   195→  }, [fetchPayload]);
   196→
   197→  // Hitung waktu mundur (hanya saat fase scan)
   198→  useEffect(() => {
   199→    if (phase !== "scan") return;
   200→    const t = setInterval(() => {
   201→      setSecondsLeft((s) => (s > 0 ? s - 1 : 0));
   202→    }, 1000);
   203→    return () => clearInterval(t);
   204→  }, [phase]);
   205→
   206→  /** Tutup layar pembayaran + notifikasi sesuai hasil pembayaran terakhir
   207→   *  (permintaan pemilik):
   208→   *  - belum lunas  -> "Pembayaran belum berhasil — silakan lanjutkan
   209→   *                    pembayaran" (merah, agar kasir/pelanggan tak bingung)
   210→   *  - sudah lunas  -> "Pembayaran berhasil — pesanan Anda sedang diproses"
   211→   *  - sudah batal  -> cukup tutup (layar batal sudah menjelaskan) */
   212→  const requestClose = useCallback(() => {
   213→    if (phase === "paid") {
   214→      toast({
   215→        title: "Pembayaran berhasil",
   216→        description: `Pesanan ${order.code} ${formatIDR(order.total)} sudah terkonfirmasi — pesanan Anda sedang diproses.`,
   217→        duration: 7000,
   218→      });
   219→    } else if (phase === "scan") {
   220→      toast({
   221→        variant: "destructive",
   222→        title: "Pembayaran belum berhasil",
   223→        description: "Silakan lanjutkan pembayaran QRIS sebelum waktu habis — pesanan belum terlunasi.",
   224→        duration: 7000,
   225→      });
   226→    }
   227→    onClose({ paid: phase === "paid" });
   228→  }, [phase, order.code, order.total, onClose]);
   229→
   230→  // Tutup dengan tombol Escape (lewat requestClose: notifikasi status ikut tampil)
   231→  useEffect(() => {
   232→    const onKey = (e: KeyboardEvent) => {
   233→      if (e.key === "Escape") requestClose();
   234→    };
   235→    window.addEventListener("keydown", onKey);
   236→    return () => window.removeEventListener("keydown", onKey);
   237→  }, [requestClose]);
   238→
   239→  // ===== Pemantauan verifikasi otomatis =====
   240→  // hasInfo (boolean) sebagai dependensi — bukan objek info — agar interval
   241→  // polling TIDAK dibongkar-pasang setiap data baru masuk.
   242→  const hasInfo = info !== null;
   243→  useEffect(() => {
   244→    if (phase !== "scan" || !hasInfo) return;
   245→    let alive = true;
   246→    const poll = async (force = false) => {
   247→      try {
   248→        const res = await apiFetch(
   249→          `/api/orders/${order.id}/qris/status${force ? "?force=1" : ""}`,
   250→          { cache: "no-store" }
   251→        );
   252→        if (!res.ok || !alive) return;
   253→        const data = await res.json();
   254→        if (data.paymentStatus === "PAID") {
   255→          setPhase("paid");
   256→          notifyPaid();
   257→          return;
   258→        }
   259→        // Pesanan batal otomatis (lewat batas 5 menit / dana telat)
   260→        if (data.cancelled === true || data.status === "CANCELLED") {
   261→          setMonitorNote((data.notice as string) ?? null);
   262→          setPhase("cancelled");
   263→          return;
   264→        }
   265→        if (data.cancelAt) setOrderDeadline(new Date(data.cancelAt as string).getTime());
   266→        setGatewayStatus((data.gatewayStatus as string) ?? null);
   267→        setMonitorNote((data.gatewayError as string) ?? null);
   268→      } catch {
   269→        // koneksi tersendat — coba lagi pada tick berikutnya
   270→      }
   271→    };
   272→    void poll();
   273→    const t = setInterval(() => void poll(), POLL_MS);
   274→    return () => {
   275→      alive = false;
   276→      clearInterval(t);
   277→    };
   278→  }, [phase, hasInfo, order.id, notifyPaid]);
   279→
   280→  /** POS: paksa sistem memeriksa penyedia sekarang (tetap otomatis, bukan menandai lunas) */
   281→  const checkNow = async () => {
   282→    if (checkingNow) return;
   283→    setCheckingNow(true);
   284→    try {
   285→      const res = await apiFetch(`/api/orders/${order.id}/qris/status?force=1`, {
   286→        cache: "no-store",
   287→      });
   288→      const data = await res.json();
   289→      if (res.ok && data.paymentStatus === "PAID") {
   290→        setPhase("paid");
   291→        notifyPaid();
   292→      }
   293→    } catch {
   294→      /* abaikan — polling reguler akan menyusul */
   295→    } finally {
   296→      setCheckingNow(false);
   297→    }
   298→  };
   299→
   300→  /** Buat ulang kode bayar (transaksi lama kedaluwarsa) — soft: QR lama
   301→   *  tetap tampil sampai kode baru siap, tanpa kedipan spinner. */
   302→  const regenerate = async () => {
   303→    if (regenerating) return;
   304→    setRegenerating(true);
   305→    await fetchPayload(true);
   306→    setRegenerating(false);
   307→  };
   308→
   309→  const expired = secondsLeft <= 0;
   310→  const codeStale = expired || gatewayStatus === "EXPIRED";
   311→  const autoVerify = info?.autoVerify ?? false;
   312→  /** Batas pesanan (5 menit) sudah terlewat — pesanan mati, jangan tawarkan buat ulang kode */
   313→  const pastDeadline = orderDeadline !== null && Date.now() >= orderDeadline;
   314→
   315→  // Portal ke <body> agar modal tidak terjebak stacking context `main` (z-10)
   316→  // — selalu tampil DI ATAS navigasi bawah liquid glass (z-50) & header (z-40).
   317→  return createPortal(
   318→    <AnimatePresence>
   319→      <motion.div
   320→        key="qris-overlay"
   321→        initial={prefersReduced ? false : { opacity: 0 }}
   322→        animate={{ opacity: 1 }}
   323→        exit={prefersReduced ? undefined : { opacity: 0 }}
   324→        className="fixed inset-0 z-[95] flex items-end justify-center bg-neutral-900/70 backdrop-blur-sm sm:items-center sm:p-4"
   325→        role="dialog"
   326→        aria-modal="true"
   327→        aria-label="Pembayaran QRIS"
   328→      >
   329→        <motion.div
   330→          initial={prefersReduced ? false : { y: 48, scale: 0.97 }}
   331→          animate={{ y: 0, scale: 1 }}
   332→          exit={prefersReduced ? undefined : { y: 48, opacity: 0 }}
   333→          transition={{ type: "spring", stiffness: 380, damping: 34 }}
   334→          className="flex max-h-[94dvh] w-full max-w-md flex-col overflow-hidden rounded-t-3xl bg-white shadow-2xl sm:rounded-3xl"
   335→        >
   336→          {/* ===== KEPALA ===== */}
   337→          <header className="relative shrink-0 bg-gradient-to-r from-orange-700 via-orange-600 to-orange-500 px-5 py-4 text-white">
   338→            <div className="flex items-center gap-3">
   339→              <span className="flex h-10 w-10 items-center justify-center rounded-2xl bg-white/20 ring-1 ring-white/30">
   340→                <ScanLine className="h-5 w-5" />
   341→              </span>
   342→              <div className="min-w-0 flex-1">
   343→                <p className="text-sm font-black tracking-wide">Pembayaran QRIS</p>
   344→                <p className="truncate text-[11px] font-semibold text-orange-100">
   345→                  {order.code} • {isPos ? "Kasir" : "Aplikasi"}
   346→                </p>
   347→              </div>
   348→              <button
   349→                onClick={requestClose}
   350→                aria-label="Tutup"
   351→                className="press-effect flex h-9 w-9 items-center justify-center rounded-full bg-white/15 hover:bg-white/30"
   352→              >
   353→                <X className="h-4.5 w-4.5" />
   354→              </button>
   355→            </div>
   356→          </header>
   357→
   358→          {/* ===== ISI ===== */}
   359→          <div className="scrollbar-slim overflow-y-auto px-5 pb-[max(1.25rem,env(safe-area-inset-bottom))] pt-4">
   360→            {phase === "paid" ? (
   361→              /* --------- BERHASIL (verifikasi otomatis) --------- */
   362→              <div className="flex flex-col items-center gap-3 py-8 text-center">
   363→                <motion.span
   364→                  initial={prefersReduced ? false : { scale: 0.5, opacity: 0 }}
   365→                  animate={{ scale: 1, opacity: 1 }}
   366→                  transition={{ type: "spring", stiffness: 300, damping: 18 }}
   367→                  className="flex h-20 w-20 items-center justify-center rounded-full bg-emerald-50 ring-4 ring-emerald-100"
   368→                >
   369→                  <BadgeCheck className="h-11 w-11 text-emerald-600" />
   370→                </motion.span>
   371→                <h2 className="text-lg font-black text-neutral-900">Pembayaran Diterima! 🎉</h2>
   372→                <p className="max-w-[300px] text-xs leading-relaxed text-neutral-500">
   373→                  {formatIDR(order.total)} untuk pesanan {order.code} sudah lunas —
   374→                  terverifikasi <strong className="font-bold text-emerald-600">otomatis oleh sistem</strong>{" "}
   375→                  saat dana masuk. Geprekmu segera dimasak ya!
   376→                </p>
   377→                {isPos && onPrintReceipt ? (
   378→                  <div className="mt-2 flex w-full items-center gap-2">
   379→                    <button
   380→                      onClick={requestClose}
   381→                      className="press-effect h-11 flex-1 rounded-xl border border-neutral-200 bg-white text-sm font-extrabold text-neutral-600 hover:bg-neutral-50"
   382→                    >
   383→                      Selesai
   384→                    </button>
   385→                    <button
   386→                      onClick={() => void onPrintReceipt()}
   387→                      className="press-effect flex h-11 flex-[1.6] items-center justify-center gap-2 rounded-xl bg-orange-600 text-sm font-extrabold text-white shadow-lg shadow-orange-500/30 hover:bg-orange-700"
   388→                    >
   389→                      <Printer className="h-4 w-4" /> Cetak Struk
   390→                    </button>
   391→                  </div>
   392→                ) : (
   393→                  <button
   394→                    onClick={requestClose}
   395→                    className="press-effect mt-2 w-full rounded-xl bg-orange-600 py-3 text-sm font-extrabold text-white shadow-lg shadow-orange-500/30 hover:bg-orange-700"
   396→                  >
   397→                    Selesai
   398→                  </button>
   399→                )}
   400→              </div>
   401→            ) : phase === "cancelled" ? (
   402→              /* --------- DIBATALKAN OTOMATIS (belum bayar 5 menit) --------- */
   403→              <div className="flex flex-col items-center gap-3 py-8 text-center">
   404→                <motion.span
   405→                  initial={prefersReduced ? false : { scale: 0.5, opacity: 0 }}
   406→                  animate={{ scale: 1, opacity: 1 }}
   407→                  transition={{ type: "spring", stiffness: 300, damping: 18 }}
   408→                  className="flex h-20 w-20 items-center justify-center rounded-full bg-red-50 ring-4 ring-red-100"
   409→                >
   410→                  <XCircle className="h-11 w-11 text-red-500" />
   411→                </motion.span>
   412→                <h2 className="text-lg font-black text-neutral-900">Pesanan Dibatalkan Otomatis</h2>
   413→                <p className="max-w-[320px] text-xs leading-relaxed text-neutral-500">
   414→                  Pembayaran QRIS untuk pesanan <strong className="font-bold text-neutral-700">{order.code}</strong>{" "}
   415→                  tidak diterima dalam <strong className="font-bold text-red-500">5 menit</strong>, jadi
   416→                  pesanan dibatalkan. Voucher yang terpakai sudah dikembalikan ke akunmu.
   417→                </p>
   418→                {monitorNote && (
   419→                  <p className="max-w-[320px] rounded-xl bg-amber-50 p-2.5 text-[11px] font-semibold leading-snug text-amber-700 ring-1 ring-amber-200">
   420→                    {monitorNote}
   421→                  </p>
   422→                )}
   423→                <p className="text-[11px] text-neutral-400">
   424→                  Masih pengin? Buat pesanan baru — dapur selalu siap menggeprek 😊
   425→                </p>
   426→                <button
   427→                  onClick={() => onClose({ paid: false })}
   428→                  className="press-effect mt-2 w-full rounded-xl bg-orange-600 py-3 text-sm font-extrabold text-white shadow-lg shadow-orange-500/30 hover:bg-orange-700"
   429→                >
   430→                  Tutup
   431→                </button>
   432→              </div>
   433→            ) : (
   434→              /* --------- PINDAI QR --------- */
   435→              <>
   436→                <div className="flex items-end justify-between gap-3">
   437→                  <div>
   438→                    <p className="text-[11px] font-bold uppercase tracking-wide text-neutral-400">
   439→                      Total Bayar
   440→                    </p>
   441→                    {/* Nominal FINAL (total + kode unik Casaku bila aktif) —
   442→                        pelanggan wajib membayar nominal ini PERSIS agar
   443→                        verifikasi otomatis mencocokkan dananya. */}
   444→                    <p className="text-2xl font-black text-orange-600">
   445→                      {formatIDR(info?.payAmount ?? order.total)}
   446→                    </p>
   447→                    {info?.uniqueCode != null && info.uniqueCode > 0 ? (
   448→                      <p className="mt-1 rounded-lg bg-amber-50 px-2 py-1 text-[10px] font-bold leading-snug text-amber-700 ring-1 ring-amber-200">
   449→                        Termasuk kode unik {formatIDR(info.uniqueCode)} — bayar PERSIS
   450→                        nominal di atas ({formatIDR(order.total)} + {formatIDR(info.uniqueCode)})
