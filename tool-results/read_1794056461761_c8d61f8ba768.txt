    33→/** Kompres & perkecil gambar banner di sisi klien (maks 1200px, JPEG) */
    34→async function fileToResizedDataUrl(file: File): Promise<string> {
    35→  const dataUrl = await new Promise<string>((resolve, reject) => {
    36→    const r = new FileReader();
    37→    r.onload = () => resolve(r.result as string);
    38→    r.onerror = () => reject(new Error("Gagal membaca file."));
    39→    r.readAsDataURL(file);
    40→  });
    41→  const img = await new Promise<HTMLImageElement>((resolve, reject) => {
    42→    const i = new Image();
    43→    i.onload = () => resolve(i);
    44→    i.onerror = () => reject(new Error("File bukan gambar yang valid."));
    45→    i.src = dataUrl;
    46→  });
    47→  const maxDim = 1200;
    48→  const scale = Math.min(1, maxDim / Math.max(img.width, img.height));
    49→  if (scale >= 1 && dataUrl.length < 250_000) return dataUrl;
    50→  const canvas = document.createElement("canvas");
    51→  canvas.width = Math.max(1, Math.round(img.width * scale));
    52→  canvas.height = Math.max(1, Math.round(img.height * scale));
    53→  const ctx = canvas.getContext("2d");
    54→  if (!ctx) return dataUrl;
    55→  ctx.drawImage(img, 0, 0, canvas.width, canvas.height);
    56→  return canvas.toDataURL("image/jpeg", 0.85);
    57→}
