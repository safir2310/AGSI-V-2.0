     1→"use client";
     2→
     3→import { useCallback, useEffect, useMemo, useRef, useState } from "react";
     4→import { AnimatePresence, motion, useReducedMotion } from "framer-motion";
     5→import {
     6→  Bike,
     7→  BadgeCheck,
     8→  ClipboardList,
     9→  Clock,
    10→  Coins,
    11→  Loader2,
    12→  LogOut,
    13→  Minus,
    14→  MonitorSmartphone,
    15→  Plus,
    16→  Receipt,
    17→  RefreshCw,
    18→  ScanLine,
    19→  Search,
    20→  ShoppingBag,
    21→  Trash2,
    22→  Utensils,
    23→  Wifi,
    24→  X,
    25→} from "lucide-react";
    26→import { Input } from "@/components/ui/input";
    27→import { ChickenBadge } from "@/components/logo/ChickenLogo";
    28→import { ProductThumb } from "@/components/shared/ProductThumb";
    29→import { QrisPaySheet, type QrisPayOrder } from "@/components/shared/QrisPaySheet";
    30→import { ReceiptSheet } from "@/components/shared/ReceiptSheet";
    31→import { refreshOrders, useProductsStore, useSettingsStore } from "@/lib/hooks";
    32→import { useNoticeStore, type AppView } from "@/lib/store";
    33→import { apiFetch } from "@/lib/api";
    34→import { formatIDR, type AppUser, type OrderData, type Product } from "@/lib/types";
    35→import { cn } from "@/lib/utils";
    36→
    37→type PosItem = { product: Product; qty: number };
    38→
    39→export function PosView({ user, go, onLogout }: { user: AppUser; go: (v: AppView) => void; onLogout: () => void }) {
    40→  const prefersReduced = useReducedMotion();
    41→  const showNotice = useNoticeStore((s) => s.show);
    42→  const { products, load } = useProductsStore();
    43→  const [query, setQuery] = useState("");
    44→  /** Popup hasil pencarian menu di bawah input cari (autocompletion) */
    45→  const [searchOpen, setSearchOpen] = useState(false);
    46→  const searchBoxRef = useRef<HTMLDivElement | null>(null);
    47→  const [category, setCategory] = useState("Semua");
    48→  /** Mode bar pencarian di bawah header: scan barcode member ATAU cari nama produk */
    49→  const [scanMode, setScanMode] = useState<"member" | "product">("product");
    50→  const [items, setItems] = useState<PosItem[]>([]);
    51→  const [customerName, setCustomerName] = useState("");
    52→  const [tableNo, setTableNo] = useState("");
    53→  /** Kode member (hasil scan barcode kartu member / ketikan kasir) */
    54→  const [memberCode, setMemberCode] = useState("");
    55→  const [memberInfo, setMemberInfo] = useState<{ name: string; memberCode: string; points: number } | null>(null);
    56→  const [memberChecking, setMemberChecking] = useState(false);
    57→  const [orderType, setOrderType] = useState<"DINE_IN" | "TAKEAWAY" | "DELIVERY">("DINE_IN");
    58→  const [payment, setPayment] = useState<"CASH" | "QRIS">("CASH");
    59→  // Saklar layanan toko (Profil Admin → Pengaturan Toko):
    60→  // nonaktif → tombol "Antar" / "QRIS" disembunyikan & pilihan lama di-reset
    61→  const deliveryEnabled = useSettingsStore((s) => s.deliveryEnabled);
    62→  const qrisEnabled = useSettingsStore((s) => s.qrisEnabled);
    63→
    64→  // Layanan dimatikan admin → pilihan lama otomatis dikembalikan
    65→  useEffect(() => {
    66→    if (!deliveryEnabled) {
    67→      setOrderType((t) => (t === "DELIVERY" ? "DINE_IN" : t));
    68→    }
    69→    if (!qrisEnabled) {
    70→      setPayment((p) => (p === "QRIS" ? "CASH" : p));
    71→    }
    72→  }, [deliveryEnabled, qrisEnabled]);
    73→  const [submitting, setSubmitting] = useState(false);
    74→  const [clock, setClock] = useState("");
    75→  const [lastOrder, setLastOrder] = useState<{
    76→    id: string;
    77→    code: string;
    78→    total: number;
    79→    name: string;
    80→    paymentMethod: string;
    81→    paymentStatus: string;
    82→  } | null>(null);
    83→  /** Transaksi QRIS yang sedang dibayar (QR tampil di layar kasir) */
    84→  const [payOrder, setPayOrder] = useState<QrisPayOrder | null>(null);
    85→  /** Pesanan yang struknya sedang dibuka (cetak / simpan bernama kode struk) */
    86→  const [receiptOrder, setReceiptOrder] = useState<OrderData | null>(null);
    87→
    88→  // Jam operasional live
    89→  useEffect(() => {
    90→    const tick = () =>
    91→      setClock(
    92→        new Date().toLocaleTimeString("id-ID", { hour: "2-digit", minute: "2-digit", second: "2-digit" })
    93→      );
    94→    tick();
    95→    const t = setInterval(tick, 1000);
    96→    return () => clearInterval(t);
    97→  }, []);
    98→
    99→  useEffect(() => {
   100→    load(true);
   101→  }, [load]);
   102→
   103→  const categories = useMemo(
   104→    () => ["Semua", ...Array.from(new Set(products.map((p) => p.category)))],
   105→    [products]
   106→  );
   107→
   108→  const q = query.trim().toLowerCase();
   109→  const filtered = products.filter((p) => {
   110→    if (!p.available) return false;
   111→    // Sedang mencari → hasil dari SEMUA kategori (saringan kategori dilewati),
   112→    // cocokkan nama produk ATAU nama kategorinya
   113→    if (q) {
   114→      return (
   115→        p.name.toLowerCase().includes(q) || p.category.toLowerCase().includes(q)
   116→      );
   117→    }
   118→    return category === "Semua" || p.category === category;
   119→  });
   120→
   121→  /** Menu yang tampil di popup pencarian (maks. 6 supaya ringkas) */
   122→  const POPUP_MAX = 6;
   123→  const popupMatches = q ? filtered.slice(0, POPUP_MAX) : [];
   124→
   125→  /** Tutup popup saat klik di luar kotak pencarian atau tekan Escape */
   126→  useEffect(() => {
   127→    if (!searchOpen) return;
   128→    const onPointerDown = (e: PointerEvent) => {
   129→      if (searchBoxRef.current && !searchBoxRef.current.contains(e.target as Node)) {
   130→        setSearchOpen(false);
   131→      }
   132→    };
   133→    const onKey = (e: KeyboardEvent) => {
   134→      if (e.key === "Escape") setSearchOpen(false);
   135→    };
   136→    window.addEventListener("pointerdown", onPointerDown);
   137→    window.addEventListener("keydown", onKey);
   138→    return () => {
   139→      window.removeEventListener("pointerdown", onPointerDown);
   140→      window.removeEventListener("keydown", onKey);
   141→    };
   142→  }, [searchOpen]);
   143→
   144→  const addItem = (p: Product) =>
   145→    setItems((cur) => {
   146→      const found = cur.find((i) => i.product.id === p.id);
   147→      if (found) return cur.map((i) => (i.product.id === p.id ? { ...i, qty: i.qty + 1 } : i));
   148→      return [...cur, { product: p, qty: 1 }];
   149→    });
   150→
   151→  /** Pilih menu dari popup: masuk keranjang, bersihkan pencarian, tutup popup */
   152→  const pickFromPopup = (p: Product) => {
   153→    addItem(p);
   154→    setQuery("");
   155→    setSearchOpen(false);
   156→  };
   157→
   158→  const changeQty = (id: string, delta: number) =>
   159→    setItems((cur) =>
   160→      cur
   161→        .map((i) => (i.product.id === id ? { ...i, qty: i.qty + delta } : i))
   162→        .filter((i) => i.qty > 0)
   163→    );
   164→
   165→  const subtotal = items.reduce((s, i) => s + i.product.price * i.qty, 0);
   166→  const totalItems = items.reduce((s, i) => s + i.qty, 0);
   167→
   168→  /** Banyak jenis menu di keranjang (≥7) → baris keranjang otomatis jadi
   169→   *  RINGKAS & TRANSPARAN (kaca) supaya daftar tetap ringan digulir dan
   170→   *  kasir tetap fokus ke total & tombol bayar saat transaksi ramai */
   171→  const compactCart = items.length > 6;
   172→
   173→  /** Mode scroll daftar keranjang: gradasi atas/bawah menandai daftar bisa
   174→   *  digulir + item yang BARU ditambahkan langsung digulirkan ke layar */
   175→  const listRef = useRef<HTMLDivElement | null>(null);
   176→  const [fade, setFade] = useState({ top: false, bottom: false });
   177→  const updateFades = useCallback(() => {
   178→    const el = listRef.current;
   179→    if (!el) return;
   180→    setFade({
   181→      top: el.scrollTop > 4,
   182→      bottom: el.scrollTop + el.clientHeight < el.scrollHeight - 4,
   183→    });
   184→  }, []);
   185→  const prevRows = useRef(0);
   186→  useEffect(() => {
   187→    const el = listRef.current;
   188→    // Jenis menu BARU masuk keranjang → gulir ke bawah agar langsung terlihat
   189→    if (items.length > prevRows.current && el) {
   190→      el.scrollTo({
   191→        top: el.scrollHeight,
   192→        behavior: prefersReduced ? "auto" : "smooth",
   193→      });
   194→    }
   195→    prevRows.current = items.length;
   196→    updateFades();
   197→  }, [items, prefersReduced, updateFades]);
   198→
   199→  /** Cari member dari kode (dipakai saat scan barcode / ketik lalu Enter) */
   200→  const lookupMember = async (raw?: string) => {
   201→    const code = (raw ?? memberCode).replace(/\D/g, "");
   202→    if (!code) {
   203→      setMemberInfo(null);
   204→      return;
   205→    }
   206→    setMemberChecking(true);
   207→    try {
   208→      const res = await apiFetch(`/api/member/lookup?code=${encodeURIComponent(code)}`, {
   209→        cache: "no-store",
   210→      });
   211→      const data = await res.json();
   212→      if (!res.ok) {
   213→        setMemberInfo(null);
   214→        showNotice({ type: "error", title: "Member Tidak Ditemukan", message: data.error ?? "Periksa kode member." });
   215→        return;
   216→      }
   217→      setMemberInfo(data.member);
   218→    } catch {
   219→      setMemberInfo(null);
   220→      showNotice({ type: "error", title: "Gagal Memeriksa", message: "Tidak dapat menghubungi server." });
   221→    } finally {
   222→      setMemberChecking(false);
   223→    }
   224→  };
   225→
   226→  const clearMember = () => {
   227→    setMemberCode("");
   228→    setMemberInfo(null);
   229→  };
   230→
   231→  /** Ambil detail pesanan lengkap lalu tampilkan sheet struk (cetak/simpan
   232→   *  dengan nama file = kode struk). Mengembalikan true bila berhasil dibuka. */
   233→  const openReceipt = useCallback(async (orderId: string): Promise<boolean> => {
   234→    try {
   235→      const res = await apiFetch(`/api/orders/${orderId}`, { cache: "no-store" });
   236→      const data = await res.json();
   237→      if (!res.ok || !data.order) return false;
   238→      setReceiptOrder(data.order as OrderData);
   239→      return true;
   240→    } catch {
   241→      return false;
   242→    }
   243→  }, []);
   244→
   245→  const submit = async () => {
   246→    if (items.length === 0) return;
   247→    setSubmitting(true);
   248→    try {
   249→      const res = await apiFetch("/api/orders", {
   250→        method: "POST",
   251→        headers: { "Content-Type": "application/json" },
   252→        body: JSON.stringify({
   253→          items: items.map((i) => ({ productId: i.product.id, qty: i.qty })),
   254→          customerName: customerName.trim() || `Walk-in${tableNo ? ` Meja ${tableNo}` : ""}`,
   255→          memberCode: memberInfo ? memberInfo.memberCode : undefined,
   256→          type: orderType,
   257→          tableNo: orderType === "DINE_IN" ? tableNo : undefined,
   258→          paymentMethod: payment,
   259→          source: "POS",
   260→        }),
   261→      });
   262→      const data = await res.json();
   263→      if (!res.ok) {
   264→        showNotice({ type: "error", title: "Transaksi Gagal", message: data.error ?? "Coba lagi." });
   265→        return;
   266→      }
   267→      setLastOrder({
   268→        id: data.order.id,
   269→        code: data.order.code,
   270→        total: data.order.total,
   271→        name: data.order.customerName,
   272→        paymentMethod: data.order.paymentMethod,
   273→        paymentStatus: data.order.paymentStatus,
   274→      });
   275→      setItems([]);
   276→      setCustomerName("");
   277→      setTableNo("");
   278→      clearMember();
   279→      refreshOrders("all");
   280→
   281→      // QRIS: tampilkan QR di layar kasir agar pelanggan bisa memindai.
   282→      // Sistem yang melunaskan otomatis saat dana masuk (webhook/polling) —
   283→      // kasir TIDAK menandai lunas.
   284→      if (
   285→        data.order.paymentMethod === "QRIS" &&
   286→        data.order.paymentStatus !== "PAID" &&
   287→        data.order.total > 0
   288→      ) {
   289→        setPayOrder({ id: data.order.id, code: data.order.code, total: data.order.total });
   290→        return;
   291→      }
   292→
   293→      // Tunai — uang sudah diterima langsung di kasir. Struk dibuka seketika
   294→      // agar bisa dicetak / disimpan (sesuai label tombol "Bayar & Cetak");
   295→      // bila gagal dimuat, cukup pemberitahuan sukses biasa.
   296→      const receiptOpened = await openReceipt(data.order.id);
   297→      if (!receiptOpened) {
   298→        showNotice({
   299→          type: "success",
   300→          title: "Transaksi Berhasil ✓",
   301→          message: `Pesanan ${data.order.code} (${formatIDR(data.order.total)}) telah dicatat dan masuk antrean dapur.`,
   302→        });
   303→      }
   304→    } catch {
   305→      showNotice({ type: "error", title: "Koneksi Bermasalah", message: "Periksa jaringan lalu ulangi." });
   306→    } finally {
   307→      setSubmitting(false);
   308→    }
   309→  };
   310→
   311→  return (
   312→    <div className="flex h-dvh flex-col bg-orange-50/40 pb-[env(safe-area-inset-bottom,0px)] pt-[env(safe-area-inset-top,0px)]">
   313→      {/* ===== TOOLBAR OPERASIONAL POS ===== */}
   314→      <header className="z-30 shrink-0 bg-gradient-to-r from-orange-700 via-orange-600 to-orange-500 text-white shadow-xl">
   315→        <div className="flex flex-wrap items-center gap-x-4 gap-y-2 px-4 py-2.5 lg:px-6">
   316→          <div className="flex items-center gap-2.5">
   317→            <ChickenBadge size={38} />
   318→            <div className="leading-tight">
   319→              <p className="text-sm font-black uppercase tracking-wide">POS Kasir</p>
   320→              <p className="text-[10px] font-semibold text-orange-100">Ayam Geprek Sambal Ijo</p>
   321→            </div>
   322→          </div>
   323→
   324→          <div className="hidden items-center gap-1.5 rounded-full bg-white/15 px-3 py-1 text-[11px] font-bold md:flex">
   325→            <Clock className="h-3.5 w-3.5" /> <span className="tabular-nums">{clock}</span>
   326→          </div>
   327→          <div className="hidden items-center gap-1.5 rounded-full bg-white/15 px-3 py-1 text-[11px] font-bold md:flex">
   328→            <Wifi className="h-3.5 w-3.5" /> Online
   329→          </div>
   330→          <div className="flex items-center gap-1.5 rounded-full bg-white/15 px-3 py-1 text-[11px] font-bold">
   331→            👤 {user.name}
   332→          </div>
   333→
   334→          <div className="ml-auto flex items-center gap-2">
   335→            {lastOrder && (
   336→              <button
   337→                onClick={() => {
   338→                  // Pesanan QRIS belum lunas -> buka lagi layar QR (pantau otomatis)
   339→                  if (
   340→                    lastOrder.paymentMethod === "QRIS" &&
   341→                    lastOrder.paymentStatus !== "PAID" &&
   342→                    lastOrder.total > 0
   343→                  ) {
   344→                    setPayOrder({
   345→                      id: lastOrder.id,
   346→                      code: lastOrder.code,
   347→                      total: lastOrder.total,
   348→                    });
   349→                    return;
   350→                  }
   351→                  // Sudah lunas (tunai/QRIS) -> buka struk untuk dicetak/diunduh
   352→                  void openReceipt(lastOrder.id);
   353→                }}
   354→                className="press-effect flex items-center gap-1.5 rounded-full bg-white/15 px-3.5 py-1.5 text-xs font-bold hover:bg-white/25"
   355→                title={
   356→                  lastOrder.paymentMethod === "QRIS" && lastOrder.paymentStatus !== "PAID"
   357→                    ? "Buka lagi layar QRIS (memantau otomatis)"
   358→                    : "Buka struk terakhir — cetak / simpan"
   359→                }
   360→              >
   361→                <Receipt className="h-4 w-4" /> {lastOrder.code}
   362→              </button>
   363→            )}
   364→            {(user.role === "ADMIN" || user.role === "OWNER") && (
   365→              <button
   366→                onClick={() => go("dashboard")}
   367→                className="press-effect flex items-center gap-1.5 rounded-full bg-white/15 px-3.5 py-1.5 text-xs font-bold hover:bg-white/25"
   368→              >
   369→                <MonitorSmartphone className="h-4 w-4" /> Mode Admin
   370→              </button>
   371→            )}
   372→            <button
   373→              onClick={() => load(true)}
   374→              aria-label="Muat ulang produk"
   375→              className="press-effect flex h-8 w-8 items-center justify-center rounded-full bg-white/15 hover:bg-white/25"
   376→            >
   377→              <RefreshCw className="h-4 w-4" />
   378→            </button>
   379→            <button
   380→              onClick={onLogout}
   381→              className="press-effect flex items-center gap-1.5 rounded-full bg-white/15 px-3.5 py-1.5 text-xs font-bold hover:bg-white/25"
   382→            >
   383→              <LogOut className="h-4 w-4" /> Keluar
   384→            </button>
   385→          </div>
   386→        </div>
   387→      </header>
   388→
   389→      {/* ===== BAR SCAN BARCODE & PENCARIAN (di bawah header) =====
   390→          Mode 1: Scan Member — scan/ketik kode member (8 digit) → member terpasang
   391→          Mode 2: Nama Produk — saring katalog menu berdasarkan nama produk */}
   392→      <div className="z-20 shrink-0 border-b border-orange-100 bg-white px-4 py-2.5 shadow-sm lg:px-5">
   393→        <div className="flex flex-wrap items-center gap-2">
   394→          {/* Pemilih mode */}
   395→          <div
   396→            role="tablist"
   397→            aria-label="Mode pencarian"
   398→            className="flex shrink-0 items-center gap-0.5 rounded-xl bg-orange-50 p-1 ring-1 ring-orange-200"
   399→          >
   400→            <button
   401→              role="tab"
   402→              aria-selected={scanMode === "member"}
   403→              onClick={() => setScanMode("member")}
   404→              className={cn(
   405→                "press-effect flex items-center gap-1.5 rounded-lg px-3 py-1.5 text-[11px] font-extrabold transition",
   406→                scanMode === "member"
   407→                  ? "bg-orange-600 text-white shadow-md"
   408→                  : "text-orange-700 hover:bg-orange-100"
   409→              )}
   410→            >
   411→              <ScanLine className="h-3.5 w-3.5" /> Scan Member
   412→            </button>
   413→            <button
   414→              role="tab"
   415→              aria-selected={scanMode === "product"}
   416→              onClick={() => setScanMode("product")}
   417→              className={cn(
   418→                "press-effect flex items-center gap-1.5 rounded-lg px-3 py-1.5 text-[11px] font-extrabold transition",
   419→                scanMode === "product"
   420→                  ? "bg-orange-600 text-white shadow-md"
   421→                  : "text-orange-700 hover:bg-orange-100"
   422→              )}
   423→            >
   424→              <Search className="h-3.5 w-3.5" /> Nama Produk
   425→            </button>
   426→          </div>
   427→
   428→          {/* Input sesuai mode */}
   429→          {scanMode === "member" ? (
   430→            <form
   431→              className="relative min-w-[220px] flex-1"
   432→              onSubmit={(e) => {
   433→                e.preventDefault();
   434→                void lookupMember();
   435→              }}
   436→            >
   437→              <ScanLine className="pointer-events-none absolute left-3 top-1/2 h-4 w-4 -translate-y-1/2 text-orange-400" />
   438→              <Input
   439→                inputMode="numeric"
   440→                autoFocus
   441→                placeholder="Scan / ketik kode member (8 digit), Enter…"
   442→                className="border-orange-200 pl-9 pr-24 focus-visible:ring-orange-400"
   443→                value={memberCode}
   444→                disabled={memberChecking}
   445→                onChange={(e) => {
   446→                  const v = e.target.value.replace(/\D/g, "").slice(0, 12);
   447→                  setMemberCode(v);
   448→                  // Kode member selalu 8 digit → cari otomatis begitu genap 8 angka
   449→                  if (v.length === 8 && !memberInfo) void lookupMember(v);
   450→                }}
   451→              />
   452→              <div className="absolute right-2 top-1/2 flex -translate-y-1/2 items-center gap-1.5">
   453→                {memberChecking && <Loader2 className="h-3.5 w-3.5 animate-spin text-orange-500" />}
   454→                {memberCode && !memberInfo && (
   455→                  <button
   456→                    type="button"
   457→                    onClick={() => setMemberCode("")}
   458→                    aria-label="Bersihkan kode"
   459→                    className="press-effect flex h-6 w-6 items-center justify-center rounded-full text-neutral-400 hover:bg-orange-50 hover:text-neutral-600"
   460→                  >
   461→                    <X className="h-3.5 w-3.5" />
   462→                  </button>
   463→                )}
   464→              </div>
   465→            </form>
   466→          ) : (
   467→            <div ref={searchBoxRef} className="relative min-w-[220px] flex-1">
   468→              <Search className="pointer-events-none absolute left-3 top-1/2 h-4 w-4 -translate-y-1/2 text-orange-400" />
   469→              <Input
   470→                autoFocus
   471→                placeholder="Cari menu dengan nama produk… (cth: geprek, mozarella)"
   472→                className="border-orange-200 pl-9 pr-9 focus-visible:ring-orange-400"
   473→                value={query}
   474→                onChange={(e) => {
   475→                  setQuery(e.target.value);
   476→                  setSearchOpen(true);
   477→                }}
   478→                onFocus={() => {
   479→                  if (query.trim()) setSearchOpen(true);
   480→                }}
   481→                onKeyDown={(e) => {
   482→                  // Enter = langsung masukkan hasil teratas ke keranjang
   483→                  if (e.key === "Enter" && popupMatches.length > 0) {
   484→                    e.preventDefault();
   485→                    pickFromPopup(popupMatches[0]);
   486→                  }
   487→                }}
   488→              />
   489→              {query && (
   490→                <button
   491→                  onClick={() => {
   492→                    setQuery("");
   493→                    setSearchOpen(false);
