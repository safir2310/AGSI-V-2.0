     1→"use client";
     2→

     4→import { motion, useReducedMotion } from "framer-motion";
     5→import {
     6→  Bike,

    24→import { QrisPaySheet, type QrisPayOrder } from "@/components/shared/QrisPaySheet";

    26→import { useCartStore, useNoticeStore, type AppView } from "@/lib/store";
    27→import { apiFetch } from "@/lib/api";
    28→import { refreshOrders, refreshVouchers, useProductsStore, useVouchersStore } from "@/lib/hooks";

    35→} from "@/lib/types";

    48→  findDeliveryArea,
    49→  formatDistance,
    50→} from "@/lib/delivery";
    51→
    52→const ORDER_TYPES = [
    53→  { id: "DINE_IN", label: "Makan di Tempat", icon: Utensils, placeholderTable: true },
    54→  { id: "TAKEAWAY", label: "Bungkus / Ambil", icon: ShoppingBag, placeholderTable: false },
    55→  { id: "DELIVERY", label: "Antar ke Alamat", icon: Bike, placeholderTable: false },
    56→] as const;
    57→
    58→const PAYMENTS = [
    59→  { id: "CASH", label: "Tunai", emoji: "💵" },
    60→  { id: "QRIS", label: "QRIS", emoji: "📱" },
    61→] as const;
    62→
    63→export function CartView({ user, go }: { user: AppUser; go: (v: AppView) => void }) {
    64→  const cart = useCartStore();
    65→  const showNotice = useNoticeStore((s) => s.show);
    66→  const prefersReduced = useReducedMotion();
    67→  const products = useProductsStore((s) => s.products);
    68→  const vouchers = useVouchersStore((s) => s.vouchers);
    69→  const loadVouchers = useVouchersStore((s) => s.load);
    70→  const pendingCode = useVouchersStore((s) => s.pendingCode);
    71→  const setPendingCode = useVouchersStore((s) => s.setPendingCode);
    72→  const [submitting, setSubmitting] = useState(false);
    73→  /** Pesanan QRIS yang sedang dibayar (layar QR terbuka) */
    74→  const [payOrder, setPayOrder] = useState<QrisPayOrder | null>(null);
    75→  /** Popup pilih alamat pengantaran (area Kec. Pidie 24151) */
    76→  const [areaOpen, setAreaOpen] = useState(false);
    77→  const [form, setForm] = useState({
    78→    name: user.name ?? "",
    79→    type: "DINE_IN" as (typeof ORDER_TYPES)[number]["id"],
    80→    tableNo: "",
    81→    address: "",
    82→    areaId: "",
    83→    phone: user.phone ?? "",
    84→    note: "",
    85→    payment: "CASH" as (typeof PAYMENTS)[number]["id"],
    86→  });
    87→
    88→  // ===== Voucher =====
    89→  const [voucherInput, setVoucherInput] = useState("");
    90→  const [appliedVoucher, setAppliedVoucher] = useState<VoucherInfo | null>(null);
    91→  const [voucherChecking, setVoucherChecking] = useState(false);
    92→  const [voucherError, setVoucherError] = useState<string | null>(null);
    93→
    94→  const subtotal = cart.totalPrice();
    95→
    96→  /** Area pengantaran terpilih + ongkir — Rp1.000/km, kelipatan jarak
    97→   *  (jarak dibulatkan KE ATAS per 1 km, dihitung ulang juga oleh server) */
    98→  const deliveryArea =
    99→    form.type === "DELIVERY" && form.areaId ? findDeliveryArea(form.areaId) : undefined;
   100→  const deliveryFee =
   101→    form.type === "DELIVERY" && deliveryArea ? (feeForArea(deliveryArea.id) ?? 0) : 0;
   102→
   103→  /** Potongan dihitung ulang otomatis mengikuti isi keranjang */
   104→  const voucherDiscount = useMemo(() => {
   105→    if (!appliedVoucher) return 0;
   106→    return computeVoucherDiscount(appliedVoucher, subtotal);
   107→  }, [appliedVoucher, subtotal]);
   108→
   109→  const total = Math.max(0, subtotal - voucherDiscount + deliveryFee);
   110→
   111→  // Muat daftar voucher untuk saran cepat di bawah input kode
   112→  useEffect(() => {
   113→    void loadVouchers();
   114→  }, [loadVouchers]);
   115→
   116→  // Lepas voucher bila belanja turun di bawah minimum
   117→  useEffect(() => {
   118→    if (appliedVoucher && subtotal < appliedVoucher.minPurchase) {
   119→      setVoucherError(
   120→        `Voucher ${appliedVoucher.code} dilepas — belanja minimum ${formatIDR(appliedVoucher.minPurchase)}.`
   121→      );
   122→      setAppliedVoucher(null);
   123→      setVoucherInput("");
   124→    }
   125→  }, [subtotal, appliedVoucher]);
   126→
   127→  // Keranjang kosong → bersihkan voucher juga
   128→  useEffect(() => {
   129→    if (cart.items.length === 0) {
   130→      setAppliedVoucher(null);
   131→      setVoucherError(null);
   132→    }
   133→  }, [cart.items.length]);
   134→
   135→  /** Tukar kode voucher → validasi server → potongan otomatis */
   136→  const applyVoucher = async (rawCode?: string) => {
   137→    const code = (rawCode ?? voucherInput).trim().toUpperCase();
   138→    if (!code) {
   139→      setVoucherError("Masukkan dulu kode vouchernya ya.");
   140→      return;
   141→    }
   142→    setVoucherChecking(true);
   143→    setVoucherError(null);
   144→    try {
   145→      const res = await apiFetch("/api/vouchers/apply", {
   146→        method: "POST",
   147→        headers: { "Content-Type": "application/json" },
   148→        body: JSON.stringify({ code, subtotal }),
   149→      });
   150→      const data = await res.json();
   151→      if (!res.ok) {
   152→        setAppliedVoucher(null);
   153→        setVoucherError(data.error ?? "Kode voucher tidak valid.");
   154→        return;
   155→      }
   156→      const voucher = data.voucher as VoucherInfo;
   157→      setAppliedVoucher(voucher);
   158→      setVoucherInput(voucher.code);
   159→      showNotice({
   160→        type: "success",
   161→        mascot: true,
   162→        title: "Voucher Terpasang! 🎟️",
   163→        message: `${voucher.code} (${voucher.title}) berhasil ditukar — potongan ${formatIDR(
   164→          data.discount
   165→        )} otomatis diterapkan ke total.`,
   166→      });
   167→    } catch {
   168→      setVoucherError("Tidak dapat menghubungi server. Coba lagi ya.");
   169→    } finally {
   170→      setVoucherChecking(false);
   171→    }
   172→  };
   173→
   174→  // Kode voucher yang "dibawa" dari beranda → isi & tukar otomatis
   175→  useEffect(() => {
   176→    if (pendingCode && cart.items.length > 0) {
   177→      const code = pendingCode;
   178→      setPendingCode(null);
   179→      setVoucherInput(code);
   180→      void applyVoucher(code);
   181→    }
   182→  }, [pendingCode, cart.items.length]);
   183→
   184→  const removeVoucher = () => {
   185→    setAppliedVoucher(null);
   186→    setVoucherInput("");
   187→    setVoucherError(null);
   188→  };
   189→
   190→  /** Voucher yang masih bisa dipakai akun ini untuk subtotal sekarang */
   191→  const quickPicks = useMemo(
   192→    () =>
   193→      vouchers
   194→        .filter((v) => !v.usedByMe && v.remaining > 0 && subtotal >= v.minPurchase)
   195→        .filter((v) => v.code !== appliedVoucher?.code)
   196→        .slice(0, 3),
   197→    [vouchers, subtotal, appliedVoucher]
   198→  );
   199→
   200→  /** Validasi → konfirmasi final (pesanan tidak dapat dibatalkan) → kirim */
   201→  const startOrder = () => {
   202→    if (!form.name.trim()) {
   203→      showNotice({
   204→        type: "error",
   205→        title: "Nama Pemesan Kosong",
   206→        message: "Isi dulu nama pemesan ya, biar pesananmu mudah dikenali dapur 😊",
   207→      });
   208→      return;
   209→    }
   210→    if (form.type === "DINE_IN" && !form.tableNo.trim()) {
   211→      showNotice({
   212→        type: "error",
   213→        title: "Nomor Meja Kosong",
   214→        message: "Isi dulu nomor meja tempat kamu duduk ya 😊",
   215→      });
   216→      return;
   217→    }
   218→    if (form.type === "DELIVERY" && !form.areaId) {
   219→      showNotice({
   220→        type: "error",
   221→        title: "Alamat Pengantaran Kosong",
   222→        message: "Pilih dulu gampong tujuan di area layanan (Kec. Pidie, Kab. Pidie, Aceh 24151).",
   223→      });
   224→      return;
   225→    }
   226→
   227→    const paymentLabel = PAYMENTS.find((p) => p.id === form.payment)?.label ?? "Tunai";
   228→    const voucherText =
   229→      appliedVoucher && voucherDiscount > 0
   230→        ? `, voucher ${appliedVoucher.code} (potongan ${formatIDR(voucherDiscount)})`
   231→        : "";
   232→    const deliveryText =
   233→      form.type === "DELIVERY" && deliveryArea
   234→        ? `, antar ke Gampong ${deliveryArea.gampong} (ongkir ${formatIDR(deliveryFee)})`
   235→        : "";
   236→    showNotice({
   237→      type: "confirm",
   238→      title: "Konfirmasi Pesanan",
   239→      message: `Atas nama ${form.name.trim()} — ${cart.totalItems()} item, pembayaran ${paymentLabel}${voucherText}${deliveryText}, total bayar ${formatIDR(
   240→        total
   241→      )}. Pesanan langsung masuk dapur dan TIDAK DAPAT dibatalkan. Lanjutkan pesanan?`,
   242→      actions: [
   243→        { label: "Ya, Pesan Sekarang", onClick: () => void submitOrder() },
   244→        { label: "Periksa Lagi" },
   245→      ],
   246→    });
   247→  };
   248→
   249→  const submitOrder = async () => {
   250→    setSubmitting(true);
   251→    try {
   252→      const res = await apiFetch("/api/orders", {
   253→        method: "POST",
   254→        headers: { "Content-Type": "application/json" },
   255→        body: JSON.stringify({
   256→          customerName: form.name.trim(),
   257→          items: cart.items.map((i) => ({ productId: i.productId, qty: i.qty })),
   258→          type: form.type,
   259→          tableNo: form.type === "DINE_IN" ? form.tableNo : undefined,
   260→          address: form.type === "DELIVERY" ? form.address.trim() || undefined : undefined,
   261→          deliveryAreaId: form.type === "DELIVERY" ? form.areaId : undefined,
   262→          phone: form.phone,
   263→          note: form.note || undefined,
   264→          paymentMethod: form.payment,
   265→          source: "APP",
   266→          voucherCode: appliedVoucher?.code || undefined,
   267→        }),
   268→      });
   269→      const data = await res.json();
   270→      if (!res.ok) {
   271→        // 401 sudah ditangani global (kembali ke halaman login)
   272→        if (res.status !== 401) {
   273→          // Voucher bisa jadi tidak lolos validasi ulang server (mis. kuota habis duluan)
   274→          if (appliedVoucher && typeof data.error === "string" && data.error.toLowerCase().includes("voucher")) {
   275→            setAppliedVoucher(null);
   276→            setVoucherInput("");
   277→          }
   278→          showNotice({ type: "error", title: "Gagal Membuat Pesanan", message: data.error ?? "Coba lagi ya." });
   279→        }
   280→        return;
   281→      }
   282→      cart.clear();
   283→      setAppliedVoucher(null);
   284→      setVoucherInput("");
   285→      setVoucherError(null);
   286→      refreshOrders("mine");
   287→      refreshVouchers();
   288→      const created = data.order;
   289→      const finalTotal = created?.total ?? total;
   290→
   291→      // Pembayaran QRIS dan masih ada tagihan → tampilkan layar pindai QR
   292→      if (
   293→        created &&
   294→        created.paymentMethod === "QRIS" &&
   295→        created.paymentStatus !== "PAID" &&
   296→        finalTotal > 0
   297→      ) {
   298→        setPayOrder({ id: created.id, code: created.code, total: finalTotal });
   299→        return;
   300→      }
   301→
   302→      const hadVoucher = !!created?.voucherCode;
   303→      showNotice({
   304→        type: "success",
   305→        mascot: true,
   306→        title: "Pesanan Berhasil! 🎉",
   307→        message: `Pesanan ${created.code} sudah masuk dapur${
   308→          hadVoucher ? ` — voucher ${created.voucherCode} terpakai` : ""
   309→        }. Total bayar ${formatIDR(finalTotal)}. Ayam geprekmu segera dimasak!`,
   310→        actions: [
   311→          {
   312→            label: "Lihat Pesanan",
   313→            onClick: () => go("orders"),
   314→          },
   315→        ],
   316→      });
   317→    } catch {
   318→      showNotice({ type: "error", title: "Koneksi Bermasalah", message: "Tidak dapat menghubungi server." });
   319→    } finally {
   320→      setSubmitting(false);
   321→    }
   322→  };
   323→
   324→  /** Layar pembayaran QRIS — dirender di kedua keadaan (keranjang isi/kosong)
   325→      agar tetap tampil setelah pesanan sukses mengosongkan keranjang */
   326→  const qrisSheet = payOrder ? (
   327→    <QrisPaySheet
   328→      order={payOrder}
   329→      variant="customer"
   330→      onClose={() => {
   331→        setPayOrder(null);
   332→        showNotice({
   333→          type: "success",
   334→          mascot: true,
   335→          title: "Pesanan Berhasil! 🎉",
   336→          message: `Pesanan ${payOrder.code} sudah masuk dapur. Selesaikan pembayaran QRIS ${formatIDR(
   337→            payOrder.total
   338→          )} lewat tombol "Bayar QRIS" di daftar pesanan bila belum.`,
   339→          actions: [
   340→            {
   341→              label: "Lihat Pesanan",
   342→              onClick: () => go("orders"),
   343→            },
   344→          ],
   345→        });
   346→      }}
   347→      onPaid={() => void refreshOrders("mine")}
   348→    />
   349→  ) : null;
   350→
   351→  if (cart.items.length === 0) {
   352→    return (
   353→      <>
   354→        {qrisSheet}
   355→        <div className="flex min-h-[70dvh] flex-col items-center justify-center gap-3 px-8 text-center">
   356→          <span className="flex h-20 w-20 items-center justify-center rounded-full bg-orange-50 text-4xl ring-4 ring-orange-100">
   357→            🛒
   358→          </span>
   359→          <h2 className="text-lg font-black text-neutral-800">Keranjang Masih Kosong</h2>
   360→          <p className="text-sm text-neutral-400">
   361→            Ayamnya kesepian nih… yuk pilih menu geprek favoritmu dulu!
   362→          </p>
   363→          <button
   364→            onClick={() => go("menu")}
   365→            className="press-effect mt-2 rounded-full bg-orange-600 px-6 py-2.5 text-sm font-extrabold text-white shadow-lg hover:bg-orange-700"
   366→          >
   367→            Lihat Menu
   368→          </button>
   369→        </div>
   370→      </>
   371→    );
   372→  }
   373→
   374→  return (
   375→    <div className="space-y-4 px-4 pt-[calc(0.75rem+env(safe-area-inset-top,0px))]">
   376→      <header>
   377→        <h1 className="text-xl font-black text-neutral-900">Keranjang 🛒</h1>
   378→        <p className="text-xs text-neutral-400">{cart.totalItems()} item siap digeprek</p>
   379→      </header>
   380→
   381→      {/* Daftar item */}
   382→      <section className="space-y-2.5">
   383→        {cart.items.map((item, i) => (
   384→          <motion.div
   385→            key={item.productId}
   386→            layout={!prefersReduced}
   387→            initial={prefersReduced ? false : { opacity: 0, x: -14 }}
   388→            animate={{ opacity: 1, x: 0 }}
   389→            transition={{ delay: i * 0.04 }}
   390→            className="flex items-center gap-3 rounded-2xl border border-orange-100 bg-white p-3 shadow-sm"
   391→          >
   392→            <ProductThumb
   393→              product={
   394→                products.find((p) => p.id === item.productId) ?? {
   395→                  id: item.productId,
   396→                  updatedAt: "",
   397→                  emoji: item.emoji,
   398→                  imageColor: item.imageColor,
   399→                  hasImage: false,
   400→                }
   401→              }
   402→              className="h-14 w-14 text-2xl"
   403→            />
   404→            <div className="min-w-0 flex-1">
   405→              <p className="line-clamp-1 text-sm font-bold text-neutral-800">{item.name}</p>
   406→              <p className="text-xs font-semibold text-orange-600">{formatIDR(item.price)}</p>
   407→            </div>
   408→            <div className="flex items-center gap-1 rounded-full bg-orange-50 p-0.5">
   409→              <button
   410→                aria-label="Kurangi"
   411→                className="press-effect flex h-7 w-7 items-center justify-center rounded-full bg-white text-orange-700 shadow-sm hover:bg-orange-100"
   412→                onClick={() => cart.setQty(item.productId, item.qty - 1)}
   413→              >
   414→                <Minus className="h-3.5 w-3.5" />
   415→              </button>
   416→              <span className="w-6 text-center text-sm font-black text-orange-800">{item.qty}</span>
   417→              <button
   418→                aria-label="Tambah"
   419→                className="press-effect flex h-7 w-7 items-center justify-center rounded-full bg-orange-600 text-white shadow-sm hover:bg-orange-700"
   420→                onClick={() => cart.setQty(item.productId, item.qty + 1)}
   421→              >
   422→                <Plus className="h-3.5 w-3.5" />
   423→              </button>
   424→            </div>
   425→            <button
   426→              aria-label={`Hapus ${item.name}`}
   427→              className="press-effect ml-1 flex h-8 w-8 items-center justify-center rounded-full text-neutral-300 transition-colors hover:bg-red-50 hover:text-red-500"
   428→              onClick={() => cart.remove(item.productId)}
   429→            >
   430→              <Trash2 className="h-4 w-4" />
   431→            </button>
   432→          </motion.div>
   433→        ))}
   434→      </section>
   435→
   436→      {/* Form checkout */}
   437→      <section className="space-y-4 rounded-2xl border border-orange-100 bg-white p-4 shadow-sm">
   438→        <h2 className="text-sm font-black uppercase tracking-wide text-neutral-700">
   439→          Detail Pesanan
   440→        </h2>
   441→
   442→        {/* Nama pemesan */}
   443→        <div className="space-y-1.5">
   444→          <Label htmlFor="custName">Nama Pemesan</Label>
   445→          <div className="relative">
   446→            <UserRound className="pointer-events-none absolute left-3 top-1/2 h-4 w-4 -translate-y-1/2 text-orange-400" />
   447→            <Input
   448→              id="custName"
   449→              placeholder="cth: Budi Santoso"
   450→              className="border-orange-200 pl-9 focus-visible:ring-orange-400"
   451→              value={form.name}
   452→              onChange={(e) => setForm({ ...form, name: e.target.value })}
   453→              maxLength={60}
   454→            />
   455→          </div>
   456→          <p className="text-[11px] text-neutral-400">
   457→            Nama ini akan tertera pada struk pesananmu.
   458→          </p>
   459→        </div>
   460→
   461→        {/* Tipe pesanan */}
   462→        <div className="grid grid-cols-3 gap-2">
   463→          {ORDER_TYPES.map((t) => (
   464→            <button
   465→              key={t.id}
   466→              type="button"
   467→              onClick={() => setForm({ ...form, type: t.id })}
   468→              className={cn(
   469→                "press-effect flex flex-col items-center gap-1 rounded-xl border p-2.5 text-[10px] font-bold transition-all",
   470→                form.type === t.id
   471→                  ? "border-orange-500 bg-orange-50 text-orange-700 ring-2 ring-orange-200"
   472→                  : "border-neutral-200 bg-white text-neutral-500 hover:border-orange-200"
   473→              )}
   474→            >
   475→              <t.icon className="h-5 w-5" strokeWidth={2.2} />
   476→              {t.label}
   477→            </button>
   478→          ))}
   479→        </div>
   480→
   481→        {form.type === "DINE_IN" && (
   482→          <div className="space-y-1.5">
   483→            <Label htmlFor="tableNo">Nomor Meja</Label>
   484→            <Input
   485→              id="tableNo"
   486→              placeholder="cth: 07"
   487→              className="border-orange-200 focus-visible:ring-orange-400"
   488→              value={form.tableNo}
   489→              onChange={(e) => setForm({ ...form, tableNo: e.target.value })}
   490→            />
   491→          </div>
   492→        )}
   493→        {form.type === "DELIVERY" && (
   494→          <div className="space-y-1.5">
   495→            <Label htmlFor="address">Alamat Pengantaran</Label>
   496→            {/* Pilih gampong (area Kec. Pidie 24151) → popup daftar area + ongkir */}
   497→            <button
   498→              type="button"
   499→              onClick={() => setAreaOpen(true)}
   500→              aria-label="Pilih alamat pengantaran"
   501→              className={cn(
   502→                "press-effect flex w-full items-center gap-3 rounded-xl border p-3 text-left transition-all",
   503→                deliveryArea
   504→                  ? "border-orange-500 bg-orange-50 ring-2 ring-orange-200"
   505→                  : "border-orange-200 bg-white hover:border-orange-300"
   506→              )}
   507→            >
   508→              <span
   509→                className={cn(
   510→                  "flex h-9 w-9 shrink-0 items-center justify-center rounded-xl",
   511→                  deliveryArea ? "bg-orange-600 text-white" : "bg-orange-100 text-orange-600"
   512→                )}
   513→              >
