     1→"use client";
     2→
     3→import { useEffect, useMemo, useState } from "react";
     4→import { motion, useReducedMotion } from "framer-motion";
     5→import {
     6→  Bike,
     7→  Loader2,
     8→  Minus,
     9→  Plus,
    10→  ShoppingBag,
    11→  Ticket,
    12→  Trash2,
    13→  TriangleAlert,
    14→  UserRound,
    15→  Utensils,
    16→  X,
    17→} from "lucide-react";
    18→import { Input } from "@/components/ui/input";
    19→import { Label } from "@/components/ui/label";
    20→import { Textarea } from "@/components/ui/textarea";
    21→import { ProductThumb } from "@/components/shared/ProductThumb";
    22→import { QrisPaySheet, type QrisPayOrder } from "@/components/shared/QrisPaySheet";
    23→import { useCartStore, useNoticeStore, type AppView } from "@/lib/store";
    24→import { apiFetch } from "@/lib/api";
    25→import { refreshOrders, refreshVouchers, useProductsStore, useVouchersStore } from "@/lib/hooks";
    26→import {
    27→  computeVoucherDiscount,
    28→  formatIDR,
    29→  formatVoucherValue,
    30→  type AppUser,
    31→  type VoucherInfo,
    32→} from "@/lib/types";
    33→import {
    34→  earnPointsForTotal,
    35→  POINTS_EARN_PER_STEP,
    36→  POINTS_EARN_STEP,
    37→  voucherValueForPoints,
    38→} from "@/lib/points-rules";
    39→import { cn } from "@/lib/utils";
    40→
    41→const ORDER_TYPES = [
    42→  { id: "DINE_IN", label: "Makan di Tempat", icon: Utensils, placeholderTable: true },
    43→  { id: "TAKEAWAY", label: "Bungkus / Ambil", icon: ShoppingBag, placeholderTable: false },
    44→  { id: "DELIVERY", label: "Antar ke Alamat", icon: Bike, placeholderTable: false },
    45→] as const;
    46→
    47→const PAYMENTS = [
    48→  { id: "CASH", label: "Tunai", emoji: "💵" },
    49→  { id: "QRIS", label: "QRIS", emoji: "📱" },
    50→] as const;
    51→
    52→export function CartView({ user, go }: { user: AppUser; go: (v: AppView) => void }) {
    53→  const cart = useCartStore();
    54→  const showNotice = useNoticeStore((s) => s.show);
    55→  const prefersReduced = useReducedMotion();
    56→  const products = useProductsStore((s) => s.products);
    57→  const vouchers = useVouchersStore((s) => s.vouchers);
    58→  const loadVouchers = useVouchersStore((s) => s.load);
    59→  const pendingCode = useVouchersStore((s) => s.pendingCode);
    60→  const setPendingCode = useVouchersStore((s) => s.setPendingCode);
    61→  const [submitting, setSubmitting] = useState(false);
    62→  /** Pesanan QRIS yang sedang dibayar (layar QR terbuka) */
    63→  const [payOrder, setPayOrder] = useState<QrisPayOrder | null>(null);
    64→  const [form, setForm] = useState({
    65→    name: user.name ?? "",
    66→    type: "DINE_IN" as (typeof ORDER_TYPES)[number]["id"],
    67→    tableNo: "",
    68→    address: "",
    69→    phone: user.phone ?? "",
    70→    note: "",
    71→    payment: "CASH" as (typeof PAYMENTS)[number]["id"],
    72→  });
    73→
    74→  // ===== Voucher =====
    75→  const [voucherInput, setVoucherInput] = useState("");
    76→  const [appliedVoucher, setAppliedVoucher] = useState<VoucherInfo | null>(null);
    77→  const [voucherChecking, setVoucherChecking] = useState(false);
    78→  const [voucherError, setVoucherError] = useState<string | null>(null);
    79→
    80→  const subtotal = cart.totalPrice();
    81→
    82→  /** Potongan dihitung ulang otomatis mengikuti isi keranjang */
    83→  const voucherDiscount = useMemo(() => {
    84→    if (!appliedVoucher) return 0;
    85→    return computeVoucherDiscount(appliedVoucher, subtotal);
    86→  }, [appliedVoucher, subtotal]);
    87→
    88→  const total = Math.max(0, subtotal - voucherDiscount);
    89→
    90→  // Muat daftar voucher untuk saran cepat di bawah input kode
    91→  useEffect(() => {
    92→    void loadVouchers();
    93→  }, [loadVouchers]);
    94→
    95→  // Lepas voucher bila belanja turun di bawah minimum
    96→  useEffect(() => {
    97→    if (appliedVoucher && subtotal < appliedVoucher.minPurchase) {
    98→      setVoucherError(
    99→        `Voucher ${appliedVoucher.code} dilepas — belanja minimum ${formatIDR(appliedVoucher.minPurchase)}.`
   100→      );
   101→      setAppliedVoucher(null);
   102→      setVoucherInput("");
   103→    }
   104→  }, [subtotal, appliedVoucher]);
   105→
   106→  // Keranjang kosong → bersihkan voucher juga
   107→  useEffect(() => {
   108→    if (cart.items.length === 0) {
   109→      setAppliedVoucher(null);
   110→      setVoucherError(null);
   111→    }
   112→  }, [cart.items.length]);
   113→
   114→  /** Tukar kode voucher → validasi server → potongan otomatis */
   115→  const applyVoucher = async (rawCode?: string) => {
   116→    const code = (rawCode ?? voucherInput).trim().toUpperCase();
   117→    if (!code) {
   118→      setVoucherError("Masukkan dulu kode vouchernya ya.");
   119→      return;
   120→    }
   121→    setVoucherChecking(true);
   122→    setVoucherError(null);
   123→    try {
   124→      const res = await apiFetch("/api/vouchers/apply", {
   125→        method: "POST",
   126→        headers: { "Content-Type": "application/json" },
   127→        body: JSON.stringify({ code, subtotal }),
   128→      });
   129→      const data = await res.json();
   130→      if (!res.ok) {
   131→        setAppliedVoucher(null);
   132→        setVoucherError(data.error ?? "Kode voucher tidak valid.");
   133→        return;
   134→      }
   135→      const voucher = data.voucher as VoucherInfo;
   136→      setAppliedVoucher(voucher);
   137→      setVoucherInput(voucher.code);
   138→      showNotice({
   139→        type: "success",
   140→        mascot: true,
   141→        title: "Voucher Terpasang! 🎟️",
   142→        message: `${voucher.code} (${voucher.title}) berhasil ditukar — potongan ${formatIDR(
   143→          data.discount
   144→        )} otomatis diterapkan ke total.`,
   145→      });
   146→    } catch {
   147→      setVoucherError("Tidak dapat menghubungi server. Coba lagi ya.");
   148→    } finally {
   149→      setVoucherChecking(false);
   150→    }
   151→  };
   152→
   153→  // Kode voucher yang "dibawa" dari beranda → isi & tukar otomatis
   154→  useEffect(() => {
   155→    if (pendingCode && cart.items.length > 0) {
   156→      const code = pendingCode;
   157→      setPendingCode(null);
   158→      setVoucherInput(code);
   159→      void applyVoucher(code);
   160→    }
   161→  }, [pendingCode, cart.items.length]);
   162→
   163→  const removeVoucher = () => {
   164→    setAppliedVoucher(null);
   165→    setVoucherInput("");
   166→    setVoucherError(null);
   167→  };
   168→
   169→  /** Voucher yang masih bisa dipakai akun ini untuk subtotal sekarang */
   170→  const quickPicks = useMemo(
   171→    () =>
   172→      vouchers
   173→        .filter((v) => !v.usedByMe && v.remaining > 0 && subtotal >= v.minPurchase)
   174→        .filter((v) => v.code !== appliedVoucher?.code)
   175→        .slice(0, 3),
   176→    [vouchers, subtotal, appliedVoucher]
   177→  );
   178→
   179→  /** Validasi → konfirmasi final (pesanan tidak dapat dibatalkan) → kirim */
   180→  const startOrder = () => {
   181→    if (!form.name.trim()) {
   182→      showNotice({
   183→        type: "error",
   184→        title: "Nama Pemesan Kosong",
   185→        message: "Isi dulu nama pemesan ya, biar pesananmu mudah dikenali dapur 😊",
   186→      });
   187→      return;
   188→    }
   189→    if (form.type === "DINE_IN" && !form.tableNo.trim()) {
   190→      showNotice({
   191→        type: "error",
   192→        title: "Nomor Meja Kosong",
   193→        message: "Isi dulu nomor meja tempat kamu duduk ya 😊",
   194→      });
   195→      return;
   196→    }
   197→    if (form.type === "DELIVERY" && !form.address.trim()) {
   198→      showNotice({
   199→        type: "error",
   200→        title: "Alamat Kosong",
   201→        message: "Tuliskan alamat lengkap pengantaran.",
   202→      });
   203→      return;
   204→    }
   205→
   206→    const paymentLabel = PAYMENTS.find((p) => p.id === form.payment)?.label ?? "Tunai";
   207→    const voucherText =
   208→      appliedVoucher && voucherDiscount > 0
   209→        ? `, voucher ${appliedVoucher.code} (potongan ${formatIDR(voucherDiscount)})`
   210→        : "";
   211→    showNotice({
   212→      type: "confirm",
   213→      title: "Konfirmasi Pesanan",
   214→      message: `Atas nama ${form.name.trim()} — ${cart.totalItems()} item, pembayaran ${paymentLabel}${voucherText}, total bayar ${formatIDR(
   215→        total
   216→      )}. Pesanan langsung masuk dapur dan TIDAK DAPAT dibatalkan. Lanjutkan pesanan?`,
   217→      actions: [
   218→        { label: "Ya, Pesan Sekarang", onClick: () => void submitOrder() },
   219→        { label: "Periksa Lagi" },
   220→      ],
   221→    });
   222→  };
   223→
   224→  const submitOrder = async () => {
   225→    setSubmitting(true);
   226→    try {
   227→      const res = await apiFetch("/api/orders", {
   228→        method: "POST",
   229→        headers: { "Content-Type": "application/json" },
   230→        body: JSON.stringify({
   231→          customerName: form.name.trim(),
   232→          items: cart.items.map((i) => ({ productId: i.productId, qty: i.qty })),
   233→          type: form.type,
   234→          tableNo: form.type === "DINE_IN" ? form.tableNo : undefined,
   235→          address: form.type === "DELIVERY" ? form.address : undefined,
   236→          phone: form.phone,
   237→          note: form.note || undefined,
   238→          paymentMethod: form.payment,
   239→          source: "APP",
   240→          voucherCode: appliedVoucher?.code || undefined,
   241→        }),
   242→      });
   243→      const data = await res.json();
   244→      if (!res.ok) {
   245→        // 401 sudah ditangani global (kembali ke halaman login)
   246→        if (res.status !== 401) {
   247→          // Voucher bisa jadi tidak lolos validasi ulang server (mis. kuota habis duluan)
   248→          if (appliedVoucher && typeof data.error === "string" && data.error.toLowerCase().includes("voucher")) {
   249→            setAppliedVoucher(null);
   250→            setVoucherInput("");
   251→          }
   252→          showNotice({ type: "error", title: "Gagal Membuat Pesanan", message: data.error ?? "Coba lagi ya." });
   253→        }
   254→        return;
   255→      }
   256→      cart.clear();
   257→      setAppliedVoucher(null);
   258→      setVoucherInput("");
   259→      setVoucherError(null);
   260→      refreshOrders("mine");
   261→      refreshVouchers();
   262→      const created = data.order;
   263→      const finalTotal = created?.total ?? total;
   264→
   265→      // Pembayaran QRIS dan masih ada tagihan → tampilkan layar pindai QR
   266→      if (
   267→        created &&
   268→        created.paymentMethod === "QRIS" &&
   269→        created.paymentStatus !== "PAID" &&
   270→        finalTotal > 0
   271→      ) {
   272→        setPayOrder({ id: created.id, code: created.code, total: finalTotal });
   273→        return;
   274→      }
   275→
   276→      const hadVoucher = !!created?.voucherCode;
   277→      showNotice({
   278→        type: "success",
   279→        mascot: true,
   280→        title: "Pesanan Berhasil! 🎉",
   281→        message: `Pesanan ${created.code} sudah masuk dapur${
   282→          hadVoucher ? ` — voucher ${created.voucherCode} terpakai` : ""
   283→        }. Total bayar ${formatIDR(finalTotal)}. Ayam geprekmu segera dimasak!`,
   284→        actions: [
   285→          {
   286→            label: "Lihat Pesanan",
   287→            onClick: () => go("orders"),
   288→          },
   289→        ],
   290→      });
   291→    } catch {
   292→      showNotice({ type: "error", title: "Koneksi Bermasalah", message: "Tidak dapat menghubungi server." });
   293→    } finally {
   294→      setSubmitting(false);
   295→    }
   296→  };
   297→
   298→  /** Layar pembayaran QRIS — dirender di kedua keadaan (keranjang isi/kosong)
   299→      agar tetap tampil setelah pesanan sukses mengosongkan keranjang */
   300→  const qrisSheet = payOrder ? (
   301→    <QrisPaySheet
   302→      order={payOrder}
   303→      variant="customer"
   304→      onClose={() => {
   305→        setPayOrder(null);
   306→        showNotice({
   307→          type: "success",
   308→          mascot: true,
   309→          title: "Pesanan Berhasil! 🎉",
   310→          message: `Pesanan ${payOrder.code} sudah masuk dapur. Selesaikan pembayaran QRIS ${formatIDR(
   311→            payOrder.total
   312→          )} lewat tombol "Bayar QRIS" di daftar pesanan bila belum.`,
   313→          actions: [
   314→            {
   315→              label: "Lihat Pesanan",
   316→              onClick: () => go("orders"),
   317→            },
   318→          ],
   319→        });
   320→      }}
   321→      onPaid={() => void refreshOrders("mine")}
   322→    />
   323→  ) : null;
   324→
   325→  if (cart.items.length === 0) {
   326→    return (
   327→      <>
   328→        {qrisSheet}
   329→        <div className="flex min-h-[70dvh] flex-col items-center justify-center gap-3 px-8 text-center">
   330→          <span className="flex h-20 w-20 items-center justify-center rounded-full bg-orange-50 text-4xl ring-4 ring-orange-100">
   331→            🛒
   332→          </span>
   333→          <h2 className="text-lg font-black text-neutral-800">Keranjang Masih Kosong</h2>
   334→          <p className="text-sm text-neutral-400">
   335→            Ayamnya kesepian nih… yuk pilih menu geprek favoritmu dulu!
   336→          </p>
   337→          <button
   338→            onClick={() => go("menu")}
   339→            className="press-effect mt-2 rounded-full bg-orange-600 px-6 py-2.5 text-sm font-extrabold text-white shadow-lg hover:bg-orange-700"
   340→          >
   341→            Lihat Menu
   342→          </button>
   343→        </div>
   344→      </>
   345→    );
   346→  }
   347→
   348→  return (
   349→    <div className="space-y-4 px-4 pt-[calc(0.75rem+env(safe-area-inset-top,0px))]">
   350→      <header>
   351→        <h1 className="text-xl font-black text-neutral-900">Keranjang 🛒</h1>
   352→        <p className="text-xs text-neutral-400">{cart.totalItems()} item siap digeprek</p>
   353→      </header>
   354→
   355→      {/* Daftar item */}
   356→      <section className="space-y-2.5">
   357→        {cart.items.map((item, i) => (
   358→          <motion.div
   359→            key={item.productId}
   360→            layout={!prefersReduced}
   361→            initial={prefersReduced ? false : { opacity: 0, x: -14 }}
   362→            animate={{ opacity: 1, x: 0 }}
   363→            transition={{ delay: i * 0.04 }}
   364→            className="flex items-center gap-3 rounded-2xl border border-orange-100 bg-white p-3 shadow-sm"
   365→          >
   366→            <ProductThumb
   367→              product={
   368→                products.find((p) => p.id === item.productId) ?? {
   369→                  id: item.productId,
   370→                  updatedAt: "",
   371→                  emoji: item.emoji,
   372→                  imageColor: item.imageColor,
   373→                  hasImage: false,
   374→                }
   375→              }
   376→              className="h-14 w-14 text-2xl"
   377→            />
   378→            <div className="min-w-0 flex-1">
   379→              <p className="line-clamp-1 text-sm font-bold text-neutral-800">{item.name}</p>
   380→              <p className="text-xs font-semibold text-orange-600">{formatIDR(item.price)}</p>
   381→            </div>
   382→            <div className="flex items-center gap-1 rounded-full bg-orange-50 p-0.5">
   383→              <button
   384→                aria-label="Kurangi"
   385→                className="press-effect flex h-7 w-7 items-center justify-center rounded-full bg-white text-orange-700 shadow-sm hover:bg-orange-100"
   386→                onClick={() => cart.setQty(item.productId, item.qty - 1)}
   387→              >
   388→                <Minus className="h-3.5 w-3.5" />
   389→              </button>
   390→              <span className="w-6 text-center text-sm font-black text-orange-800">{item.qty}</span>
   391→              <button
   392→                aria-label="Tambah"
   393→                className="press-effect flex h-7 w-7 items-center justify-center rounded-full bg-orange-600 text-white shadow-sm hover:bg-orange-700"
   394→                onClick={() => cart.setQty(item.productId, item.qty + 1)}
   395→              >
   396→                <Plus className="h-3.5 w-3.5" />
   397→              </button>
   398→            </div>
   399→            <button
   400→              aria-label={`Hapus ${item.name}`}
   401→              className="press-effect ml-1 flex h-8 w-8 items-center justify-center rounded-full text-neutral-300 transition-colors hover:bg-red-50 hover:text-red-500"
   402→              onClick={() => cart.remove(item.productId)}
   403→            >
   404→              <Trash2 className="h-4 w-4" />
   405→            </button>
   406→          </motion.div>
   407→        ))}
   408→      </section>
   409→
   410→      {/* Form checkout */}
   411→      <section className="space-y-4 rounded-2xl border border-orange-100 bg-white p-4 shadow-sm">
   412→        <h2 className="text-sm font-black uppercase tracking-wide text-neutral-700">
   413→          Detail Pesanan
   414→        </h2>
   415→
   416→        {/* Nama pemesan */}
   417→        <div className="space-y-1.5">
   418→          <Label htmlFor="custName">Nama Pemesan</Label>
   419→          <div className="relative">
   420→            <UserRound className="pointer-events-none absolute left-3 top-1/2 h-4 w-4 -translate-y-1/2 text-orange-400" />
   421→            <Input
   422→              id="custName"
   423→              placeholder="cth: Budi Santoso"
   424→              className="border-orange-200 pl-9 focus-visible:ring-orange-400"
   425→              value={form.name}
   426→              onChange={(e) => setForm({ ...form, name: e.target.value })}
   427→              maxLength={60}
   428→            />
   429→          </div>
   430→          <p className="text-[11px] text-neutral-400">
   431→            Nama ini akan tertera pada struk pesananmu.
   432→          </p>
   433→        </div>
   434→
   435→        {/* Tipe pesanan */}
   436→        <div className="grid grid-cols-3 gap-2">
   437→          {ORDER_TYPES.map((t) => (
   438→            <button
   439→              key={t.id}
   440→              type="button"
   441→              onClick={() => setForm({ ...form, type: t.id })}
   442→              className={cn(
   443→                "press-effect flex flex-col items-center gap-1 rounded-xl border p-2.5 text-[10px] font-bold transition-all",
   444→                form.type === t.id
   445→                  ? "border-orange-500 bg-orange-50 text-orange-700 ring-2 ring-orange-200"
   446→                  : "border-neutral-200 bg-white text-neutral-500 hover:border-orange-200"
   447→              )}
   448→            >
   449→              <t.icon className="h-5 w-5" strokeWidth={2.2} />
   450→              {t.label}
   451→            </button>
   452→          ))}
   453→        </div>
   454→
   455→        {form.type === "DINE_IN" && (
   456→          <div className="space-y-1.5">
   457→            <Label htmlFor="tableNo">Nomor Meja</Label>
   458→            <Input
   459→              id="tableNo"
   460→              placeholder="cth: 07"
   461→              className="border-orange-200 focus-visible:ring-orange-400"
   462→              value={form.tableNo}
   463→              onChange={(e) => setForm({ ...form, tableNo: e.target.value })}
   464→            />
   465→          </div>
   466→        )}
   467→        {form.type === "DELIVERY" && (
   468→          <div className="space-y-1.5">
   469→            <Label htmlFor="address">Alamat Pengantaran</Label>
   470→            <Textarea
   471→              id="address"
   472→              placeholder="Nama jalan, nomor rumah, patokan…"
   473→              className="border-orange-200 focus-visible:ring-orange-400"
   474→              value={form.address}
   475→              onChange={(e) => setForm({ ...form, address: e.target.value })}
   476→            />
   477→          </div>
   478→        )}
   479→        <div className="space-y-1.5">
   480→          <Label htmlFor="phone">No. HP (untuk kabari pesanan)</Label>
   481→          <Input
   482→            id="phone"
   483→            placeholder="08xx-xxxx-xxxx"
   484→            className="border-orange-200 focus-visible:ring-orange-400"
   485→            value={form.phone}
   486→            onChange={(e) => setForm({ ...form, phone: e.target.value })}
   487→          />
   488→        </div>
   489→        <div className="space-y-1.5">
   490→          <Label htmlFor="note">Catatan (opsional)</Label>
   491→          <Textarea
   492→            id="note"
   493→            placeholder="cth: sambal jangan terlalu pedas, ayam dipisah…"
   494→            className="min-h-[64px] border-orange-200 focus-visible:ring-orange-400"
   495→            value={form.note}
   496→            onChange={(e) => setForm({ ...form, note: e.target.value })}
   497→          />
   498→        </div>
   499→
   500→        {/* Metode bayar */}
   501→        <div className="space-y-1.5">
   502→          <Label>Metode Pembayaran</Label>
   503→          <div className="grid grid-cols-2 gap-2">
   504→            {PAYMENTS.map((p) => (
   505→              <button
   506→                key={p.id}
   507→                type="button"
   508→                onClick={() => setForm({ ...form, payment: p.id })}
   509→                className={cn(
   510→                  "press-effect flex items-center justify-center gap-2 rounded-xl border p-2.5 text-xs font-bold transition-all",
   511→                  form.payment === p.id
   512→                    ? "border-orange-500 bg-orange-50 text-orange-700 ring-2 ring-orange-200"
   513→                    : "border-neutral-200 text-neutral-500 hover:border-orange-200"
   514→                )}
   515→              >
   516→                <span className="text-base">{p.emoji}</span> {p.label}
   517→              </button>
   518→            ))}
   519→          </div>
   520→        </div>
   521→      </section>
   522→
   523→      {/* Ringkasan */}
   524→      <section className="rounded-2xl border border-orange-100 bg-gradient-to-b from-orange-50/70 to-white p-4 shadow-sm">
   525→        <div className="flex items-center justify-between gap-3 text-sm text-neutral-500">
   526→          <span className="shrink-0">Atas nama</span>
   527→          <span className="max-w-[65%] truncate text-right font-semibold text-neutral-700">
   528→            {form.name.trim() || "—"}
   529→          </span>
   530→        </div>
   531→        <div className="mt-1 flex items-center justify-between text-sm text-neutral-500">
   532→          <span>Subtotal</span>
   533→          <span className="font-semibold text-neutral-700">{formatIDR(subtotal)}</span>
   534→        </div>
   535→
   536→        {/* ===== Tukar kode voucher ===== */}
   537→        <div className="mt-2.5 space-y-1.5">
   538→          <Label htmlFor="voucherCode" className="text-[11px] font-bold uppercase tracking-wide text-neutral-500">
   539→            <span className="flex items-center gap-1">
   540→              <Ticket className="h-3.5 w-3.5 text-orange-500" /> Kode Voucher
   541→            </span>
   542→          </Label>
   543→
   544→          {appliedVoucher ? (
   545→            <motion.div
   546→              initial={prefersReduced ? false : { opacity: 0, scale: 0.98 }}
   547→              animate={{ opacity: 1, scale: 1 }}
   548→              className="flex items-center gap-2.5 rounded-xl border border-green-300 bg-green-50 p-2.5"
   549→            >
   550→              <span className="flex h-9 w-9 shrink-0 items-center justify-center rounded-lg bg-green-600 text-white">
   551→                <Ticket className="h-4 w-4" />
   552→              </span>
   553→              <div className="min-w-0 flex-1">
   554→                <p className="truncate text-xs font-black text-green-900">
   555→                  {appliedVoucher.code} — {formatVoucherValue(appliedVoucher)}{" "}
   556→                  {appliedVoucher.type === "PERCENT" ? "diskon" : "potongan"}
   557→                </p>
   558→                <p className="truncate text-[10px] font-semibold text-green-700">
   559→                  {appliedVoucher.title}
   560→                  {voucherDiscount > 0 ? ` • hemat ${formatIDR(voucherDiscount)}` : ""}
   561→                </p>
   562→              </div>
   563→              <button
   564→                type="button"
   565→                aria-label="Lepas voucher"
   566→                onClick={removeVoucher}
   567→                className="press-effect flex h-8 w-8 shrink-0 items-center justify-center rounded-full bg-white text-neutral-400 shadow-sm hover:bg-red-50 hover:text-red-500"
   568→              >
   569→                <X className="h-4 w-4" />
   570→              </button>
   571→            </motion.div>
   572→          ) : (
   573→            <>
   574→              <div className="flex gap-2">
   575→                <div className="relative flex-1">
   576→                  <Ticket className="pointer-events-none absolute left-3 top-1/2 h-4 w-4 -translate-y-1/2 text-orange-400" />
   577→                  <Input
   578→                    id="voucherCode"
   579→                    placeholder="cth: GEPREK10"
   580→                    className="border-orange-200 pl-9 font-mono text-sm font-bold uppercase tracking-wider focus-visible:ring-orange-400"
   581→                    value={voucherInput}
   582→                    maxLength={24}
   583→                    autoComplete="off"
   584→                    onChange={(e) =>
   585→                      setVoucherInput(e.target.value.toUpperCase().replace(/[^A-Z0-9-]/g, ""))
   586→                    }
   587→                    onKeyDown={(e) => {
   588→                      if (e.key === "Enter") {
   589→                        e.preventDefault();
   590→                        void applyVoucher();
   591→                      }
   592→                    }}
   593→                  />
   594→                </div>
   595→                <button
   596→                  type="button"
   597→                  onClick={() => void applyVoucher()}
   598→                  disabled={voucherChecking}
   599→                  className="press-effect flex h-10 shrink-0 items-center gap-1.5 rounded-xl bg-orange-600 px-4 text-xs font-extrabold text-white shadow-md hover:bg-orange-700 disabled:opacity-60"
   600→                >
   601→                  {voucherChecking ? (
   602→                    <Loader2 className="h-4 w-4 animate-spin" />
   603→                  ) : (
   604→                    "Tukar"
   605→                  )}
   606→                </button>
   607→              </div>
   608→              {voucherError && (
   609→                <p className="text-[11px] font-semibold leading-snug text-red-500">⚠️ {voucherError}</p>
   610→              )}
   611→              {quickPicks.length > 0 && (
   612→                <div className="flex flex-wrap gap-1.5 pt-0.5">
   613→                  {quickPicks.map((v) => (
   614→                    <button
   615→                      key={v.id}
   616→                      type="button"
   617→                      onClick={() => {
   618→                        setVoucherInput(v.code);
   619→                        void applyVoucher(v.code);
   620→                      }}
   621→                      className="press-effect rounded-full border border-orange-200 bg-orange-50 px-2.5 py-1 text-[10px] font-bold text-orange-700 transition hover:bg-orange-100"
   622→                    >
   623→                      🎟️ {v.code} · {formatVoucherValue(v)}
   624→                    </button>
   625→                  ))}
   626→                </div>
   627→              )}
   628→            </>
   629→          )}
   630→        </div>
   631→
   632→        {appliedVoucher && voucherDiscount > 0 && (
   633→          <div className="mt-1.5 flex items-center justify-between text-sm">
   634→            <span className="flex items-center gap-1 font-semibold text-green-700">
   635→              <Ticket className="h-3.5 w-3.5" /> Potongan voucher
   636→            </span>
   637→            <span className="font-black text-green-600">−{formatIDR(voucherDiscount)}</span>
   638→          </div>
   639→        )}
   640→        {appliedVoucher && voucherDiscount === 0 && (
   641→          <p className="mt-1 text-[11px] font-semibold text-amber-600">
   642→            Potongan voucher baru berlaku saat belanja minimal {formatIDR(appliedVoucher.minPurchase)}.
   643→          </p>
   644→        )}
   645→
   646→        <div className="mt-1 flex items-center justify-between text-sm text-neutral-500">
   647→          <span>Biaya layanan</span>
   648→          <span className="font-semibold text-green-600">Gratis</span>
   649→        </div>
   650→        <div className="my-2.5 border-t border-dashed border-orange-200" />
   651→        <div className="flex items-center justify-between">
   652→          <span className="text-sm font-bold text-neutral-700">Total Bayar</span>
   653→          <motion.span
   654→            key={total}
   655→            initial={prefersReduced ? false : { scale: 1.06 }}
   656→            animate={{ scale: 1 }}
   657→            className="text-xl font-black text-orange-600"
   658→          >
   659→            {formatIDR(total)}
   660→          </motion.span>
   661→        </div>
   662→        <p className="mt-1 text-[11px] text-neutral-400">
   663→          {earnPointsForTotal(total) > 0 ? (
   664→            <>
   665→              🔖 Dapat {earnPointsForTotal(total).toLocaleString("id-ID")} poin member (setara{" "}
   666→              {formatIDR(voucherValueForPoints(earnPointsForTotal(total)))}) saat pesanan selesai
   667→            </>
   668→          ) : (
   669→            <>
   670→              🔖 Tiap belanja {formatIDR(POINTS_EARN_STEP)} dapat {POINTS_EARN_PER_STEP.toLocaleString("id-ID")} poin
   671→              member saat pesanan selesai
   672→            </>
   673→          )}
   674→        </p>
   675→
   676→        {/* Peringatan: pesanan tidak dapat dibatalkan */}
   677→        <div className="mt-3 flex items-start gap-2 rounded-xl bg-amber-50 p-2.5 text-[11px] font-semibold leading-relaxed text-amber-800 ring-1 ring-amber-200">
   678→          <TriangleAlert className="mt-0.5 h-3.5 w-3.5 shrink-0" />
   679→          <span>
   680→            Pesanan yang sudah dikirim langsung masuk dapur dan{" "}
   681→            <strong>TIDAK DAPAT dibatalkan</strong>. Pastikan nama, jenis pesanan,
   682→            dan totalnya sudah benar ya!
   683→          </span>
   684→        </div>
   685→
   686→        <motion.button
   687→          whileTap={prefersReduced ? undefined : { scale: 0.97 }}
   688→          disabled={submitting}
   689→          onClick={startOrder}
   690→          className="press-effect mt-3.5 flex h-12 w-full items-center justify-center gap-2 rounded-xl bg-gradient-to-r from-orange-600 to-orange-500 text-sm font-extrabold text-white shadow-lg shadow-orange-500/35 transition hover:brightness-105 disabled:opacity-60"
   691→        >
   692→          {submitting ? (
   693→            <>
   694→              <Loader2 className="h-4 w-4 animate-spin" /> Memproses…
   695→            </>
   696→          ) : (
   697→            <>
   698→              {appliedVoucher && voucherDiscount > 0 && <Ticket className="h-4 w-4" />}
   699→              {form.payment === "QRIS" ? "Pesan & Bayar QRIS" : "Pesan Sekarang"} — {formatIDR(total)}
   700→            </>
   701→          )}
   702→        </motion.button>
   703→      </section>
   704→
   705→      {/* ===== LAYAR PEMBAYARAN QRIS ===== */}
   706→      {qrisSheet}
   707→    </div>
   708→  );
