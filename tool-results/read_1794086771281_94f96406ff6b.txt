    24→    24→/** Pesanan yang akan dibayar via QRIS */
    25→    25→export type QrisPayOrder = { id: string; code: string; total: number };
    26→    26→
    27→    27→type Phase = "scan" | "paid" | "cancelled";
    28→    28→
    29→    29→type QrisInfo = {
    30→    30→  payload: string | null;
    31→    31→  imageUrl: string | null;
    32→    32→  mode: "GATEWAY" | "LOCAL";
    33→    33→  autoVerify: boolean;
    34→    34→  /** gateway QRIS bawaan (sandbox) aktif — tombol simulasi dana ditampilkan */
    35→    35→  sandbox: boolean;
    36→    36→  /** penyedia verifikasi: "midtrans" | "midtrans-sandbox" | "internal" | null */
    37→    37→  provider: string | null;
    38→    38→  /** Alasan penyedia utama (Midtrans) ditolak sehingga QR berasal dari fallback uji lokal */
    39→    39→  fallbackReason: string | null;
    40→    40→  /** Sandbox Midtrans: tautan simulator uji pembayaran (QR berisi URL uji) */
    41→    41→  sandboxSimulatorUrl: string | null;
    42→    42→};
    43→    43→
    44→    44→/** Interval polling verifikasi otomatis */
    45→    45→const POLL_MS = 4000;
    46→    46→/** Batas waktu pembayaran — 30 menit sejak pesanan dibuat.
    47→    47→ *  Lewat dari itu pesanan batal otomatis (server membatalkan saat sweep/poll). */
    48→    48→const DURATION_SEC = 30 * 60; // 30 menit
    49→    49→
    50→    50→function mmss(sec: number): string {
    51→    51→  const m = Math.floor(sec / 60);
    52→    52→  const s = sec % 60;
    53→    53→  return `${String(m).padStart(2, "0")}:${String(s).padStart(2, "0")}`;
    54→    54→}
    55→    55→
    56→    56→/**
    57→    57→ * Layar pembayaran QRIS — pelanggan maupun kasir memakai layar yang sama.
    58→    58→ *
    59→    59→ * VERIFIKASI OTOMATIS: setelah QR tampil, layar ini memantau status ke server
    60→    60→ * tiap beberapa detik; server bertanya ke penyedia QRIS "sudah masuk belum
    61→    61→ * dananya?" dan melunaskan pesanan begitu dana masuk. TIDAK ADA tombol
    62→    62→ * "Saya Sudah Bayar" / "Uang Sudah Masuk" — kasir memang tidak boleh
    63→    63→ * (dan tidak bisa) melunaskan pesanan QRIS secara manual.
    64→    64→ */
    65→    65→export function QrisPaySheet({
    66→    66→  order,
    67→    67→  variant,
    68→    68→  onClose,
    69→    69→  onPaid,
    70→    70→  onPrintReceipt,
    71→    71→}: {
    72→    72→  order: QrisPayOrder;
    73→    73→  variant: "customer" | "pos";
    74→    74→  onClose: () => void;
    75→    75→  /** dipanggil saat status pembayaran berubah menjadi lunas (mis. refresh daftar) */
    76→    76→  onPaid?: () => void | Promise<void>;
    77→    77→  /** POS: cetak struk setelah lunas (bila disediakan halaman pemanggil) */
    78→    78→  onPrintReceipt?: () => void | Promise<void>;
    79→    79→}) {
    80→    80→  const prefersReduced = useReducedMotion();
    81→    81→  const isPos = variant === "pos";
    82→    82→
    83→    83→  const [info, setInfo] = useState<QrisInfo | null>(null);
    84→    84→  const [loadError, setLoadError] = useState<string | null>(null);
    85→    85→  const [phase, setPhase] = useState<Phase>("scan");
    86→    86→  const [secondsLeft, setSecondsLeft] = useState(DURATION_SEC);
    87→    87→  const [checkingNow, setCheckingNow] = useState(false);
    88→    88→  const [regenerating, setRegenerating] = useState(false);
    89→    89→  /** true saat simulasi dana masuk (sandbox) sedang diproses */
    90→    90→  const [simulating, setSimulating] = useState(false);
    91→    91→  /** pesan error simulasi (ditampilkan di bawah tombolnya) */
    92→    92→  const [simulateError, setSimulateError] = useState<string | null>(null);
    93→    93→  /** status terakhir dari penyedia (mis. "EXPIRED") */
    94→    94→  const [gatewayStatus, setGatewayStatus] = useState<string | null>(null);
    95→    95→  /** gangguan sementara saat memantau (diciutkan agar tidak bikin cemas) */
    96→    96→  const [monitorNote, setMonitorNote] = useState<string | null>(null);
    97→    97→  /** batas waktu pesanan (createdAt + 30 menit) dalam ms epoch */
    98→    98→  const [orderDeadline, setOrderDeadline] = useState<number | null>(null);
    99→    99→  const paidNotified = useRef(false);
   100→   100→  /** onPaid disimpan di ref: identitas prop baru dari re-render parent
   101→   101→   *  (mis. jam kasir yang berdetik tiap detik) TIDAK boleh memicu ulang
   102→   102→   *  efek fetch/polling di bawah — itulah penyebab layar QR berkedip. */
   103→   103→  const onPaidRef = useRef(onPaid);
   104→   104→  useEffect(() => {
   105→   105→    onPaidRef.current = onPaid;
   106→   106→  }, [onPaid]);
   107→   107→
   108→   108→  /** Stabil antar-render — aman dipakai sebagai dependensi efek */
   109→   109→  const notifyPaid = useCallback(() => {
   110→   110→    if (paidNotified.current) return;
   111→   111→    paidNotified.current = true;
   112→   112→    void onPaidRef.current?.();
   113→   113→  }, []);
   114→   114→
   115→   115→  /** Penomoran permintaan: respons yang datang terlambat dari fetch lama
   116→   116→   *  tidak boleh menimpa data fetch terbaru (cegah QR "maju-mundur"). */
   117→   117→  const fetchSeq = useRef(0);
   118→   118→  /** Ambil kode QRIS dari server (gateway bila aktif, fallback lokal).
   119→   119→   *  soft=true → QR lama TETAP tampil selagi data baru diambil (kode
   120→   120→   *  langsung tertukar saat respons tiba) sehingga layar tak berkedip. */
   121→   121→  const fetchPayload = useCallback(async (soft = false) => {
   122→   122→    const seq = ++fetchSeq.current;
   123→   123→    setLoadError(null);
   124→   124→    setGatewayStatus(null);
   125→   125→    if (!soft) setInfo(null);
   126→   126→    try {
   127→   127→      const res = await apiFetch(`/api/orders/${order.id}/qris`, { cache: "no-store" });
   128→   128→      const data = await res.json();
   129→   129→      if (seq !== fetchSeq.current) return; // respons kedaluwarsa — abaikan
   130→   130→      if (!res.ok) {
   131→   131→        // Pesanan ternyata sudah lunas → langsung tampilkan sukses
   132→   132→        if (typeof data.error === "string" && data.error.toLowerCase().includes("lunas")) {
   133→   133→          setPhase("paid");
   134→   134→          notifyPaid();
   135→   135→          return;
   136→   136→        }
   137→   137→        // Pesanan dibatalkan otomatis (lewat batas 30 menit) → tampilan batal
   138→   138→        if (data.cancelled === true) {
   139→   139→          setPhase("cancelled");
   140→   140→          return;
   141→   141→        }
   142→   142→        setLoadError(data.error ?? "Gagal memuat kode QRIS.");
   143→   143→        return;
   144→   144→      }
   145→   145→      if (data.paid === true) {
   146→   146→        setPhase("paid");
   147→   147→        notifyPaid();
   148→   148→        return;
   149→   149→      }
   150→   150→      const expiresAt = data.expiresAt ? new Date(data.expiresAt as string) : null;
   151→   151→      const cancelMs = data.cancelAt ? new Date(data.cancelAt as string).getTime() : null;
   152→   152→      if (cancelMs) setOrderDeadline(cancelMs);
   153→   153→      setInfo({
   154→   154→        payload: (data.payload as string) ?? null,
   155→   155→        imageUrl: (data.imageUrl as string) ?? null,
   156→   156→        mode: data.mode === "GATEWAY" ? "GATEWAY" : "LOCAL",
   157→   157→        autoVerify: !!data.autoVerify,
   158→   158→        sandbox: data.sandbox === true,
   159→   159→        provider: typeof data.provider === "string" ? data.provider : null,
   160→   160→        fallbackReason:
   161→   161→          typeof data.fallbackReason === "string" ? data.fallbackReason : null,
   162→   162→        sandboxSimulatorUrl:
   163→   163→          typeof data.sandboxSimulatorUrl === "string" ? data.sandboxSimulatorUrl : null,
   164→   164→      });
   165→   165→      // Hitung mundur pakai batas TERKETAT: umur kode gateway vs batas pesanan
   166→   166→      const nowMs = Date.now();
   167→   167→      const gatewaySec = expiresAt
   168→   168→        ? Math.max(0, Math.floor((expiresAt.getTime() - nowMs) / 1000))
   169→   169→        : DURATION_SEC;
   170→   170→      const orderSec = cancelMs
   171→   171→        ? Math.max(0, Math.floor((cancelMs - nowMs) / 1000))
   172→   172→        : DURATION_SEC;
   173→   173→      setSecondsLeft(Math.max(0, Math.min(DURATION_SEC, gatewaySec, orderSec)));
   174→   174→    } catch {
   175→   175→      if (seq !== fetchSeq.current) return; // respons kedaluwarsa — abaikan
   176→   176→      setLoadError("Tidak dapat menghubungi server. Periksa koneksi lalu coba lagi.");
   177→   177→    }
   178→   178→    // notifyPaid stabil ([]) — dependensi ini tidak berubah antar-render
   179→   179→    // parent, sehingga QR hanya diambil SEKALI per pesanan (bukan tiap detik).
   180→   180→  }, [order.id, notifyPaid]);
   181→   181→
   182→   182→  useEffect(() => {
   183→   183→    void fetchPayload();
   184→   184→  }, [fetchPayload]);
   185→   185→
   186→   186→  // Hitung waktu mundur (hanya saat fase scan)
   187→   187→  useEffect(() => {
   188→   188→    if (phase !== "scan") return;
   189→   189→    const t = setInterval(() => {
   190→   190→      setSecondsLeft((s) => (s > 0 ? s - 1 : 0));
   191→   191→    }, 1000);
   192→   192→    return () => clearInterval(t);
   193→   193→  }, [phase]);
   194→   194→
   195→   195→  // Tutup dengan tombol Escape
   196→   196→  useEffect(() => {
   197→   197→    const onKey = (e: KeyboardEvent) => {
   198→   198→      if (e.key === "Escape") onClose();
   199→   199→    };
   200→   200→    window.addEventListener("keydown", onKey);
   201→   201→    return () => window.removeEventListener("keydown", onKey);
   202→   202→  }, [onClose]);
   203→   203→
   204→   204→  // ===== Pemantauan verifikasi otomatis =====
   205→   205→  // hasInfo (boolean) sebagai dependensi — bukan objek info — agar interval
   206→   206→  // polling TIDAK dibongkar-pasang setiap data baru masuk.
   207→   207→  const hasInfo = info !== null;
   208→   208→  useEffect(() => {
   209→   209→    if (phase !== "scan" || !hasInfo) return;
   210→   210→    let alive = true;
   211→   211→    const poll = async (force = false) => {
   212→   212→      try {
   213→   213→        const res = await apiFetch(
   214→   214→          `/api/orders/${order.id}/qris/status${force ? "?force=1" : ""}`,
   215→   215→          { cache: "no-store" }
   216→   216→        );
   217→   217→        if (!res.ok || !alive) return;
   218→   218→        const data = await res.json();
   219→   219→        if (data.paymentStatus === "PAID") {
   220→   220→          setPhase("paid");
   221→   221→          notifyPaid();
   222→   222→          return;
   223→   223→        }
   224→   224→        // Pesanan batal otomatis (lewat batas 30 menit / dana telat)
   225→   225→        if (data.cancelled === true || data.status === "CANCELLED") {
   226→   226→          setMonitorNote((data.notice as string) ?? null);
   227→   227→          setPhase("cancelled");
   228→   228→          return;
   229→   229→        }
   230→   230→        if (data.cancelAt) setOrderDeadline(new Date(data.cancelAt as string).getTime());
   231→   231→        setGatewayStatus((data.gatewayStatus as string) ?? null);
   232→   232→        setMonitorNote((data.gatewayError as string) ?? null);
   233→   233→      } catch {
   234→   234→        // koneksi tersendat — coba lagi pada tick berikutnya
   235→   235→      }
   236→   236→    };
   237→   237→    void poll();
   238→   238→    const t = setInterval(() => void poll(), POLL_MS);
   239→   239→    return () => {
   240→   240→      alive = false;
   241→   241→      clearInterval(t);
   242→   242→    };
   243→   243→  }, [phase, hasInfo, order.id, notifyPaid]);
   244→   244→
   245→   245→  /** POS: paksa sistem memeriksa penyedia sekarang (tetap otomatis, bukan menandai lunas) */
   246→   246→  const checkNow = async () => {
   247→   247→    if (checkingNow) return;
   248→   248→    setCheckingNow(true);
   249→   249→    try {
   250→   250→      const res = await apiFetch(`/api/orders/${order.id}/qris/status?force=1`, {
   251→   251→        cache: "no-store",
   252→   252→      });
   253→   253→      const data = await res.json();
   254→   254→      if (res.ok && data.paymentStatus === "PAID") {
   255→   255→        setPhase("paid");
   256→   256→        notifyPaid();
   257→   257→      }
   258→   258→    } catch {
   259→   259→      /* abaikan — polling reguler akan menyusul */
   260→   260→    } finally {
   261→   261→      setCheckingNow(false);
   262→   262→    }
   263→   263→  };
   264→   264→
   265→   265→  /** Buat ulang kode bayar (transaksi lama kedaluwarsa) — soft: QR lama
   266→   266→   *  tetap tampil sampai kode baru siap, tanpa kedipan spinner. */
   267→   267→  const regenerate = async () => {
   268→   268→    if (regenerating) return;
   269→   269→    setRegenerating(true);
   270→   270→    await fetchPayload(true);
   271→   271→    setRegenerating(false);
   272→   272→  };
   273→   273→
   274→   274→  /** SANDBOX: tiru notifikasi "dana masuk" dari dompet digital ke gateway
   275→   275→   *  bawaan. Pesanan tetap dilunaskan OLEH SISTEM di server (jalur yang sama
   276→   276→   *  dengan verifikasi otomatis) — ini bukan tombol "tandai lunas" manual. */
   277→   277→  const simulatePay = async () => {
   278→   278→    if (simulating) return;
   279→   279→    setSimulating(true);
   280→   280→    setSimulateError(null);
   281→   281→    try {
   282→   282→      const res = await apiFetch(`/api/orders/${order.id}/qris/simulate-pay`, {
   283→   283→        method: "POST",
   284→   284→      });
   285→   285→      const data = await res.json();
   286→   286→      if (res.ok && data.paymentStatus === "PAID") {
   287→   287→        setPhase("paid");
   288→   288→        notifyPaid();
   289→   289→        return;
   290→   290→      }
   291→   291→      if (data.cancelled === true) {
   292→   292→        setMonitorNote(
   293→   293→          typeof data.error === "string" ? data.error : null
   294→   294→        );
   295→   295→        setPhase("cancelled");
   296→   296→        return;
   297→   297→      }
   298→   298→      setSimulateError(
   299→   299→        typeof data.error === "string" ? data.error : "Simulasi gagal — coba lagi."
   300→   300→      );
   301→   301→    } catch {
   302→   302→      setSimulateError("Tidak dapat menghubungi server. Periksa koneksi lalu coba lagi.");
   303→   303→    } finally {
   304→   304→      setSimulating(false);
   305→   305→    }
   306→   306→  };
   307→   307→
   308→   308→  const expired = secondsLeft <= 0;
   309→   309→  const codeStale = expired || gatewayStatus === "EXPIRED";
   310→   310→  const autoVerify = info?.autoVerify ?? false;
   311→   311→  /** Batas pesanan (30 menit) sudah terlewat — pesanan mati, jangan tawarkan buat ulang kode */
   312→   312→  const pastDeadline = orderDeadline !== null && Date.now() >= orderDeadline;
   313→   313→
   314→   314→  // Portal ke <body> agar modal tidak terjebak stacking context `main` (z-10)
   315→   315→  // — selalu tampil DI ATAS navigasi bawah liquid glass (z-50) & header (z-40).
   316→   316→  return createPortal(
   317→   317→    <AnimatePresence>
   318→   318→      <motion.div
   319→   319→        key="qris-overlay"
   320→   320→        initial={prefersReduced ? false : { opacity: 0 }}
   321→   321→        animate={{ opacity: 1 }}
   322→   322→        exit={prefersReduced ? undefined : { opacity: 0 }}
   323→   323→        className="fixed inset-0 z-[95] flex items-end justify-center bg-neutral-900/70 backdrop-blur-sm sm:items-center sm:p-4"
   324→   324→        role="dialog"
   325→   325→        aria-modal="true"
   326→   326→        aria-label="Pembayaran QRIS"
   327→   327→      >
   328→   328→        <motion.div
   329→   329→          initial={prefersReduced ? false : { y: 48, scale: 0.97 }}
   330→   330→          animate={{ y: 0, scale: 1 }}
   331→   331→          exit={prefersReduced ? undefined : { y: 48, opacity: 0 }}
   332→   332→          transition={{ type: "spring", stiffness: 380, damping: 34 }}
   333→   333→          className="flex max-h-[94dvh] w-full max-w-md flex-col overflow-hidden rounded-t-3xl bg-white shadow-2xl sm:rounded-3xl"
   334→   334→        >
   335→   335→          {/* ===== KEPALA ===== */}
   336→   336→          <header className="relative shrink-0 bg-gradient-to-r from-orange-700 via-orange-600 to-orange-500 px-5 py-4 text-white">
   337→   337→            <div className="flex items-center gap-3">
   338→   338→              <span className="flex h-10 w-10 items-center justify-center rounded-2xl bg-white/20 ring-1 ring-white/30">
   339→   339→                <ScanLine className="h-5 w-5" />
   340→   340→              </span>
   341→   341→              <div className="min-w-0 flex-1">
   342→   342→                <p className="text-sm font-black tracking-wide">Pembayaran QRIS</p>
   343→   343→                <p className="truncate text-[11px] font-semibold text-orange-100">
   344→   344→                  {order.code} • {isPos ? "Kasir" : "Aplikasi"}
   345→   345→                </p>
   346→   346→              </div>
   347→   347→              <button
   348→   348→                onClick={onClose}
   349→   349→                aria-label="Tutup"
   350→   350→                className="press-effect flex h-9 w-9 items-center justify-center rounded-full bg-white/15 hover:bg-white/30"
   351→   351→              >
   352→   352→                <X className="h-4.5 w-4.5" />
   353→   353→              </button>
   354→   354→            </div>
   355→   355→          </header>
   356→   356→
   357→   357→          {/* ===== ISI ===== */}
   358→   358→          <div className="scrollbar-slim overflow-y-auto px-5 pb-[max(1.25rem,env(safe-area-inset-bottom))] pt-4">
   359→   359→            {phase === "paid" ? (
   360→   360→              /* --------- BERHASIL (verifikasi otomatis) --------- */
   361→   361→              <div className="flex flex-col items-center gap-3 py-8 text-center">
   362→   362→                <motion.span
   363→   363→                  initial={prefersReduced ? false : { scale: 0.5, opacity: 0 }}
   364→   364→                  animate={{ scale: 1, opacity: 1 }}
   365→   365→                  transition={{ type: "spring", stiffness: 300, damping: 18 }}
   366→   366→                  className="flex h-20 w-20 items-center justify-center rounded-full bg-emerald-50 ring-4 ring-emerald-100"
   367→   367→                >
   368→   368→                  <BadgeCheck className="h-11 w-11 text-emerald-600" />
   369→   369→                </motion.span>
   370→   370→                <h2 className="text-lg font-black text-neutral-900">Pembayaran Diterima! 🎉</h2>
   371→   371→                <p className="max-w-[300px] text-xs leading-relaxed text-neutral-500">
   372→   372→                  {formatIDR(order.total)} untuk pesanan {order.code} sudah lunas —
   373→   373→                  terverifikasi <strong className="font-bold text-emerald-600">otomatis oleh sistem</strong>{" "}
   374→   374→                  saat dana masuk. Geprekmu segera dimasak ya!
   375→   375→                </p>
   376→   376→                {isPos && onPrintReceipt ? (
   377→   377→                  <div className="mt-2 flex w-full items-center gap-2">
   378→   378→                    <button
   379→   379→                      onClick={onClose}
   380→   380→                      className="press-effect h-11 flex-1 rounded-xl border border-neutral-200 bg-white text-sm font-extrabold text-neutral-600 hover:bg-neutral-50"
   381→   381→                    >
   382→   382→                      Selesai
   383→   383→                    </button>
   384→   384→                    <button
   385→   385→                      onClick={() => void onPrintReceipt()}
   386→   386→                      className="press-effect flex h-11 flex-[1.6] items-center justify-center gap-2 rounded-xl bg-orange-600 text-sm font-extrabold text-white shadow-lg shadow-orange-500/30 hover:bg-orange-700"
   387→   387→                    >
   388→   388→                      <Printer className="h-4 w-4" /> Cetak Struk
   389→   389→                    </button>
   390→   390→                  </div>
   391→   391→                ) : (
   392→   392→                  <button
   393→   393→                    onClick={onClose}
   394→   394→                    className="press-effect mt-2 w-full rounded-xl bg-orange-600 py-3 text-sm font-extrabold text-white shadow-lg shadow-orange-500/30 hover:bg-orange-700"
   395→   395→                  >
   396→   396→                    Selesai
   397→   397→                  </button>
   398→   398→                )}
   399→   399→              </div>
   400→   400→            ) : phase === "cancelled" ? (
   401→   401→              /* --------- DIBATALKAN OTOMATIS (belum bayar 30 menit) --------- */
   402→   402→              <div className="flex flex-col items-center gap-3 py-8 text-center">
