     1→"use client";
     2→
     3→import { useCallback, useEffect, useRef, useState } from "react";
     4→import { createPortal } from "react-dom";
     5→import { AnimatePresence, motion, useReducedMotion } from "framer-motion";
     6→import { QRCodeSVG } from "qrcode.react";
     7→import {
     8→  BadgeCheck,
     9→  ExternalLink,
    10→  FlaskConical,
    11→  Loader2,
    12→  Printer,
    13→  RefreshCw,
    14→  ScanLine,
    15→  ShieldCheck,
    16→  Timer,
    17→  TriangleAlert,
    18→  X,
    19→  XCircle,
    20→} from "lucide-react";
    21→import { apiFetch } from "@/lib/api";

    24→import { cn } from "@/lib/utils";
    25→
    26→/** Pesanan yang akan dibayar lewat gateway (QRIS / E-Wallet) */
    27→export type QrisPayOrder = {
    28→  id: string;
    29→  code: string;
    30→  total: number;
    31→  /** Metode digital: "QRIS" (semua kanal) | "EWALLET" (halaman DOKU
    32→   *  difilter ke kanal e-wallet). Default "QRIS" bila tidak dikirim. */
    33→  method?: "QRIS" | "EWALLET";
    34→};
    35→
    36→type Phase = "scan" | "paid" | "cancelled";
    37→
    38→type QrisInfo = {
    39→  payload: string | null;
    40→  imageUrl: string | null;
    41→  /** Jalur DOKU Checkout: URL halaman pembayaran DOKU (status tetap dipantau) */
    42→  paymentUrl: string | null;
    43→  mode: "GATEWAY" | "LOCAL";
    44→  autoVerify: boolean;
    45→  /** gateway QRIS bawaan (sandbox) aktif — tombol simulasi dana ditampilkan */
    46→  sandbox: boolean;
    47→  /** penyedia verifikasi: "casaku" | "midtrans" | "midtrans-sandbox" | "internal" | null */
    48→  provider: string | null;
    49→  /** Alasan penyedia utama (mis. Casaku/Midtrans) ditolak sehingga QR berasal dari fallback uji lokal */
    50→  fallbackReason: string | null;
    51→  /** Galat gateway saat mode LOCAL — ditampilkan jujur (kanal belum aktif, dll.) */
    52→  gatewayError: string | null;
    53→  /** Sandbox Midtrans: tautan simulator uji pembayaran (QR berisi URL uji) */
    54→  sandboxSimulatorUrl: string | null;
    55→  /** Nominal FINAL yang harus dibayar (total + kode unik Casaku bila aktif) */
    56→  payAmount: number | null;
    57→  /** Kode unik yang disematkan Casaku ke nominal (bila aktif) */
    58→  uniqueCode: number | null;
    59→};
    60→
    61→/** Interval polling verifikasi otomatis */
    62→const POLL_MS = 4000;
    63→/** Batas waktu pembayaran QRIS — 5 MENIT sejak pesanan dibuat (permintaan
    64→ *  pemilik). Lewat dari itu pesanan batal otomatis (server membatalkan
    65→ *  saat sweep/poll) dan kode bayar hangus. */
    66→const DURATION_SEC = 5 * 60; // 5 menit
    67→
    68→function mmss(sec: number): string {
    69→  const m = Math.floor(sec / 60);
    70→  const s = sec % 60;
    71→  return `${String(m).padStart(2, "0")}:${String(s).padStart(2, "0")}`;
    72→}
    73→
    74→/**
    75→ * Layar pembayaran QRIS — pelanggan maupun kasir memakai layar yang sama.
    76→ *
    77→ * VERIFIKASI OTOMATIS: setelah QR tampil, layar ini memantau status ke server
    78→ * tiap beberapa detik; server bertanya ke penyedia QRIS "sudah masuk belum
    79→ * dananya?" dan melunaskan pesanan begitu dana masuk. TIDAK ADA tombol
    80→ * "Saya Sudah Bayar" / "Uang Sudah Masuk" — kasir memang tidak boleh
    81→ * (dan tidak bisa) melunaskan pesanan QRIS secara manual.
    82→ */
    83→export function QrisPaySheet({
    84→  order,
    85→  variant,
    86→  onClose,
    87→  onPaid,
    88→  onPrintReceipt,
    89→}: {
    90→  order: QrisPayOrder;
    91→  variant: "customer" | "pos";
    92→  /** Dipanggil saat layar ditutup. result.paid memberi tahu apakah pembayaran
    93→   *  SUDAH lunas saat ditutup (false = belum dibayar / sudah batal) —
    94→   *  pemanggil TIDAK boleh menampilkan pesan sukses bila paid=false. */
    95→  onClose: (result: { paid: boolean }) => void;
    96→  /** dipanggil saat status pembayaran berubah menjadi lunas (mis. refresh daftar) */
    97→  onPaid?: () => void | Promise<void>;
    98→  /** POS: cetak struk setelah lunas (bila disediakan halaman pemanggil) */
    99→  onPrintReceipt?: () => void | Promise<void>;
   100→}) {
   101→  const prefersReduced = useReducedMotion();
   102→  const isPos = variant === "pos";
   103→
   104→  const [info, setInfo] = useState<QrisInfo | null>(null);
   105→  const [loadError, setLoadError] = useState<string | null>(null);
   106→  const [phase, setPhase] = useState<Phase>("scan");
   107→  const [secondsLeft, setSecondsLeft] = useState(DURATION_SEC);
   108→  const [checkingNow, setCheckingNow] = useState(false);
   109→  const [regenerating, setRegenerating] = useState(false);
   110→  /** status terakhir dari penyedia (mis. "EXPIRED") */
   111→  const [gatewayStatus, setGatewayStatus] = useState<string | null>(null);
   112→  /** gangguan sementara saat memantau (diciutkan agar tidak bikin cemas) */
   113→  const [monitorNote, setMonitorNote] = useState<string | null>(null);
   114→  /** batas waktu pesanan (createdAt + 5 menit) dalam ms epoch */
   115→  const [orderDeadline, setOrderDeadline] = useState<number | null>(null);
   116→  const paidNotified = useRef(false);
   117→  /** onPaid disimpan di ref: identitas prop baru dari re-render parent
   118→   *  (mis. jam kasir yang berdetik tiap detik) TIDAK boleh memicu ulang
   119→   *  efek fetch/polling di bawah — itulah penyebab layar QR berkedip. */
   120→  const onPaidRef = useRef(onPaid);
   121→  useEffect(() => {
   122→    onPaidRef.current = onPaid;
   123→  }, [onPaid]);
   124→
   125→  /** Stabil antar-render — aman dipakai sebagai dependensi efek */
   126→  const notifyPaid = useCallback(() => {
   127→    if (paidNotified.current) return;
   128→    paidNotified.current = true;
   129→    void onPaidRef.current?.();
   130→  }, []);
   131→
   132→  /** Penomoran permintaan: respons yang datang terlambat dari fetch lama
   133→   *  tidak boleh menimpa data fetch terbaru (cegah QR "maju-mundur"). */
   134→  const fetchSeq = useRef(0);
   135→  /** Ambil kode QRIS dari server (gateway bila aktif, fallback lokal).
   136→   *  soft=true → QR lama TETAP tampil selagi data baru diambil (kode
   137→   *  langsung tertukar saat respons tiba) sehingga layar tak berkedip. */
   138→  const fetchPayload = useCallback(async (soft = false) => {
   139→    const seq = ++fetchSeq.current;
   140→    setLoadError(null);
   141→    setGatewayStatus(null);
   142→    if (!soft) setInfo(null);
   143→    try {
   144→      const res = await apiFetch(`/api/orders/${order.id}/qris`, { cache: "no-store" });
   145→      const data = await res.json();
   146→      if (seq !== fetchSeq.current) return; // respons kedaluwarsa — abaikan
   147→      if (!res.ok) {
   148→        // Pesanan ternyata sudah lunas → langsung tampilkan sukses
   149→        if (typeof data.error === "string" && data.error.toLowerCase().includes("lunas")) {
   150→          setPhase("paid");
   151→          notifyPaid();
   152→          return;
   153→        }
   154→        // Pesanan dibatalkan otomatis (lewat batas 5 menit) → tampilan batal
   155→        if (data.cancelled === true) {
   156→          setPhase("cancelled");
   157→          return;
   158→        }
   159→        setLoadError(data.error ?? "Gagal memuat kode QRIS.");
   160→        return;
   161→      }
   162→      if (data.paid === true) {
   163→        setPhase("paid");
   164→        notifyPaid();
   165→        return;
   166→      }
   167→      const expiresAt = data.expiresAt ? new Date(data.expiresAt as string) : null;
   168→      const cancelMs = data.cancelAt ? new Date(data.cancelAt as string).getTime() : null;
   169→      if (cancelMs) setOrderDeadline(cancelMs);
   170→      setInfo({
   171→        payload: (data.payload as string) ?? null,
   172→        imageUrl: (data.imageUrl as string) ?? null,
   173→        paymentUrl:
   174→          typeof data.paymentUrl === "string" ? data.paymentUrl : null,
   175→        mode: data.mode === "GATEWAY" ? "GATEWAY" : "LOCAL",
   176→        autoVerify: !!data.autoVerify,
   177→        sandbox: data.sandbox === true,
   178→        provider: typeof data.provider === "string" ? data.provider : null,
   179→        fallbackReason:
   180→          typeof data.fallbackReason === "string" ? data.fallbackReason : null,
   181→        gatewayError:
   182→          typeof data.gatewayError === "string" ? data.gatewayError : null,
   183→        sandboxSimulatorUrl:
   184→          typeof data.sandboxSimulatorUrl === "string" ? data.sandboxSimulatorUrl : null,
   185→        payAmount: typeof data.amount === "number" ? data.amount : null,
   186→        uniqueCode: typeof data.uniqueCode === "number" ? data.uniqueCode : null,
   187→      });
   188→      // Hitung mundur pakai batas TERKETAT: umur kode gateway vs batas pesanan
   189→      const nowMs = Date.now();
   190→      const gatewaySec = expiresAt
   191→        ? Math.max(0, Math.floor((expiresAt.getTime() - nowMs) / 1000))
   192→        : DURATION_SEC;
   193→      const orderSec = cancelMs
   194→        ? Math.max(0, Math.floor((cancelMs - nowMs) / 1000))
   195→        : DURATION_SEC;
   196→      setSecondsLeft(Math.max(0, Math.min(DURATION_SEC, gatewaySec, orderSec)));
   197→    } catch {
   198→      if (seq !== fetchSeq.current) return; // respons kedaluwarsa — abaikan
   199→      setLoadError("Tidak dapat menghubungi server. Periksa koneksi lalu coba lagi.");
   200→    }
   201→    // notifyPaid stabil ([]) — dependensi ini tidak berubah antar-render
   202→    // parent, sehingga QR hanya diambil SEKALI per pesanan (bukan tiap detik).
   203→  }, [order.id, notifyPaid]);
   204→
   205→  useEffect(() => {
   206→    void fetchPayload();
   207→  }, [fetchPayload]);
   208→
   209→  // Hitung waktu mundur (hanya saat fase scan)
   210→  useEffect(() => {
   211→    if (phase !== "scan") return;
   212→    const t = setInterval(() => {
   213→      setSecondsLeft((s) => (s > 0 ? s - 1 : 0));
   214→    }, 1000);
   215→    return () => clearInterval(t);
   216→  }, [phase]);
   217→
   218→  /** Tutup layar pembayaran + notifikasi sesuai hasil pembayaran terakhir
   219→   *  (permintaan pemilik):
   220→   *  - belum lunas  -> "Pembayaran belum berhasil — silakan lanjutkan
   221→   *                    pembayaran" (merah, agar kasir/pelanggan tak bingung)
   222→   *  - sudah lunas  -> "Pembayaran berhasil — pesanan Anda sedang diproses"
   223→   *  - sudah batal  -> cukup tutup (layar batal sudah menjelaskan) */
   224→  const requestClose = useCallback(() => {
   225→    if (phase === "paid") {
   226→      toast({
   227→        title: "Pembayaran berhasil",
   228→        description: `Pesanan ${order.code} ${formatIDR(order.total)} sudah terkonfirmasi — pesanan Anda sedang diproses.`,
   229→        duration: 7000,
   230→      });
   231→    } else if (phase === "scan") {
   232→      toast({
   233→        variant: "destructive",
   234→        title: "Pembayaran belum berhasil",
   235→        description: "Silakan lanjutkan pembayaran QRIS sebelum waktu habis — pesanan belum terlunasi.",
   236→        duration: 7000,
   237→      });
   238→    }
   239→    onClose({ paid: phase === "paid" });
   240→  }, [phase, order.code, order.total, onClose]);
   241→
   242→  // Tutup dengan tombol Escape (lewat requestClose: notifikasi status ikut tampil)
   243→  useEffect(() => {
   244→    const onKey = (e: KeyboardEvent) => {
   245→      if (e.key === "Escape") requestClose();
   246→    };
   247→    window.addEventListener("keydown", onKey);
   248→    return () => window.removeEventListener("keydown", onKey);
   249→  }, [requestClose]);
   250→
   251→  // ===== Pemantauan verifikasi otomatis =====
   252→  // hasInfo (boolean) sebagai dependensi — bukan objek info — agar interval
   253→  // polling TIDAK dibongkar-pasang setiap data baru masuk.
   254→  const hasInfo = info !== null;
   255→  useEffect(() => {
   256→    if (phase !== "scan" || !hasInfo) return;
   257→    let alive = true;
   258→    const poll = async (force = false) => {
   259→      try {
   260→        const res = await apiFetch(
   261→          `/api/orders/${order.id}/qris/status${force ? "?force=1" : ""}`,
   262→          { cache: "no-store" }
   263→        );
   264→        if (!res.ok || !alive) return;
   265→        const data = await res.json();
   266→        if (data.paymentStatus === "PAID") {
   267→          setPhase("paid");
   268→          notifyPaid();
   269→          return;
   270→        }
   271→        // Pesanan batal otomatis (lewat batas 5 menit / dana telat)
   272→        if (data.cancelled === true || data.status === "CANCELLED") {
   273→          setMonitorNote((data.notice as string) ?? null);
   274→          setPhase("cancelled");
   275→          return;
   276→        }
   277→        if (data.cancelAt) setOrderDeadline(new Date(data.cancelAt as string).getTime());
   278→        setGatewayStatus((data.gatewayStatus as string) ?? null);
   279→        setMonitorNote((data.gatewayError as string) ?? null);
   280→      } catch {
   281→        // koneksi tersendat — coba lagi pada tick berikutnya
   282→      }
   283→    };
   284→    void poll();
   285→    const t = setInterval(() => void poll(), POLL_MS);
   286→    return () => {
   287→      alive = false;
   288→      clearInterval(t);
   289→    };
   290→  }, [phase, hasInfo, order.id, notifyPaid]);
   291→
   292→  /** POS: paksa sistem memeriksa penyedia sekarang (tetap otomatis, bukan menandai lunas) */
   293→  const checkNow = async () => {
   294→    if (checkingNow) return;
   295→    setCheckingNow(true);
   296→    try {
   297→      const res = await apiFetch(`/api/orders/${order.id}/qris/status?force=1`, {
   298→        cache: "no-store",
   299→      });
   300→      const data = await res.json();
   301→      if (res.ok && data.paymentStatus === "PAID") {
   302→        setPhase("paid");
   303→        notifyPaid();
   304→      }
   305→    } catch {
   306→      /* abaikan — polling reguler akan menyusul */
   307→    } finally {
   308→      setCheckingNow(false);
   309→    }
   310→  };
   311→
   312→  /** Buat ulang kode bayar (transaksi lama kedaluwarsa) — soft: QR lama
   313→   *  tetap tampil sampai kode baru siap, tanpa kedipan spinner. */
   314→  const regenerate = async () => {
   315→    if (regenerating) return;
   316→    setRegenerating(true);
   317→    await fetchPayload(true);
   318→    setRegenerating(false);
   319→  };
   320→
   321→  const expired = secondsLeft <= 0;
   322→  const codeStale = expired || gatewayStatus === "EXPIRED";
   323→  const autoVerify = info?.autoVerify ?? false;
   324→  /** Batas pesanan (5 menit) sudah terlewat — pesanan mati, jangan tawarkan buat ulang kode */
   325→  const pastDeadline = orderDeadline !== null && Date.now() >= orderDeadline;
   326→
   327→  // Portal ke <body> agar modal tidak terjebak stacking context `main` (z-10)
   328→  // — selalu tampil DI ATAS navigasi bawah liquid glass (z-50) & header (z-40).
   329→  return createPortal(
   330→    <AnimatePresence>
   331→      <motion.div
   332→        key="qris-overlay"
   333→        initial={prefersReduced ? false : { opacity: 0 }}
   334→        animate={{ opacity: 1 }}
   335→        exit={prefersReduced ? undefined : { opacity: 0 }}
   336→        className="fixed inset-0 z-[95] flex items-end justify-center bg-neutral-900/70 backdrop-blur-sm sm:items-center sm:p-4"
   337→        role="dialog"
   338→        aria-modal="true"
   339→        aria-label={order.method === "EWALLET" ? "Pembayaran E-Wallet" : "Pembayaran QRIS"}
   340→      >
   341→        <motion.div
   342→          initial={prefersReduced ? false : { y: 48, scale: 0.97 }}
   343→          animate={{ y: 0, scale: 1 }}
   344→          exit={prefersReduced ? undefined : { y: 48, opacity: 0 }}
   345→          transition={{ type: "spring", stiffness: 380, damping: 34 }}
   346→          className="flex max-h-[94dvh] w-full max-w-md flex-col overflow-hidden rounded-t-3xl bg-white shadow-2xl sm:rounded-3xl"
   347→        >
   348→          {/* ===== KEPALA ===== */}
   349→          <header className="relative shrink-0 bg-gradient-to-r from-orange-700 via-orange-600 to-orange-500 px-5 py-4 text-white">
   350→            <div className="flex items-center gap-3">
   351→              <span className="flex h-10 w-10 items-center justify-center rounded-2xl bg-white/20 ring-1 ring-white/30">
   352→                <ScanLine className="h-5 w-5" />
   353→              </span>
   354→              <div className="min-w-0 flex-1">
   355→                <p className="text-sm font-black tracking-wide">
   356→                  {order.method === "EWALLET" ? "Pembayaran E-Wallet" : "Pembayaran QRIS"}
   357→                </p>
   358→                <p className="truncate text-[11px] font-semibold text-orange-100">
   359→                  {order.code} • {isPos ? "Kasir" : "Aplikasi"}
   360→                </p>
   361→              </div>
   362→              <button
   363→                onClick={requestClose}
   364→                aria-label="Tutup"
   365→                className="press-effect flex h-9 w-9 items-center justify-center rounded-full bg-white/15 hover:bg-white/30"
   366→              >
   367→                <X className="h-4.5 w-4.5" />
   368→              </button>
   369→            </div>
   370→          </header>
   371→
   372→          {/* ===== ISI ===== */}
   373→          <div className="scrollbar-slim overflow-y-auto px-5 pb-[max(1.25rem,env(safe-area-inset-bottom))] pt-4">
   374→            {phase === "paid" ? (
   375→              /* --------- BERHASIL (verifikasi otomatis) --------- */
   376→              <div className="flex flex-col items-center gap-3 py-8 text-center">
   377→                <motion.span
   378→                  initial={prefersReduced ? false : { scale: 0.5, opacity: 0 }}
   379→                  animate={{ scale: 1, opacity: 1 }}
   380→                  transition={{ type: "spring", stiffness: 300, damping: 18 }}
   381→                  className="flex h-20 w-20 items-center justify-center rounded-full bg-emerald-50 ring-4 ring-emerald-100"
   382→                >
   383→                  <BadgeCheck className="h-11 w-11 text-emerald-600" />
   384→                </motion.span>
   385→                <h2 className="text-lg font-black text-neutral-900">Pembayaran Diterima! 🎉</h2>
   386→                <p className="max-w-[300px] text-xs leading-relaxed text-neutral-500">
   387→                  {formatIDR(order.total)} untuk pesanan {order.code} sudah lunas —
   388→                  terverifikasi <strong className="font-bold text-emerald-600">otomatis oleh sistem</strong>{" "}
   389→                  saat dana masuk. Geprekmu segera dimasak ya!
   390→                </p>
   391→                {isPos && onPrintReceipt ? (
   392→                  <div className="mt-2 flex w-full items-center gap-2">
   393→                    <button
   394→                      onClick={requestClose}
   395→                      className="press-effect h-11 flex-1 rounded-xl border border-neutral-200 bg-white text-sm font-extrabold text-neutral-600 hover:bg-neutral-50"
   396→                    >
   397→                      Selesai
   398→                    </button>
   399→                    <button
   400→                      onClick={() => void onPrintReceipt()}
   401→                      className="press-effect flex h-11 flex-[1.6] items-center justify-center gap-2 rounded-xl bg-orange-600 text-sm font-extrabold text-white shadow-lg shadow-orange-500/30 hover:bg-orange-700"
   402→                    >
   403→                      <Printer className="h-4 w-4" /> Cetak Struk
   404→                    </button>
   405→                  </div>
   406→                ) : (
   407→                  <button
   408→                    onClick={requestClose}
   409→                    className="press-effect mt-2 w-full rounded-xl bg-orange-600 py-3 text-sm font-extrabold text-white shadow-lg shadow-orange-500/30 hover:bg-orange-700"
   410→                  >
   411→                    Selesai
   412→                  </button>
   413→                )}
   414→              </div>
   415→            ) : phase === "cancelled" ? (
   416→              /* --------- DIBATALKAN OTOMATIS (belum bayar 5 menit) --------- */
   417→              <div className="flex flex-col items-center gap-3 py-8 text-center">
   418→                <motion.span
   419→                  initial={prefersReduced ? false : { scale: 0.5, opacity: 0 }}
   420→                  animate={{ scale: 1, opacity: 1 }}
   421→                  transition={{ type: "spring", stiffness: 300, damping: 18 }}
   422→                  className="flex h-20 w-20 items-center justify-center rounded-full bg-red-50 ring-4 ring-red-100"
   423→                >
   424→                  <XCircle className="h-11 w-11 text-red-500" />
   425→                </motion.span>
   426→                <h2 className="text-lg font-black text-neutral-900">Pesanan Dibatalkan Otomatis</h2>
   427→                <p className="max-w-[320px] text-xs leading-relaxed text-neutral-500">
   428→                  Pembayaran untuk pesanan <strong className="font-bold text-neutral-700">{order.code}</strong>{" "}
   429→                  tidak diterima dalam <strong className="font-bold text-red-500">5 menit</strong>, jadi
   430→                  pesanan dibatalkan. Voucher yang terpakai sudah dikembalikan ke akunmu.
   431→                </p>
   432→                {monitorNote && (
   433→                  <p className="max-w-[320px] rounded-xl bg-amber-50 p-2.5 text-[11px] font-semibold leading-snug text-amber-700 ring-1 ring-amber-200">
   434→                    {monitorNote}
   435→                  </p>
   436→                )}
   437→                <p className="text-[11px] text-neutral-400">
   438→                  Masih pengin? Buat pesanan baru — dapur selalu siap menggeprek 😊
   439→                </p>
   440→                <button
   441→                  onClick={() => onClose({ paid: false })}
   442→                  className="press-effect mt-2 w-full rounded-xl bg-orange-600 py-3 text-sm font-extrabold text-white shadow-lg shadow-orange-500/30 hover:bg-orange-700"
   443→                >
   444→                  Tutup
   445→                </button>
   446→              </div>
   447→            ) : (
   448→              /* --------- PINDAI QR --------- */
   449→              <>
   450→                <div className="flex items-end justify-between gap-3">
   451→                  <div>
   452→                    <p className="text-[11px] font-bold uppercase tracking-wide text-neutral-400">
   453→                      Total Bayar
   454→                    </p>
