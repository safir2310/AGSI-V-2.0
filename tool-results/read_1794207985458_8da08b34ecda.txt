    51→
    52→const ORDER_TYPES = [
    53→  { id: "DINE_IN", label: "Makan di Tempat", icon: Utensils, placeholderTable: true },
    54→  { id: "TAKEAWAY", label: "Bungkus / Ambil", icon: ShoppingBag, placeholderTable: false },
    55→  { id: "DELIVERY", label: "Antar ke Alamat", icon: Bike, placeholderTable: false },
    56→] as const;
    57→
    58→const PAYMENTS = [
    59→  { id: "CASH", label: "Tunai", emoji: "💵" },
    60→  { id: "QRIS", label: "QRIS", emoji: "📱" },
    61→] as const;
    62→
    63→export function CartView({ user, go }: { user: AppUser; go: (v: AppView) => void }) {
    64→  const cart = useCartStore();
    65→  const showNotice = useNoticeStore((s) => s.show);
    66→  const prefersReduced = useReducedMotion();
    67→  const products = useProductsStore((s) => s.products);
    68→  const vouchers = useVouchersStore((s) => s.vouchers);
    69→  const loadVouchers = useVouchersStore((s) => s.load);
    70→  const pendingCode = useVouchersStore((s) => s.pendingCode);
    71→  const setPendingCode = useVouchersStore((s) => s.setPendingCode);
    72→  // Saklar layanan toko (Profil Admin → Pengaturan Toko):
    73→  // nonaktif → opsi antar / QRIS disembunyikan & pilihan lama di-reset
    74→  const deliveryEnabled = useSettingsStore((s) => s.deliveryEnabled);
    75→  const qrisEnabled = useSettingsStore((s) => s.qrisEnabled);
    76→  const [submitting, setSubmitting] = useState(false);
    77→  /** Pesanan QRIS yang sedang dibayar (layar QR terbuka) */
    78→  const [payOrder, setPayOrder] = useState<QrisPayOrder | null>(null);
    79→  /** Popup pilih alamat pengantaran (area Kec. Pidie 24151) */
    80→  const [areaOpen, setAreaOpen] = useState(false);
    81→  const [form, setForm] = useState({
    82→    name: user.name ?? "",
    83→    type: "DINE_IN" as (typeof ORDER_TYPES)[number]["id"],
    84→    tableNo: "",
    85→    address: "",
    86→    areaId: "",
    87→    phone: user.phone ?? "",
    88→    note: "",
    89→    payment: "CASH" as (typeof PAYMENTS)[number]["id"],
    90→  });
    91→
    92→  // ===== Voucher =====
    93→  const [voucherInput, setVoucherInput] = useState("");
    94→  const [appliedVoucher, setAppliedVoucher] = useState<VoucherInfo | null>(null);
    95→  const [voucherChecking, setVoucherChecking] = useState(false);
    96→  const [voucherError, setVoucherError] = useState<string | null>(null);
    97→
    98→  const subtotal = cart.totalPrice();
    99→
   100→  // Layanan dimatikan admin → pilihan lama otomatis dikembalikan
   101→  useEffect(() => {
   102→    if (!deliveryEnabled) {
   103→      setForm((f) => (f.type === "DELIVERY" ? { ...f, type: "DINE_IN", areaId: "", address: "" } : f));
   104→    }
   105→    if (!qrisEnabled) {
   106→      setForm((f) => (f.payment === "QRIS" ? { ...f, payment: "CASH" } : f));
   107→    }
   108→  }, [deliveryEnabled, qrisEnabled]);
   109→
   110→  /** Opsi jenis pesanan & pembayaran yang tampil — mengikuti saklar admin */
   111→  const availableTypes = ORDER_TYPES.filter((t) => t.id !== "DELIVERY" || deliveryEnabled);
   112→  const availablePayments = PAYMENTS.filter((p) => p.id !== "QRIS" || qrisEnabled);
   113→
   114→  /** Area pengantaran terpilih + ongkir — Rp1.000/km, kelipatan jarak
   115→   *  (jarak dibulatkan KE ATAS per 1 km, dihitung ulang juga oleh server) */
   116→  const deliveryArea =
   117→    form.type === "DELIVERY" && form.areaId ? findDeliveryArea(form.areaId) : undefined;
   118→  const deliveryFee =
   119→    form.type === "DELIVERY" && deliveryArea ? (feeForArea(deliveryArea.id) ?? 0) : 0;
   120→
   121→  /** Potongan dihitung ulang otomatis mengikuti isi keranjang */
   122→  const voucherDiscount = useMemo(() => {
   123→    if (!appliedVoucher) return 0;
   124→    return computeVoucherDiscount(appliedVoucher, subtotal);
   125→  }, [appliedVoucher, subtotal]);
   126→
   127→  const total = Math.max(0, subtotal - voucherDiscount + deliveryFee);
   128→
   129→  // Muat daftar voucher untuk saran cepat di bawah input kode
   130→  useEffect(() => {
   131→    void loadVouchers();
   132→  }, [loadVouchers]);
   133→
   134→  // Lepas voucher bila belanja turun di bawah minimum
   135→  useEffect(() => {
   136→    if (appliedVoucher && subtotal < appliedVoucher.minPurchase) {
   137→      setVoucherError(
   138→        `Voucher ${appliedVoucher.code} dilepas — belanja minimum ${formatIDR(appliedVoucher.minPurchase)}.`
   139→      );
   140→      setAppliedVoucher(null);
   141→      setVoucherInput("");
   142→    }
   143→  }, [subtotal, appliedVoucher]);
   144→
   145→  // Keranjang kosong → bersihkan voucher juga
   146→  useEffect(() => {
   147→    if (cart.items.length === 0) {
   148→      setAppliedVoucher(null);
   149→      setVoucherError(null);
   150→    }
   151→  }, [cart.items.length]);
   152→
   153→  /** Tukar kode voucher → validasi server → potongan otomatis */
   154→  const applyVoucher = async (rawCode?: string) => {
   155→    const code = (rawCode ?? voucherInput).trim().toUpperCase();
   156→    if (!code) {
   157→      setVoucherError("Masukkan dulu kode vouchernya ya.");
   158→      return;
   159→    }
   160→    setVoucherChecking(true);
   161→    setVoucherError(null);
   162→    try {
   163→      const res = await apiFetch("/api/vouchers/apply", {
   164→        method: "POST",
   165→        headers: { "Content-Type": "application/json" },
   166→        body: JSON.stringify({ code, subtotal }),
   167→      });
   168→      const data = await res.json();
   169→      if (!res.ok) {
   170→        setAppliedVoucher(null);
   171→        setVoucherError(data.error ?? "Kode voucher tidak valid.");
   172→        return;
   173→      }
   174→      const voucher = data.voucher as VoucherInfo;
   175→      setAppliedVoucher(voucher);
   176→      setVoucherInput(voucher.code);
   177→      showNotice({
   178→        type: "success",
   179→        mascot: true,
   180→        title: "Voucher Terpasang! 🎟️",
   181→        message: `${voucher.code} (${voucher.title}) berhasil ditukar — potongan ${formatIDR(
   182→          data.discount
   183→        )} otomatis diterapkan ke total.`,
   184→      });
   185→    } catch {
   186→      setVoucherError("Tidak dapat menghubungi server. Coba lagi ya.");
   187→    } finally {
   188→      setVoucherChecking(false);
   189→    }
   190→  };
   191→
   192→  // Kode voucher yang "dibawa" dari beranda → isi & tukar otomatis
   193→  useEffect(() => {
   194→    if (pendingCode && cart.items.length > 0) {
   195→      const code = pendingCode;
   196→      setPendingCode(null);
   197→      setVoucherInput(code);
   198→      void applyVoucher(code);
   199→    }
   200→  }, [pendingCode, cart.items.length]);
   201→
   202→  const removeVoucher = () => {
   203→    setAppliedVoucher(null);
   204→    setVoucherInput("");
   205→    setVoucherError(null);
   206→  };
   207→
   208→  /** Voucher yang masih bisa dipakai akun ini untuk subtotal sekarang */
   209→  const quickPicks = useMemo(
   210→    () =>
   211→      vouchers
   212→        .filter((v) => !v.usedByMe && v.remaining > 0 && subtotal >= v.minPurchase)
   213→        .filter((v) => v.code !== appliedVoucher?.code)
   214→        .slice(0, 3),
   215→    [vouchers, subtotal, appliedVoucher]
   216→  );
   217→
   218→  /** Validasi → konfirmasi final (pesanan tidak dapat dibatalkan) → kirim */
   219→  const startOrder = () => {
   220→    if (!form.name.trim()) {
   221→      showNotice({
   222→        type: "error",
   223→        title: "Nama Pemesan Kosong",
   224→        message: "Isi dulu nama pemesan ya, biar pesananmu mudah dikenali dapur 😊",
   225→      });
   226→      return;
   227→    }
   228→    if (form.type === "DINE_IN" && !form.tableNo.trim()) {
   229→      showNotice({
   230→        type: "error",
   231→        title: "Nomor Meja Kosong",
   232→        message: "Isi dulu nomor meja tempat kamu duduk ya 😊",
   233→      });
   234→      return;
   235→    }
   236→    if (form.type === "DELIVERY" && !form.areaId) {
   237→      showNotice({
   238→        type: "error",
   239→        title: "Alamat Pengantaran Kosong",
   240→        message: "Pilih dulu gampong tujuan di area layanan (Kec. Pidie, Kab. Pidie, Aceh 24151).",
   241→      });
   242→      return;
   243→    }
   244→
   245→    const paymentLabel = PAYMENTS.find((p) => p.id === form.payment)?.label ?? "Tunai";
   246→    const voucherText =
   247→      appliedVoucher && voucherDiscount > 0
   248→        ? `, voucher ${appliedVoucher.code} (potongan ${formatIDR(voucherDiscount)})`
   249→        : "";
   250→    const deliveryText =
   251→      form.type === "DELIVERY" && deliveryArea
   252→        ? `, antar ke Gampong ${deliveryArea.gampong} (ongkir ${formatIDR(deliveryFee)})`
   253→        : "";
   254→    showNotice({
   255→      type: "confirm",
   256→      title: "Konfirmasi Pesanan",
   257→      message: `Atas nama ${form.name.trim()} — ${cart.totalItems()} item, pembayaran ${paymentLabel}${voucherText}${deliveryText}, total bayar ${formatIDR(
   258→        total
   259→      )}. Pesanan langsung masuk dapur dan TIDAK DAPAT dibatalkan. Lanjutkan pesanan?`,
   260→      actions: [
   261→        { label: "Ya, Pesan Sekarang", onClick: () => void submitOrder() },
   262→        { label: "Periksa Lagi" },
   263→      ],
   264→    });
   265→  };
   266→
   267→  const submitOrder = async () => {
   268→    setSubmitting(true);
   269→    try {
   270→      const res = await apiFetch("/api/orders", {
   271→        method: "POST",
   272→        headers: { "Content-Type": "application/json" },
   273→        body: JSON.stringify({
   274→          customerName: form.name.trim(),
   275→          items: cart.items.map((i) => ({ productId: i.productId, qty: i.qty })),
   276→          type: form.type,
   277→          tableNo: form.type === "DINE_IN" ? form.tableNo : undefined,
   278→          address: form.type === "DELIVERY" ? form.address.trim() || undefined : undefined,
   279→          deliveryAreaId: form.type === "DELIVERY" ? form.areaId : undefined,
   280→          phone: form.phone,
   281→          note: form.note || undefined,
   282→          paymentMethod: form.payment,
   283→          source: "APP",
   284→          voucherCode: appliedVoucher?.code || undefined,
   285→        }),
   286→      });
   287→      const data = await res.json();
   288→      if (!res.ok) {
   289→        // 401 sudah ditangani global (kembali ke halaman login)
   290→        if (res.status !== 401) {
   291→          // Voucher bisa jadi tidak lolos validasi ulang server (mis. kuota habis duluan)
   292→          if (appliedVoucher && typeof data.error === "string" && data.error.toLowerCase().includes("voucher")) {
   293→            setAppliedVoucher(null);
   294→            setVoucherInput("");
   295→          }
   296→          showNotice({ type: "error", title: "Gagal Membuat Pesanan", message: data.error ?? "Coba lagi ya." });
   297→        }
   298→        return;
   299→      }
   300→      cart.clear();
   301→      setAppliedVoucher(null);
   302→      setVoucherInput("");
   303→      setVoucherError(null);
   304→      refreshOrders("mine");
   305→      refreshVouchers();
   306→      const created = data.order;
   307→      const finalTotal = created?.total ?? total;
   308→
   309→      // Pembayaran QRIS dan masih ada tagihan → tampilkan layar pindai QR
   310→      if (
   311→        created &&
   312→        created.paymentMethod === "QRIS" &&
   313→        created.paymentStatus !== "PAID" &&
   314→        finalTotal > 0
   315→      ) {
   316→        setPayOrder({ id: created.id, code: created.code, total: finalTotal });
   317→        return;
   318→      }
   319→
   320→      const hadVoucher = !!created?.voucherCode;
   321→      showNotice({
   322→        type: "success",
   323→        mascot: true,
   324→        title: "Pesanan Berhasil! 🎉",
   325→        message: `Pesanan ${created.code} sudah masuk dapur${
   326→          hadVoucher ? ` — voucher ${created.voucherCode} terpakai` : ""
   327→        }. Total bayar ${formatIDR(finalTotal)}. Ayam geprekmu segera dimasak!`,
   328→        actions: [
   329→          {
   330→            label: "Lihat Pesanan",
   331→            onClick: () => go("orders"),
   332→          },
   333→        ],
   334→      });
   335→    } catch {
   336→      showNotice({ type: "error", title: "Koneksi Bermasalah", message: "Tidak dapat menghubungi server." });
   337→    } finally {
   338→      setSubmitting(false);
   339→    }
   340→  };
   341→
   342→  /** Layar pembayaran QRIS — dirender di kedua keadaan (keranjang isi/kosong)
   343→      agar tetap tampil setelah pesanan sukses mengosongkan keranjang */
   344→  const qrisSheet = payOrder ? (
   345→    <QrisPaySheet
   346→      order={payOrder}
   347→      variant="customer"
   348→      onClose={({ paid }) => {
   349→        setPayOrder(null);
   350→        // Pembayaran QRIS belum berhasil → TIDAK ada popup "Pesanan Berhasil"
   351→        // (permintaan pemilik). Toast peringatan sudah tampil dari QrisPaySheet;
   352→        // pesanan tetap bisa dilunasi lewat tombol "Bayar QRIS" di daftar pesanan.
   353→        if (!paid) return;
   354→        showNotice({
   355→          type: "success",
   356→          mascot: true,
   357→          title: "Pesanan Berhasil! 🎉",
   358→          message: `Pesanan ${payOrder.code} sudah dibayar & masuk dapur. Total bayar ${formatIDR(
   359→            payOrder.total
   360→          )}. Ayam geprekmu segera dimasak!`,
   361→          actions: [
   362→            {
   363→              label: "Lihat Pesanan",
   364→              onClick: () => go("orders"),
   365→            },
   366→          ],
   367→        });
   368→      }}
   369→      onPaid={() => void refreshOrders("mine")}
   370→    />
   371→  ) : null;
   372→
   373→  if (cart.items.length === 0) {
   374→    return (
   375→      <>
   376→        {qrisSheet}
   377→        <div className="flex min-h-[70dvh] flex-col items-center justify-center gap-3 px-8 text-center">
   378→          <span className="flex h-20 w-20 items-center justify-center rounded-full bg-orange-50 text-4xl ring-4 ring-orange-100">
   379→            🛒
   380→          </span>
   381→          <h2 className="text-lg font-black text-neutral-800">Keranjang Masih Kosong</h2>
   382→          <p className="text-sm text-neutral-400">
   383→            Ayamnya kesepian nih… yuk pilih menu geprek favoritmu dulu!
   384→          </p>
   385→          <button
   386→            onClick={() => go("menu")}
   387→            className="press-effect mt-2 rounded-full bg-orange-600 px-6 py-2.5 text-sm font-extrabold text-white shadow-lg hover:bg-orange-700"
   388→          >
   389→            Lihat Menu
   390→          </button>
   391→        </div>
   392→      </>
   393→    );
   394→  }
   395→
   396→  return (
   397→    <div className="space-y-4 px-4 pt-[calc(0.75rem+env(safe-area-inset-top,0px))]">
   398→      <header>
   399→        <h1 className="text-xl font-black text-neutral-900">Keranjang 🛒</h1>
   400→        <p className="text-xs text-neutral-400">{cart.totalItems()} item siap digeprek</p>
   401→      </header>
   402→
   403→      {/* Daftar item */}
   404→      <section className="space-y-2.5">
   405→        {cart.items.map((item, i) => (
   406→          <motion.div
   407→            key={item.productId}
   408→            layout={!prefersReduced}
   409→            initial={prefersReduced ? false : { opacity: 0, x: -14 }}
   410→            animate={{ opacity: 1, x: 0 }}
   411→            transition={{ delay: i * 0.04 }}
   412→            className="flex items-center gap-3 rounded-2xl border border-orange-100 bg-white p-3 shadow-sm"
   413→          >
   414→            <ProductThumb
   415→              product={
   416→                products.find((p) => p.id === item.productId) ?? {
   417→                  id: item.productId,
   418→                  updatedAt: "",
   419→                  emoji: item.emoji,
   420→                  imageColor: item.imageColor,
   421→                  hasImage: false,
   422→                }
   423→              }
   424→              className="h-14 w-14 text-2xl"
   425→            />
   426→            <div className="min-w-0 flex-1">
   427→              <p className="line-clamp-1 text-sm font-bold text-neutral-800">{item.name}</p>
   428→              <p className="text-xs font-semibold text-orange-600">{formatIDR(item.price)}</p>
   429→            </div>
   430→            <div className="flex items-center gap-1 rounded-full bg-orange-50 p-0.5">
   431→              <button
   432→                aria-label="Kurangi"
   433→                className="press-effect flex h-7 w-7 items-center justify-center rounded-full bg-white text-orange-700 shadow-sm hover:bg-orange-100"
   434→                onClick={() => cart.setQty(item.productId, item.qty - 1)}
   435→              >
   436→                <Minus className="h-3.5 w-3.5" />
   437→              </button>
   438→              <span className="w-6 text-center text-sm font-black text-orange-800">{item.qty}</span>
   439→              <button
   440→                aria-label="Tambah"
   441→                className="press-effect flex h-7 w-7 items-center justify-center rounded-full bg-orange-600 text-white shadow-sm hover:bg-orange-700"
   442→                onClick={() => cart.setQty(item.productId, item.qty + 1)}
   443→              >
   444→                <Plus className="h-3.5 w-3.5" />
   445→              </button>
   446→            </div>
   447→            <button
   448→              aria-label={`Hapus ${item.name}`}
   449→              className="press-effect ml-1 flex h-8 w-8 items-center justify-center rounded-full text-neutral-300 transition-colors hover:bg-red-50 hover:text-red-500"
   450→              onClick={() => cart.remove(item.productId)}
   451→            >
   452→              <Trash2 className="h-4 w-4" />
   453→            </button>
   454→          </motion.div>
   455→        ))}
   456→      </section>
   457→
   458→      {/* Form checkout */}
   459→      <section className="space-y-4 rounded-2xl border border-orange-100 bg-white p-4 shadow-sm">
   460→        <h2 className="text-sm font-black uppercase tracking-wide text-neutral-700">
   461→          Detail Pesanan
   462→        </h2>
   463→
   464→        {/* Nama pemesan */}
   465→        <div className="space-y-1.5">
   466→          <Label htmlFor="custName">Nama Pemesan</Label>
   467→          <div className="relative">
   468→            <UserRound className="pointer-events-none absolute left-3 top-1/2 h-4 w-4 -translate-y-1/2 text-orange-400" />
   469→            <Input
   470→              id="custName"
   471→              placeholder="cth: Budi Santoso"
   472→              className="border-orange-200 pl-9 focus-visible:ring-orange-400"
   473→              value={form.name}
   474→              onChange={(e) => setForm({ ...form, name: e.target.value })}
   475→              maxLength={60}
   476→            />
   477→          </div>
   478→          <p className="text-[11px] text-neutral-400">
   479→            Nama ini akan tertera pada struk pesananmu.
   480→          </p>
   481→        </div>
   482→
   483→        {/* Tipe pesanan — "Antar" hilang bila layanan dimatikan admin */}
   484→        <div className={cn("grid gap-2", availableTypes.length === 3 ? "grid-cols-3" : "grid-cols-2")}>
   485→          {availableTypes.map((t) => (
   486→            <button
   487→              key={t.id}
   488→              type="button"
   489→              onClick={() => setForm({ ...form, type: t.id })}
   490→              className={cn(
   491→                "press-effect flex flex-col items-center gap-1 rounded-xl border p-2.5 text-[10px] font-bold transition-all",
   492→                form.type === t.id
   493→                  ? "border-orange-500 bg-orange-50 text-orange-700 ring-2 ring-orange-200"
   494→                  : "border-neutral-200 bg-white text-neutral-500 hover:border-orange-200"
   495→              )}
   496→            >
   497→              <t.icon className="h-5 w-5" strokeWidth={2.2} />
   498→              {t.label}
   499→            </button>
   500→          ))}
   501→        </div>
   502→        {!deliveryEnabled && (
