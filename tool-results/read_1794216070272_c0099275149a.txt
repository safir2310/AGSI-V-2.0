     1→     1→/**
     2→     2→ * Penyedia QRIS via DOKU (https://api-sandbox.doku.com / https://api.doku.com).
     3→     3→ *
     4→     4→ * DOKU punya DUA jalur integrasi — adapter ini memakai yang kredensialnya
     5→     5→ * tersedia (jalur SNAP REST lebih diutamakan bila lengkap):
     6→     6→ *
     7→     7→ *  1) JALUR SNAP REST — QRIS MPM Direct API (lihat developers.doku.com):
     8→     8→ *     a. Token B2B      : POST /authorization/v1/access-token/b2b
     9→     9→ *        X-SIGNATURE    : HMAC_SHA512(clientSecret, clientId+"|"+timestamp)
    10→    10→ *                         atau SHA256withRSA(privateKey, sama string) bila
    11→    11→ *                         DOKU_PRIVATE_KEY diset.
    12→    12→ *     b. Generate QRIS  : POST /snap-adapter/b2b/v1.0/qr/qr-mpm-generate
    13→    13→ *        X-SIGNATURE    : HMAC_SHA512(clientSecret, "POST:<path>:<accessToken>:
    14→    14→ *                         <hex(sha256(minifiedBody))>:<timestamp>")
    15→    15→ *        -> response.qrContent = string QRIS EMVCo siap dirender.
    16→    16→ *     c. Query status   : POST /snap-adapter/b2b/v1.0/qr/qr-mpm-query
    17→    17→ *        -> latestTransactionStatus: "00" sukses, "01"/"02"/"03" menunggu,
    18→    18→ *           "05"/"06"/"07"/"08" gagal/kedaluwarsa.
    19→    19→ *     d. Expire QR      : POST /snap-adapter/b2b/v1.0/qr/qr-expire
    20→    20→ *
    21→    21→ *  2) JALUR API KEY AGENTIC (MCP DOKU — JSON-RPC POST /doku-mcp-server/mcp,
    22→    22→ *     Authorization: Basic base64(apiKey + ":")):
    23→    23→ *     a. tools/call create_qris_payment -> QRIS dinamis (amount, invoice).
    24→    24→ *     b. tools/call get_transaction_by_invoice_number -> cari status transaksi.
    25→    25→ *     Kedua tool dipanggil langsung dari backend (server-to-server).
    26→    26→ *
    27→    27→ * Alur pelunasan (kebijakan toko — hanya SISTEM yang boleh melunaskan):
    28→    28→ *   polling /api/orders/[id]/qris/status memanggil getQrisDokuStatus()
    29→    29→ *   (query ke DOKU) -> PAID hanya bila DOKU sendiri mengatakan dana masuk;
    30→    30→ *   webhook DOKU (/api/webhooks/qris) juga SELALU konfirmasi ulang lewat
    31→    31→ *   query sebelum memanggil markOrderPaidBySystem() — fail-closed.

    68→    68→/**
    69→    69→ * Waktu WIB berformat SNAP "YYYY-MM-DDTHH:mm:ss+07:00" — format yang

    72→    72→function wibTimestamp(d: Date = new Date()): string {

    90→    90→/** Pemetaan kode status SNAP DOKU -> status baku aplikasi */
    91→    91→function snapStatusToGateway(raw: unknown): GatewayStatus {
    92→    92→  const s = String(raw ?? "").trim();
    93→    93→  switch (s) {
    94→    94→    case "00": // Success
    95→    95→      return "PAID";
    96→    96→    case "01": // Initiated
    97→    97→    case "02": // Paying
    98→    98→    case "03": // Pending
    99→    99→    case "09": // Denied... masih bisa berubah — perlakukan menunggu
   100→   100→      return "PENDING";
   101→   101→    case "04": // Refunded
   102→   102→    case "05": // Canceled
   103→   103→    case "06": // Failed
   104→   104→    case "07": // Void
   105→   105→    case "08": // Expired
   106→   106→      return "EXPIRED";
   107→   107→    default:
   108→   108→      return "UNKNOWN";
   109→   109→  }
   110→   110→}

   112→   112→/** Cari string QRIS EMVCo di dalam objek respons apa pun (toleran struktur) */
   113→   113→function findQrString(value: unknown, depth = 0): string | undefined {
   114→   114→  if (depth > 6 || value == null) return undefined;
   115→   115→  if (typeof value === "string") {
   116→   116→    const v = value.trim();
   117→   117→    if (v.startsWith("000201") || v.includes("ID.CO.QRIS.WWW")) return v;
   118→   118→    return undefined;
   119→   119→  }
   120→   120→  if (Array.isArray(value)) {
   121→   121→    for (const item of value) {
   122→   122→      const hit = findQrString(item, depth + 1);
   123→   123→      if (hit) return hit;
   124→   124→    }
   125→   125→    return undefined;
   126→   126→  }
   127→   127→  if (typeof value === "object") {
   128→   128→    for (const v of Object.values(value as Record<string, unknown>)) {
   129→   129→      const hit = findQrString(v, depth + 1);
   130→   130→      if (hit) return hit;
   131→   131→    }
   132→   132→  }
   133→   133→  return undefined;
   134→   134→}

   136→   136→/** Urai teks hasil tool MCP — berupa JSON string (kadang dibungkus) */
   137→   137→function parseMcpText(text: string): Record<string, unknown> | null {
   138→   138→  const trimmed = text.trim();
   139→   139→  // Kadang respons diawali "..." langsung JSON; coba parse utuh dulu
   140→   140→  try {
   141→   141→    const parsed = JSON.parse(trimmed) as unknown;
   142→   142→    if (parsed && typeof parsed === "object" && !Array.isArray(parsed)) {
   143→   143→      return parsed as Record<string, unknown>;
   144→   144→    }
   145→   145→  } catch {
   146→   146→    // bukan JSON utuh — coba ekstrak bagian {..} pertama
   147→   147→  }
   148→   148→  const start = trimmed.indexOf("{");
   149→   149→  const end = trimmed.lastIndexOf("}");
   150→   150→  if (start >= 0 && end > start) {
   151→   151→    try {
   152→   152→      const parsed = JSON.parse(trimmed.slice(start, end + 1)) as unknown;
   153→   153→      if (parsed && typeof parsed === "object" && !Array.isArray(parsed)) {
   154→   154→        return parsed as Record<string, unknown>;
   155→   155→      }
   156→   156→    } catch {
   157→   157→      return null;
   158→   158→    }
   159→   159→  }
   160→   160→  return null;
   161→   161→}

   163→   163→/** Ambil pesan galat terbaca dari objek galat DOKU */
   164→   164→function messageOf(obj: Record<string, unknown> | null, fallback: string): string {
   165→   165→  if (!obj) return fallback;
   166→   166→  const parts: string[] = [];
   167→   167→  for (const key of ["responseMessage", "message", "error", "errorMessage"]) {
   168→   168→    const v = obj[key];
   169→   169→    if (typeof v === "string" && v.trim()) parts.push(v.trim());
   170→   170→  }
   171→   171→  return parts.join(" — ") || fallback;
   172→   172→}

   175→   175→//  JALUR SNAP REST — token, generate, query, expire

   177→   177→
   178→   178→// Cache token B2B (berlaku 900 detik) — satu token dipakai ulang semua request
   179→   179→let cachedToken: { token: string; expiresAt: number } | null = null;
   180→   180→
   181→   181→/** HMAC_SHA512 base64 — signature simetris SNAP */
   182→   182→function hmac512(secret: string, stringToSign: string): string {
   183→   183→  return createHmac("sha512", secret).update(stringToSign).digest("base64");
   184→   184→}
   185→   185→
   186→   186→/** SHA256withRSA base64 — signature asimetris SNAP (bila private key diset) */
   187→   187→function rsaSign(privateKeyPem: string, stringToSign: string): string {
   188→   188→  const signer = createSign("RSA-SHA256");
   189→   189→  signer.update(stringToSign);
   190→   190→  return signer.sign(privateKeyPem, "base64");
   191→   191→}
   192→   192→
   193→   193→/** Normalisasi PEM satu baris (env Vercel) -> PEM utuh */
   194→   194→function normalizePem(raw: string): string {
   195→   195→  const v = raw.trim();
   196→   196→  if (!v) return "";
   197→   197→  const withBreaks = v.includes("\\n") ? v.replace(/\\n/g, "\n") : v;
   198→   198→  if (withBreaks.includes("-----BEGIN")) return withBreaks.trim();
   199→   199→  return `-----BEGIN PRIVATE KEY-----\n${withBreaks}\n-----END PRIVATE KEY-----`;
   200→   200→}
   201→   201→
   202→   202→/**
   203→   203→ * POST /authorization/v1/access-token/b2b — token B2B (cache 13 menit).
   204→   204→ * Signature: simetris HMAC (default) atau asimetris RSA (bila private key).
   205→   205→ */
   206→   206→async function getAccessToken(): Promise<string> {
   207→   207→  if (cachedToken && cachedToken.expiresAt > Date.now() + 30_000) {
   208→   208→    return cachedToken.token;
   209→   209→  }
   210→   210→  if (!DOKU_CLIENT_ID || !DOKU_CLIENT_SECRET) {
   211→   211→    throw new QrisGatewayError("Kredensial DOKU (Client ID/Secret) belum lengkap.");
   212→   212→  }
   213→   213→  const path = "/authorization/v1/access-token/b2b";
   214→   214→  const timestamp = wibTimestamp();
   215→   215→  const stringToSign = `${DOKU_CLIENT_ID}|${timestamp}`;
   216→   216→  const signature = DOKU_PRIVATE_KEY
   217→   217→    ? rsaSign(normalizePem(DOKU_PRIVATE_KEY), stringToSign)
   218→   218→    : hmac512(DOKU_CLIENT_SECRET, stringToSign);
   219→   219→
   220→   220→  const controller = new AbortController();
   221→   221→  const timer = setTimeout(() => controller.abort(), 12_000);
   222→   222→  let httpStatus = 0;
   223→   223→  let body: Record<string, unknown>;
   224→   224→  try {
   225→   225→    const res = await fetch(`${DOKU_API_BASE}${path}`, {
   226→   226→      method: "POST",
   227→   227→      headers: {
   228→   228→        "Content-Type": "application/json",
   229→   229→        "X-CLIENT-KEY": DOKU_CLIENT_ID,
   230→   230→        "X-TIMESTAMP": timestamp,
   231→   231→        "X-SIGNATURE": signature,
   232→   232→      },
   233→   233→      body: JSON.stringify({ grantType: "client_credentials" }),
   234→   234→      signal: controller.signal,
   235→   235→      cache: "no-store",
   236→   236→    });
   237→   237→    httpStatus = res.status;
   238→   238→    body = (await res.json().catch(() => ({}))) as Record<string, unknown>;
   239→   239→  } catch (err) {
   240→   240→    if (err instanceof Error && err.name === "AbortError") {
   241→   241→      throw new QrisGatewayError("DOKU tidak merespons saat ambil token (timeout).");
   242→   242→    }
   243→   243→    throw new QrisGatewayError("Tidak dapat menghubungi DOKU (token).");
   244→   244→  } finally {
   245→   245→    clearTimeout(timer);
   246→   246→  }
   247→   247→
   248→   248→  const token = typeof body.accessToken === "string" ? body.accessToken : "";
   249→   249→  if (httpStatus >= 300 || !token) {
   250→   250→    throw new QrisGatewayError(
   251→   251→      `DOKU menolak permintaan token (HTTP ${httpStatus}). ${messageOf(body, "Periksa DOKU_CLIENT_ID / DOKU_CLIENT_SECRET.")}`,
   252→   252→      httpStatus === 401 ? 401 : undefined
   253→   253→    );
   254→   254→  }
   255→   255→  const expiresIn = Number(body.expiresIn ?? 900) || 900;
   256→   256→  cachedToken = { token, expiresAt: Date.now() + expiresIn * 1000 };
   257→   257→  return token;
   258→   258→}

   260→   260→/** POST ke endpoint SNAP dengan header signature lengkap */
   261→   261→async function snapPost(
   262→   262→  path: string,
   263→   263→  payload: Record<string, unknown>,
   264→   264→  timeoutMs = 15_000
   265→   265→): Promise<Record<string, unknown>> {
   266→   266→  const accessToken = await getAccessToken();
   267→   267→  const timestamp = wibTimestamp();
   268→   268→  const minified = JSON.stringify(payload);
   269→   269→  const bodyHash = createHash("sha256").update(minified).digest("hex").toLowerCase();
   270→   270→  const stringToSign = `POST:${path}:${accessToken}:${bodyHash}:${timestamp}`;
   271→   271→  const signature = hmac512(DOKU_CLIENT_SECRET, stringToSign);
   272→   272→
   273→   273→  const controller = new AbortController();
   274→   274→  const timer = setTimeout(() => controller.abort(), timeoutMs);
   275→   275→  let httpStatus = 0;
   276→   276→  let body: Record<string, unknown>;
   277→   277→  try {
   278→   278→    const res = await fetch(`${DOKU_API_BASE}${path}`, {
   279→   279→      method: "POST",
   280→   280→      headers: {
   281→   281→        "Content-Type": "application/json",
   282→   282→        Accept: "application/json",
   283→   283→        "X-PARTNER-ID": DOKU_CLIENT_ID,
   284→   284→        "X-EXTERNAL-ID": externalId(),
   285→   285→        "X-TIMESTAMP": timestamp,
   286→   286→        "X-SIGNATURE": signature,
   287→   287→        Authorization: `Bearer ${accessToken}`,
   288→   288→        "CHANNEL-ID": "H2H",
   289→   289→      },
   290→   290→      body: minified,
   291→   291→      signal: controller.signal,
   292→   292→      cache: "no-store",
   293→   293→    });
   294→   294→    httpStatus = res.status;
   295→   295→    body = (await res.json().catch(() => ({}))) as Record<string, unknown>;
   296→   296→  } catch (err) {
   297→   297→    if (err instanceof Error && err.name === "AbortError") {
   298→   298→      throw new QrisGatewayError("DOKU tidak merespons (timeout).");
   299→   299→    }
   300→   300→    throw new QrisGatewayError("Tidak dapat menghubungi DOKU. Periksa koneksi internet server.");
   301→   301→  } finally {
   302→   302→    clearTimeout(timer);
   303→   303→  }
   304→   304→
   305→   305→  if (httpStatus >= 400) {
   306→   306→    throw new QrisGatewayError(
   307→   307→      `DOKU menolak ${path} (HTTP ${httpStatus}). ${messageOf(body, "")}`,
   308→   308→      httpStatus === 404 ? 404 : httpStatus === 401 ? 401 : undefined
   309→   309→    );
   310→   310→  }
   311→   311→  return body;
   312→   312→}

   371→   371→
   372→   372→/** Panggil satu tool MCP DOKU; kembalikan objek hasil terurai */
   373→   373→async function mcpCallTool(
   374→   374→  toolName: string,
   375→   375→  toolRequest: Record<string, unknown>
   376→   376→): Promise<Record<string, unknown> | null> {
   377→   377→  if (!DOKU_APIKEY_READY) {
   378→   378→    throw new QrisGatewayError("API key DOKU belum dikonfigurasi di .env (DOKU_API_KEY).");
   379→   379→  }
   380→   380→  const basic = Buffer.from(`${DOKU_API_KEY}:`).toString("base64");
   381→   381→  const id = ++mcpRequestId;
   382→   382→
   383→   383→  const controller = new AbortController();
   384→   384→  const timer = setTimeout(() => controller.abort(), 30_000);
   385→   385→  let httpStatus = 0;
   386→   386→  let raw = "";
   387→   387→  try {
   388→   388→    const res = await fetch(`${DOKU_API_BASE}${DOKU_MCP_PATH}`, {
   389→   389→      method: "POST",
   390→   390→      headers: {
   391→   391→        Authorization: `Basic ${basic}`,
   392→   392→        "Content-Type": "application/json",
   393→   393→        Accept: "application/json, text/event-stream",
   394→   394→      },
   395→   395→      body: JSON.stringify({
   396→   396→        jsonrpc: "2.0",
   397→   397→        id,
   398→   398→        method: "tools/call",
   399→   399→        params: { name: toolName, arguments: { toolRequest } },
   400→   400→      }),
   401→   401→      signal: controller.signal,
   402→   402→      cache: "no-store",
   403→   403→    });
   404→   404→    httpStatus = res.status;
   405→   405→    raw = await res.text();
   406→   406→  } catch (err) {
   407→   407→    if (err instanceof Error && err.name === "AbortError") {
   408→   408→      throw new QrisGatewayError("MCP DOKU tidak merespons (timeout).");
   409→   409→    }
   410→   410→    throw new QrisGatewayError("Tidak dapat menghubungi MCP DOKU. Periksa koneksi internet server.");
   411→   411→  } finally {
   412→   412→    clearTimeout(timer);
   413→   413→  }
   414→   414→
   415→   415→  if (httpStatus >= 400) {
   416→   416→    throw new QrisGatewayError(
   417→   417→      `MCP DOKU menolak panggilan (HTTP ${httpStatus}).`,
   418→   418→      httpStatus === 401 ? 401 : undefined
   419→   419→    );
   420→   420→  }
   421→   421→
   422→   422→  // Respons MCP bisa JSON murni ATAU SSE ("data: {...}") — urai toleran
   423→   423→  let envelope: Record<string, unknown> | null = null;
   424→   424→  const direct = parseMcpText(raw);
   425→   425→  if (direct && typeof direct.result === "object") {
   426→   426→    envelope = direct;
   427→   427→  } else {
   428→   428→    for (const line of raw.split("\n")) {
   429→   429→      const trimmedLine = line.trim();

   519→   519→/** Status via MCP get_transaction_by_invoice_number */
   520→   520→async function mcpQueryStatus(partnerReferenceNo: string): Promise<GatewayStatus> {
   521→   521→  const now = new Date();
   522→   522→  const ddmm = (d: Date) =>
   523→   523→    `${String(d.getDate()).padStart(2, "0")}-${String(d.getMonth() + 1).padStart(2, "0")}-${d.getFullYear()}`;
   524→   524→  const start = new Date(now.getTime() - 3 * 24 * 60 * 60 * 1000);
   525→   525→  const end = new Date(now.getTime() + 24 * 60 * 60 * 1000);
   526→   526→
   527→   527→  let parsed: Record<string, unknown> | null;
   528→   528→  try {
   529→   529→    parsed = await mcpCallTool("get_transaction_by_invoice_number", {
   530→   530→      invoiceNumber: partnerReferenceNo,
   531→   531→      startDate: ddmm(start),
   532→   532→      endDate: ddmm(end),
   533→   533→      pageNumber: 1,
   534→   534→      pageSize: 10,
   535→   535→    });
   536→   536→  } catch (err) {
   537→   537→    const msg = err instanceof Error ? err.message.toLowerCase() : "";
   538→   538→    // DOKU: laporan transaksi kosong / belum ada catatan = transaksi belum
   539→   539→    // ada (belum dibayar) — bukan galat; QR tetap menunggu pembayaran.
   540→   540→    if (msg.includes("searchfailed") || msg.includes("unable to complete search")) {
   541→   541→      return "PENDING";
   542→   542→    }
   543→   543→    throw err;
   544→   544→  }
   545→   545→  if (!parsed) return "PENDING";
   546→   546→
   547→   547→  // DOKU: transaksi belum tercatat (belum dibayar) -> tetap menunggu.
   548→   548→  const entry = findTransactionEntry(parsed, partnerReferenceNo);
   549→   549→  if (!entry) return "PENDING";
   550→   550→
   551→   551→  // Kode SNAP (00/03/...) diutamakan; kalau tidak ada, pakai kata status.
   552→   552→  const codeKeys = [
   553→   553→    "latestTransactionStatus",
   554→   554→    "transactionStatus",
   555→   555→    "trxStatus",
   556→   556→    "status_code",
   557→   557→    "statusCode",
   558→   558→  ];
   559→   559→  for (const k of codeKeys) {
   560→   560→    if (entry[k] != null) {
   561→   561→      const mapped = snapStatusToGateway(entry[k]);
   562→   562→      if (mapped !== "UNKNOWN") return mapped;
   563→   563→    }
   564→   564→  }
   565→   565→  const wordKeys = ["transactionStatusDesc", "status", "transaction_status", "paymentStatus"];
   566→   566→  for (const k of wordKeys) {
   567→   567→    const v = entry[k];
   568→   568→    if (typeof v === "string" && v.trim()) {
   569→   569→      const s = v.trim().toLowerCase();
   570→   570→      if (s === "success" || s === "success payment" || s === "paid") return "PAID";
   571→   571→      if (s.includes("success")) return "PAID";
   572→   572→      if (s.includes("pending") || s.includes("waiting") || s.includes("unpaid")) return "PENDING";
   573→   573→      if (s.includes("expired") || s.includes("cancel") || s.includes("failed") || s.includes("refund"))
   574→   574→        return "EXPIRED";
   575→   575→    }
   576→   576→  }
   577→   577→  return "PENDING";
   578→   578→}

   581→   581→//  JALUR CHECKOUT LEGACY (Non-SNAP) — produksi, TERBUKTI HIDUP
   582→   582→// ============================================================
   583→   583→// Bila kredensial SNAP belum terdaftar di gateway SNAP (4017300
   584→   584→// "Unauthorized. Unknown Client") tetapi Client ID + Secret Key VALID di
   585→   585→// gateway legacy Non-SNAP (divalidasi lewat Check Status API), transaksi
   586→   586→// dibuat via DOKU CHECKOUT:
   587→   587→//   POST /checkout/v1/payment        -> response.payment.url (halaman bayar
   588→   588→//     DOKU berisi kanal aktif merchant — QRIS tampil di sana begitu kanal
   589→   589→//     QRIS akun diaktifkan DOKU; sementara VA/retail/e-money aktif)
   590→   590→//   GET  /orders/v1/status/{invoice} -> transaction.status
   591→   591→// Signature legacy: HMAC-SHA256 atas "Client-Id\nRequest-Id\nRequest-
   592→   592→// Timestamp\nRequest-Target\nDigest" (HMACSHA256= prefix).

   594→   594→/** Header autentikasi legacy Non-SNAP — nilai TERKIRIM persis (HMACSHA256=) */
   595→   595→function legacyHeaders(
   596→   596→  target: string,
   597→   597→  bodyRaw: string | null
   598→   598→): Record<string, string> {
   599→   599→  const requestId = externalId();
   600→   600→  const ts = new Date().toISOString().replace(/\.\d{3}Z$/, "Z");
   601→   601→  const parts = [
   602→   602→    `Client-Id:${DOKU_CLIENT_ID}`,
   603→   603→    `Request-Id:${requestId}`,
   604→   604→    `Request-Timestamp:${ts}`,
   605→   605→    `Request-Target:${target}`,
   606→   606→  ];
   607→   607→  if (bodyRaw !== null) {
   608→   608→    parts.push(
   609→   609→      `Digest:${createHash("sha256").update(bodyRaw).digest("base64")}`
   610→   610→    );
   611→   611→  }
   612→   612→  return {
   613→   613→    "Client-Id": DOKU_CLIENT_ID,
   614→   614→    "Request-Id": requestId,
   615→   615→    "Request-Timestamp": ts,
   616→   616→    "Request-Target": target,
   617→   617→    Signature: `HMACSHA256=${createHmac("sha256", DOKU_CLIENT_SECRET)
   618→   618→      .update(parts.join("\n"))
   619→   619→      .digest("base64")}`,
   620→   620→    ...(bodyRaw !== null ? { "Content-Type": "application/json" } : {}),
   621→   621→  };
   622→   622→}

   624→   624→/** Pesan galat terbaca dari respons legacy (message bisa string/array) */
   625→   625→function legacyErrorMessage(
   626→   626→  body: Record<string, unknown>,
   627→   627→  fallback: string
   628→   628→): string {
   629→   629→  const err = body.error;
   630→   630→  if (err && typeof err === "object" && !Array.isArray(err)) {
   631→   631→    const m = (err as Record<string, unknown>).message;
   632→   632→    if (typeof m === "string" && m.trim()) return m;
   633→   633→  }
   634→   634→  const msg = body.message;
   635→   635→  if (typeof msg === "string" && msg.trim()) return msg;
   636→   636→  if (Array.isArray(msg)) {
   637→   637→    const joined = msg.filter((v) => typeof v === "string").join(", ");
   638→   638→    if (joined) return joined;
   639→   639→  }
   640→   640→  if (typeof body.responseMessage === "string" && body.responseMessage.trim()) {
   641→   641→    return body.responseMessage;
   642→   642→  }
   643→   643→  return fallback;
   644→   644→}

   664→   664→    });

   667→   667→  } catch (err) {

   676→   676→  }
   677→   677→  if (httpStatus >= 300) {
   678→   678→    throw new QrisGatewayError(
   679→   679→      `DOKU Checkout menolak ${path} (HTTP ${httpStatus}). ${legacyErrorMessage(body, "")}`.trim(),
   680→   680→      httpStatus === 404 ? 404 : httpStatus === 401 ? 401 : undefined
   681→   681→    );
   682→   682→  }
   683→   683→  return body;
   684→   684→}

   869→   869→  const raw =
   870→   870→    (transaction && typeof transaction.status === "string"
   871→   871→      ? transaction.status
   872→   872→      : null) ??
   873→   873→    (order && typeof order.status === "string"
   874→   874→      ? order.status
   875→   875→      : null) ??
   876→   876→    "";
   877→   877→  const s = raw.toUpperCase();
   878→   878→  if (!s) return "UNKNOWN";
   879→   879→  if (
   880→   880→    s === "SUCCESS" ||
   881→   881→    s === "ORDER_PAID" ||
   882→   882→    s === "PAID" ||
   883→   883→    s === "SETTLEMENT" ||
   884→   884→    s === "COMPLETED"
   885→   885→  ) {
   886→   886→    return "PAID";
   887→   887→  }
   888→   888→  if (
   889→   889→    s === "PENDING" ||
   890→   890→    s === "ORDER_GENERATED" ||
   891→   891→    s === "WAITING" ||
   892→   892→    s === "INITIATED"
   893→   893→  ) {
   894→   894→    return "PENDING";
   895→   895→  }
   896→   896→  if (
   897→   897→    s === "EXPIRED" ||
   898→   898→    s === "ORDER_EXPIRED" ||
   899→   899→    s === "ORDER_CANCELLED" ||
   900→   900→    s === "CANCELLED" ||
   901→   901→    s === "CANCELED" ||
   902→   902→    s === "FAILED" ||
   903→   903→    s === "TIMEOUT" ||
   904→   904→    s === "VOID"
   905→   905→  ) {
   906→   906→    return "EXPIRED";
   907→   907→  }
   908→   908→  return normalizeGatewayStatus(raw);
   909→   909→}

   922→   922→ *   2) CHECKOUT legacy (Client ID + Secret) — halaman bayar DOKU

   925→   925→export async function createQrisDokuPayment(opts: {
   926→   926→  amount: number;
   927→   927→  reference: string;
   928→   928→}): Promise<GatewayTransaction> {
   929→   929→  const amount = Math.round(Number(opts.amount));
   930→   930→  if (!Number.isFinite(amount) || amount <= 0) {
   931→   931→    throw new QrisGatewayError("Nominal QRIS harus angka lebih dari 0.");
   932→   932→  }
   933→   933→  const partnerReferenceNo = partnerReferenceNoOf(opts.reference);
   934→   934→
   935→   935→  if (DOKU_SNAP_READY) {
   936→   936→    const validityPeriod = wibTimestamp(
   937→   937→      new Date(Date.now() + DOKU_QRIS_EXPIRY_MINUTES * 60_000)
   938→   938→    );
   939→   939→    const qrString = await snapGenerateQris({
   940→   940→      partnerReferenceNo,
   941→   941→      amount,
   942→   942→      validityPeriod,
   943→   943→    });
   944→   944→    return {
   945→   945→      trxId: partnerReferenceNo,
   946→   946→      qrString,
   947→   947→      expiresAt: new Date(Date.now() + DOKU_QRIS_EXPIRY_MINUTES * 60_000).toISOString(),
   948→   948→    };
   949→   949→  }
   950→   950→
   951→   951→  if (DOKU_CHECKOUT_READY) {
   952→   952→    return checkoutCreateQris({ partnerReferenceNo, amount });
   953→   953→  }
   954→   954→
   955→   955→  const validityPeriod = wibTimestamp(
   956→   956→    new Date(Date.now() + DOKU_QRIS_EXPIRY_MINUTES * 60_000)
   957→   957→  );
   958→   958→  const qrString = await mcpCreateQris({ partnerReferenceNo, amount, validityPeriod });
   959→   959→  return {
   960→   960→    trxId: partnerReferenceNo,
   961→   961→    qrString,
   962→   962→    expiresAt: new Date(Date.now() + DOKU_QRIS_EXPIRY_MINUTES * 60_000).toISOString(),
   963→   963→  };
   964→   964→}

   983→   983→export type DokuCancelResult = {
   984→   984→  cancelled: boolean;
   985→   985→  alreadyPaid: boolean;
   986→   986→  providerStatus: string | null;
   987→   987→};

   989→   989→/**
   990→   990→ * Batalkan QRIS di DOKU (qr-expire). Jalur MCP tidak punya tool cancel —
   991→   991→ * dikembalikan no-op aman: QR otomatis mati saat validityPeriod terlewat.
   992→   992→ */
   993→   993→export async function cancelQrisDokuPayment(trxId: string): Promise<DokuCancelResult> {
   994→   994→  const partnerReferenceNo = partnerReferenceNoOf(trxId);
   995→   995→  if (!DOKU_SNAP_READY) {
   996→   996→    // Jalur CHECKOUT/MCP tidak punya cancel yang dipakai — QR/halaman bayar
   997→   997→    // otomatis mati saat payment_due_date terlewat (sama dengan validityPeriod).
   998→   998→    return { cancelled: false, alreadyPaid: false, providerStatus: "no_cancel_tool" };
   999→   999→  }
  1000→  1000→  try {
  1001→  1001→    // Cek dulu: sudah dibayar? Jangan mematikan QR yang barusan dibayar.
  1002→  1002→    const status = await snapQueryStatus(partnerReferenceNo);
  1003→  1003→    if (status === "PAID") {
  1004→  1004→      return { cancelled: false, alreadyPaid: true, providerStatus: "00" };
  1005→  1005→    }
  1006→  1006→    if (status === "EXPIRED") {
  1007→  1007→      return { cancelled: true, alreadyPaid: false, providerStatus: "expired" };
  1008→  1008→    }
  1009→  1009→    await snapExpireQris(partnerReferenceNo);
  1010→  1010→    return { cancelled: true, alreadyPaid: false, providerStatus: "expired_by_merchant" };
  1011→  1011→  } catch (err) {
  1012→  1012→    if (err instanceof QrisGatewayError && err.status === 404) {
  1013→  1013→      return { cancelled: true, alreadyPaid: false, providerStatus: "not_found" };
  1014→  1014→    }
  1015→  1015→    throw err;
  1016→  1016→  }
  1017→  1017→}

  1019→  1019→/**
  1020→  1020→ * Verifikasi signature HTTP NOTIFICATION DOKU (HMAC-SHA512 simetris).
  1021→  1021→ * Rumus SNAP: HMAC_SHA512(clientSecret, "POST:<path>:<accessToken>:
  1022→  1022→ * <hex(sha256(body))>:<timestamp>"). Notifikasi tidak membawa access token —
  1023→  1023→ * dicoba juga varian tanpa komponen token (fail-closed: hanya cocok yang benar).
  1024→  1024→ */
  1025→  1025→export function verifyDokuWebhookSignature(opts: {
  1026→  1026→  signature: string;
  1027→  1027→  timestamp: string;
  1028→  1028→  rawBody: string;
  1029→  1029→  paths: string[];
  1030→  1030→}): boolean {
  1031→  1031→  if (!DOKU_CLIENT_SECRET) return false;
  1032→  1032→  const got = opts.signature.trim();
  1033→  1033→  if (!got) return false;
  1034→  1034→  const bodyHash = createHash("sha256")
  1035→  1035→    .update(opts.rawBody)
  1036→  1036→    .digest("hex")
  1037→  1037→    .toLowerCase();
  1038→  1038→  const candidates: string[] = [];
  1039→  1039→  for (const p of opts.paths) {
  1040→  1040→    candidates.push(`POST:${p}::${bodyHash}:${opts.timestamp}`);
  1041→  1041→    candidates.push(`POST:${p}:::${opts.timestamp}`);
  1042→  1042→  }
  1043→  1043→  for (const s of candidates) {
  1044→  1044→    const expected = hmac512(DOKU_CLIENT_SECRET, s);
  1045→  1045→    if (expected.length === got.length && expected === got) return true;
  1046→  1046→  }
  1047→  1047→  return false;
  1048→  1048→}
  1049→  1049→
  1050→  1050→/**
  1051→  1051→ * Verifikasi signature NOTIFIKASI legacy (Non-SNAP / DOKU Checkout).
  1052→  1052→ * Header: Client-Id, Request-Id, Request-Timestamp, Request-Target, Signature.
  1053→  1053→ * Rumus: "HMACSHA256=" + Base64(HMAC-SHA256(secret,
  1054→  1054→ *   "Client-Id:..\nRequest-Id:..\nRequest-Timestamp:..\nRequest-Target:..\nDigest:<base64(sha256(body))>"))
  1055→  1055→ * Fail-closed: hanya signature yang benar diterima.
  1056→  1056→ */
  1057→  1057→export function verifyDokuLegacyWebhookSignature(opts: {
  1058→  1058→  signature: string;
  1059→  1059→  clientId?: string;
  1060→  1060→  requestId: string;
  1061→  1061→  timestamp: string;
  1062→  1062→  target: string;
  1063→  1063→  rawBody: string;
  1064→  1064→}): boolean {
  1065→  1065→  if (!DOKU_CLIENT_SECRET) return false;
  1066→  1066→  const got = opts.signature.trim();
  1067→  1067→  if (!got) return false;
  1068→  1068→  const digest = createHash("sha256").update(opts.rawBody).digest("base64");
  1069→  1069→  const parts = [
  1070→  1070→    `Client-Id:${opts.clientId?.trim() || DOKU_CLIENT_ID}`,
  1071→  1071→    `Request-Id:${opts.requestId}`,
  1072→  1072→    `Request-Timestamp:${opts.timestamp}`,
  1073→  1073→    `Request-Target:${opts.target.startsWith("/") ? opts.target : `/${opts.target}`}`,
  1074→  1074→    `Digest:${digest}`,
  1075→  1075→  ];
  1076→  1076→  const expected = `HMACSHA256=${createHmac("sha256", DOKU_CLIENT_SECRET)
  1077→  1077→    .update(parts.join("\n"))
  1078→  1078→    .digest("base64")}`;
  1079→  1079→  return expected.length === got.length && expected === got;
  1080→  1080→}
