    63→    63→export function CartView({ user, go }: { user: AppUser; go: (v: AppView) => void }) {
    64→    64→  const cart = useCartStore();
    65→    65→  const showNotice = useNoticeStore((s) => s.show);
    66→    66→  const prefersReduced = useReducedMotion();

    68→    68→  const vouchers = useVouchersStore((s) => s.vouchers);
    69→    69→  const loadVouchers = useVouchersStore((s) => s.load);
    70→    70→  const pendingCode = useVouchersStore((s) => s.pendingCode);
    71→    71→  const setPendingCode = useVouchersStore((s) => s.setPendingCode);
    72→    72→  const [submitting, setSubmitting] = useState(false);
    73→    73→  /** Pesanan QRIS yang sedang dibayar (layar QR terbuka) */
    74→    74→  const [payOrder, setPayOrder] = useState<QrisPayOrder | null>(null);
    75→    75→  /** Popup pilih alamat pengantaran (area Kec. Pidie 24151) */

    79→    79→    type: "DINE_IN" as (typeof ORDER_TYPES)[number]["id"],

    81→    81→    address: "",
    82→    82→    areaId: "",

    84→    84→    note: "",

    87→    87→

   123→   123→      setVoucherInput("");

   130→   130→      setAppliedVoucher(null);
   131→   131→      setVoucherError(null);
   132→   132→    }

   161→   161→        mascot: true,

   164→   164→          data.discount

   166→   166→      });
   167→   167→    } catch {
   168→   168→      setVoucherError("Tidak dapat menghubungi server. Coba lagi ya.");
   169→   169→    } finally {
   170→   170→      setVoucherChecking(false);
   171→   171→    }
   172→   172→  };
   173→   173→
   174→   174→  // Kode voucher yang "dibawa" dari beranda → isi & tukar otomatis

   181→   181→    }
   182→   182→  }, [pendingCode, cart.items.length]);
   183→   183→
   184→   184→  const removeVoucher = () => {
   185→   185→    setAppliedVoucher(null);
   186→   186→    setVoucherInput("");
   187→   187→    setVoucherError(null);
   188→   188→  };
   189→   189→
   190→   190→  /** Voucher yang masih bisa dipakai akun ini untuk subtotal sekarang */

   192→   192→    () =>
   193→   193→      vouchers
   194→   194→        .filter((v) => !v.usedByMe && v.remaining > 0 && subtotal >= v.minPurchase)

   196→   196→        .slice(0, 3),
   197→   197→    [vouchers, subtotal, appliedVoucher]
   198→   198→  );
   199→   199→
   200→   200→  /** Validasi → konfirmasi final (pesanan tidak dapat dibatalkan) → kirim */

   203→   203→      showNotice({
   204→   204→        type: "error",
   205→   205→        title: "Nama Pemesan Kosong",

   207→   207→      });

   209→   209→    }
   210→   210→    if (form.type === "DINE_IN" && !form.tableNo.trim()) {
   211→   211→      showNotice({
   212→   212→        type: "error",
   213→   213→        title: "Nomor Meja Kosong",
   214→   214→        message: "Isi dulu nomor meja tempat kamu duduk ya 😊",
   215→   215→      });
   216→   216→      return;
   217→   217→    }
   218→   218→    if (form.type === "DELIVERY" && !form.areaId) {
   219→   219→      showNotice({

   221→   221→        title: "Alamat Pengantaran Kosong",

   228→   228→    const voucherText =
   229→   229→      appliedVoucher && voucherDiscount > 0

   231→   231→        : "";

   233→   233→      form.type === "DELIVERY" && deliveryArea

   302→   302→      const hadVoucher = !!created?.voucherCode;
   303→   303→      showNotice({
   304→   304→        type: "success",
   305→   305→        mascot: true,
   306→   306→        title: "Pesanan Berhasil! 🎉",
   307→   307→        message: `Pesanan ${created.code} sudah masuk dapur${
   308→   308→          hadVoucher ? ` — voucher ${created.voucherCode} terpakai` : ""
   309→   309→        }. Total bayar ${formatIDR(finalTotal)}. Ayam geprekmu segera dimasak!`,
   310→   310→        actions: [
   311→   311→          {
   312→   312→            label: "Lihat Pesanan",
   313→   313→            onClick: () => go("orders"),
   314→   314→          },
   315→   315→        ],
   316→   316→      });
   317→   317→    } catch {
   318→   318→      showNotice({ type: "error", title: "Koneksi Bermasalah", message: "Tidak dapat menghubungi server." });
   319→   319→    } finally {
   320→   320→      setSubmitting(false);
   321→   321→    }
   322→   322→  };
   323→   323→

   326→   326→  const qrisSheet = payOrder ? (
   327→   327→    <QrisPaySheet
   328→   328→      order={payOrder}
   329→   329→      variant="customer"
   330→   330→      onClose={() => {
   331→   331→        setPayOrder(null);
   332→   332→        showNotice({
   333→   333→          type: "success",
   334→   334→          mascot: true,
   335→   335→          title: "Pesanan Berhasil! 🎉",
   336→   336→          message: `Pesanan ${payOrder.code} sudah masuk dapur. Selesaikan pembayaran QRIS ${formatIDR(
   337→   337→            payOrder.total
   338→   338→          )} lewat tombol "Bayar QRIS" di daftar pesanan bila belum.`,
   339→   339→          actions: [
   340→   340→            {
   341→   341→              label: "Lihat Pesanan",
   342→   342→              onClick: () => go("orders"),
   343→   343→            },
   344→   344→          ],
   345→   345→        });
   346→   346→      }}
   347→   347→      onPaid={() => void refreshOrders("mine")}
   348→   348→    />
   349→   349→  ) : null;
   350→   350→
   351→   351→  if (cart.items.length === 0) {
   352→   352→    return (
   353→   353→      <>
   354→   354→        {qrisSheet}
   355→   355→        <div className="flex min-h-[70dvh] flex-col items-center justify-center gap-3 px-8 text-center">
   356→   356→          <span className="flex h-20 w-20 items-center justify-center rounded-full bg-orange-50 text-4xl ring-4 ring-orange-100">
   357→   357→            🛒
   358→   358→          </span>
   359→   359→          <h2 className="text-lg font-black text-neutral-800">Keranjang Masih Kosong</h2>
   360→   360→          <p className="text-sm text-neutral-400">
   361→   361→            Ayamnya kesepian nih… yuk pilih menu geprek favoritmu dulu!
   362→   362→          </p>
   363→   363→          <button
   364→   364→            onClick={() => go("menu")}
   365→   365→            className="press-effect mt-2 rounded-full bg-orange-600 px-6 py-2.5 text-sm font-extrabold text-white shadow-lg hover:bg-orange-700"
   366→   366→          >
   367→   367→            Lihat Menu
   368→   368→          </button>
   369→   369→        </div>
   370→   370→      </>
   371→   371→    );
   372→   372→  }
   373→   373→
   374→   374→  return (
   375→   375→    <div className="space-y-4 px-4 pt-[calc(0.75rem+env(safe-area-inset-top,0px))]">
   376→   376→      <header>
   377→   377→        <h1 className="text-xl font-black text-neutral-900">Keranjang 🛒</h1>
   378→   378→        <p className="text-xs text-neutral-400">{cart.totalItems()} item siap digeprek</p>
   379→   379→      </header>
   380→   380→
   381→   381→      {/* Daftar item */}
   382→   382→      <section className="space-y-2.5">
   383→   383→        {cart.items.map((item, i) => (
   384→   384→          <motion.div
   385→   385→            key={item.productId}
   386→   386→            layout={!prefersReduced}
   387→   387→            initial={prefersReduced ? false : { opacity: 0, x: -14 }}
   388→   388→            animate={{ opacity: 1, x: 0 }}
   389→   389→            transition={{ delay: i * 0.04 }}
   390→   390→            className="flex items-center gap-3 rounded-2xl border border-orange-100 bg-white p-3 shadow-sm"
   391→   391→          >
   392→   392→            <ProductThumb
   393→   393→              product={
   394→   394→                products.find((p) => p.id === item.productId) ?? {
   395→   395→                  id: item.productId,
   396→   396→                  updatedAt: "",
   397→   397→                  emoji: item.emoji,
   398→   398→                  imageColor: item.imageColor,
   399→   399→                  hasImage: false,
   400→   400→                }
   401→   401→              }
   402→   402→              className="h-14 w-14 text-2xl"
   403→   403→            />
   404→   404→            <div className="min-w-0 flex-1">
   405→   405→              <p className="line-clamp-1 text-sm font-bold text-neutral-800">{item.name}</p>

   407→   407→            </div>
   408→   408→            <div className="flex items-center gap-1 rounded-full bg-orange-50 p-0.5">

   414→   414→                <Minus className="h-3.5 w-3.5" />
   415→   415→              </button>
   416→   416→              <span className="w-6 text-center text-sm font-black text-orange-800">{item.qty}</span>
