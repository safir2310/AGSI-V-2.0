     1→/**
     2→ * Penyedia QRIS via DOKU (https://api-sandbox.doku.com / https://api.doku.com).
     3→ *
     4→ * DOKU punya DUA jalur integrasi — adapter ini memakai yang kredensialnya
     5→ * tersedia (jalur SNAP REST lebih diutamakan bila lengkap):
     6→ *
     7→ *  1) JALUR SNAP REST — QRIS MPM Direct API (lihat developers.doku.com):
     8→ *     a. Token B2B      : POST /authorization/v1/access-token/b2b
     9→ *        X-SIGNATURE    : HMAC_SHA512(clientSecret, clientId+"|"+timestamp)
    10→ *                         atau SHA256withRSA(privateKey, sama string) bila
    11→ *                         DOKU_PRIVATE_KEY diset.
    12→ *     b. Generate QRIS  : POST /snap-adapter/b2b/v1.0/qr/qr-mpm-generate
    13→ *        X-SIGNATURE    : HMAC_SHA512(clientSecret, "POST:<path>:<accessToken>:
    14→ *                         <hex(sha256(minifiedBody))>:<timestamp>")
    15→ *        -> response.qrContent = string QRIS EMVCo siap dirender.
    16→ *     c. Query status   : POST /snap-adapter/b2b/v1.0/qr/qr-mpm-query
    17→ *        -> latestTransactionStatus: "00" sukses, "01"/"02"/"03" menunggu,
    18→ *           "05"/"06"/"07"/"08" gagal/kedaluwarsa.
    19→ *     d. Expire QR      : POST /snap-adapter/b2b/v1.0/qr/qr-expire
    20→ *
    21→ *  2) JALUR API KEY AGENTIC (MCP DOKU — JSON-RPC POST /doku-mcp-server/mcp,
    22→ *     Authorization: Basic base64(apiKey + ":")):
    23→ *     a. tools/call create_qris_payment -> QRIS dinamis (amount, invoice).
    24→ *     b. tools/call get_transaction_by_invoice_number -> cari status transaksi.
    25→ *     Kedua tool dipanggil langsung dari backend (server-to-server).
    26→ *
    27→ * Alur pelunasan (kebijakan toko — hanya SISTEM yang boleh melunaskan):
    28→ *   polling /api/orders/[id]/qris/status memanggil getQrisDokuStatus()
    29→ *   (query ke DOKU) -> PAID hanya bila DOKU sendiri mengatakan dana masuk;
    30→ *   webhook DOKU (/api/webhooks/qris) juga SELALU konfirmasi ulang lewat
    31→ *   query sebelum memanggil markOrderPaidBySystem() — fail-closed.
    32→ */

    68→/**
    69→ * Waktu WIB berformat SNAP "YYYY-MM-DDTHH:mm:ss+07:00" — format yang
    70→ * diminta X-TIMESTAMP & validityPeriod DOKU (waktu lokal Jakarta UTC+7).
    71→ */
    72→function wibTimestamp(d: Date = new Date()): string {
    73→  const shifted = new Date(d.getTime() + 7 * 60 * 60 * 1000);
    74→  return `${shifted.toISOString().slice(0, 19)}+07:00`;
    75→}

    90→/** Pemetaan kode status SNAP DOKU -> status baku aplikasi */
    91→function snapStatusToGateway(raw: unknown): GatewayStatus {
    92→  const s = String(raw ?? "").trim();
    93→  switch (s) {
    94→    case "00": // Success
    95→      return "PAID";
    96→    case "01": // Initiated
    97→    case "02": // Paying
    98→    case "03": // Pending
    99→    case "09": // Denied... masih bisa berubah — perlakukan menunggu
   100→      return "PENDING";
   101→    case "04": // Refunded
   102→    case "05": // Canceled
   103→    case "06": // Failed
   104→    case "07": // Void
   105→    case "08": // Expired
   106→      return "EXPIRED";
   107→    default:
   108→      return "UNKNOWN";
   109→  }
   110→}

   112→/** Cari string QRIS EMVCo di dalam objek respons apa pun (toleran struktur) */
   113→function findQrString(value: unknown, depth = 0): string | undefined {
   114→  if (depth > 6 || value == null) return undefined;
   115→  if (typeof value === "string") {
   116→    const v = value.trim();
   117→    if (v.startsWith("000201") || v.includes("ID.CO.QRIS.WWW")) return v;
   118→    return undefined;
   119→  }
   120→  if (Array.isArray(value)) {
   121→    for (const item of value) {
   122→      const hit = findQrString(item, depth + 1);
   123→      if (hit) return hit;
   124→    }
   125→    return undefined;
   126→  }
   127→  if (typeof value === "object") {
   128→    for (const v of Object.values(value as Record<string, unknown>)) {
   129→      const hit = findQrString(v, depth + 1);
   130→      if (hit) return hit;
   131→    }
   132→  }
   133→  return undefined;
   134→}

   136→/** Urai teks hasil tool MCP — berupa JSON string (kadang dibungkus) */
   137→function parseMcpText(text: string): Record<string, unknown> | null {
   138→  const trimmed = text.trim();
   139→  // Kadang respons diawali "..." langsung JSON; coba parse utuh dulu
   140→  try {
   141→    const parsed = JSON.parse(trimmed) as unknown;
   142→    if (parsed && typeof parsed === "object" && !Array.isArray(parsed)) {
   143→      return parsed as Record<string, unknown>;
   144→    }
   145→  } catch {
   146→    // bukan JSON utuh — coba ekstrak bagian {..} pertama
   147→  }
   148→  const start = trimmed.indexOf("{");
   149→  const end = trimmed.lastIndexOf("}");
   150→  if (start >= 0 && end > start) {
   151→    try {
   152→      const parsed = JSON.parse(trimmed.slice(start, end + 1)) as unknown;
   153→      if (parsed && typeof parsed === "object" && !Array.isArray(parsed)) {
   154→        return parsed as Record<string, unknown>;
   155→      }
   156→    } catch {
   157→      return null;
   158→    }
   159→  }
   160→  return null;
   161→}

   163→/** Ambil pesan galat terbaca dari objek galat DOKU */
   164→function messageOf(obj: Record<string, unknown> | null, fallback: string): string {
   165→  if (!obj) return fallback;
   166→  const parts: string[] = [];
   167→  for (const key of ["responseMessage", "message", "error", "errorMessage"]) {
   168→    const v = obj[key];
   169→    if (typeof v === "string" && v.trim()) parts.push(v.trim());
   170→  }
   171→  return parts.join(" — ") || fallback;
   172→}

   178→// Cache token B2B (berlaku 900 detik) — satu token dipakai ulang semua request
   179→let cachedToken: { token: string; expiresAt: number } | null = null;
   180→
   181→/** HMAC_SHA512 base64 — signature simetris SNAP */
   182→function hmac512(secret: string, stringToSign: string): string {
   183→  return createHmac("sha512", secret).update(stringToSign).digest("base64");
   184→}
   185→
   186→/** SHA256withRSA base64 — signature asimetris SNAP (bila private key diset) */
   187→function rsaSign(privateKeyPem: string, stringToSign: string): string {
   188→  const signer = createSign("RSA-SHA256");
   189→  signer.update(stringToSign);
   190→  return signer.sign(privateKeyPem, "base64");
   191→}
   192→
   193→/** Normalisasi PEM satu baris (env Vercel) -> PEM utuh */
   194→function normalizePem(raw: string): string {
   195→  const v = raw.trim();
   196→  if (!v) return "";
   197→  const withBreaks = v.includes("\\n") ? v.replace(/\\n/g, "\n") : v;
   198→  if (withBreaks.includes("-----BEGIN")) return withBreaks.trim();
   199→  return `-----BEGIN PRIVATE KEY-----\n${withBreaks}\n-----END PRIVATE KEY-----`;
   200→}

   202→/**
   203→ * POST /authorization/v1/access-token/b2b — token B2B (cache 13 menit).
   204→ * Signature: simetris HMAC (default) atau asimetris RSA (bila private key).
   205→ */
   206→async function getAccessToken(): Promise<string> {
   207→  if (cachedToken && cachedToken.expiresAt > Date.now() + 30_000) {
   208→    return cachedToken.token;
   209→  }
   210→  if (!DOKU_CLIENT_ID || !DOKU_CLIENT_SECRET) {
   211→    throw new QrisGatewayError("Kredensial DOKU (Client ID/Secret) belum lengkap.");
   212→  }
   213→  const path = "/authorization/v1/access-token/b2b";
   214→  const timestamp = wibTimestamp();
   215→  const stringToSign = `${DOKU_CLIENT_ID}|${timestamp}`;
   216→  const signature = DOKU_PRIVATE_KEY
   217→    ? rsaSign(normalizePem(DOKU_PRIVATE_KEY), stringToSign)
   218→    : hmac512(DOKU_CLIENT_SECRET, stringToSign);
   219→
   220→  const controller = new AbortController();
   221→  const timer = setTimeout(() => controller.abort(), 12_000);
   222→  let httpStatus = 0;
   223→  let body: Record<string, unknown>;
   224→  try {
   225→    const res = await fetch(`${DOKU_API_BASE}${path}`, {
   226→      method: "POST",
   227→      headers: {
   228→        "Content-Type": "application/json",
   229→        "X-CLIENT-KEY": DOKU_CLIENT_ID,
   230→        "X-TIMESTAMP": timestamp,
   231→        "X-SIGNATURE": signature,
   232→      },
   233→      body: JSON.stringify({ grantType: "client_credentials" }),
   234→      signal: controller.signal,
   235→      cache: "no-store",
   236→    });
   237→    httpStatus = res.status;
   238→    body = (await res.json().catch(() => ({}))) as Record<string, unknown>;
   239→  } catch (err) {
   240→    if (err instanceof Error && err.name === "AbortError") {
   241→      throw new QrisGatewayError("DOKU tidak merespons saat ambil token (timeout).");
   242→    }
   243→    throw new QrisGatewayError("Tidak dapat menghubungi DOKU (token).");
   244→  } finally {
   245→    clearTimeout(timer);
   246→  }
   247→
   248→  const token = typeof body.accessToken === "string" ? body.accessToken : "";
   249→  if (httpStatus >= 300 || !token) {
   250→    throw new QrisGatewayError(
   251→      `DOKU menolak permintaan token (HTTP ${httpStatus}). ${messageOf(body, "Periksa DOKU_CLIENT_ID / DOKU_CLIENT_SECRET.")}`,
   252→      httpStatus === 401 ? 401 : undefined
   253→    );
   254→  }
   255→  const expiresIn = Number(body.expiresIn ?? 900) || 900;
   256→  cachedToken = { token, expiresAt: Date.now() + expiresIn * 1000 };
   257→  return token;
   258→}

   260→/** POST ke endpoint SNAP dengan header signature lengkap */
   261→async function snapPost(
   262→  path: string,
   263→  payload: Record<string, unknown>,
   264→  timeoutMs = 15_000
   265→): Promise<Record<string, unknown>> {
   266→  const accessToken = await getAccessToken();
   267→  const timestamp = wibTimestamp();
   268→  const minified = JSON.stringify(payload);
   269→  const bodyHash = createHash("sha256").update(minified).digest("hex").toLowerCase();
   270→  const stringToSign = `POST:${path}:${accessToken}:${bodyHash}:${timestamp}`;
   271→  const signature = hmac512(DOKU_CLIENT_SECRET, stringToSign);
   272→
   273→  const controller = new AbortController();
   274→  const timer = setTimeout(() => controller.abort(), timeoutMs);
   275→  let httpStatus = 0;
   276→  let body: Record<string, unknown>;
   277→  try {
   278→    const res = await fetch(`${DOKU_API_BASE}${path}`, {
   279→      method: "POST",
   280→      headers: {
   281→        "Content-Type": "application/json",
   282→        Accept: "application/json",
   283→        "X-PARTNER-ID": DOKU_CLIENT_ID,
   284→        "X-EXTERNAL-ID": externalId(),
   285→        "X-TIMESTAMP": timestamp,
   286→        "X-SIGNATURE": signature,
   287→        Authorization: `Bearer ${accessToken}`,
   288→        "CHANNEL-ID": "H2H",
   289→      },
   290→      body: minified,
   291→      signal: controller.signal,
   292→      cache: "no-store",
   293→    });
   294→    httpStatus = res.status;
   295→    body = (await res.json().catch(() => ({}))) as Record<string, unknown>;
   296→  } catch (err) {
   297→    if (err instanceof Error && err.name === "AbortError") {
   298→      throw new QrisGatewayError("DOKU tidak merespons (timeout).");
   299→    }
   300→    throw new QrisGatewayError("Tidak dapat menghubungi DOKU. Periksa koneksi internet server.");
   301→  } finally {
   302→    clearTimeout(timer);
   303→  }
   304→
   305→  if (httpStatus >= 400) {
   306→    throw new QrisGatewayError(
   307→      `DOKU menolak ${path} (HTTP ${httpStatus}). ${messageOf(body, "")}`,
   308→      httpStatus === 404 ? 404 : httpStatus === 401 ? 401 : undefined
   309→    );
   310→  }
   311→  return body;
   312→}

   372→/** Panggil satu tool MCP DOKU; kembalikan objek hasil terurai */
   373→async function mcpCallTool(
   374→  toolName: string,
   375→  toolRequest: Record<string, unknown>
   376→): Promise<Record<string, unknown> | null> {
   377→  if (!DOKU_APIKEY_READY) {
   378→    throw new QrisGatewayError("API key DOKU belum dikonfigurasi di .env (DOKU_API_KEY).");
   379→  }
   380→  const basic = Buffer.from(`${DOKU_API_KEY}:`).toString("base64");
   381→  const id = ++mcpRequestId;
   382→
   383→  const controller = new AbortController();
   384→  const timer = setTimeout(() => controller.abort(), 30_000);
   385→  let httpStatus = 0;
   386→  let raw = "";
   387→  try {
   388→    const res = await fetch(`${DOKU_API_BASE}${DOKU_MCP_PATH}`, {
   389→      method: "POST",
   390→      headers: {
   391→        Authorization: `Basic ${basic}`,
   392→        "Content-Type": "application/json",
   393→        Accept: "application/json, text/event-stream",
   394→      },
   395→      body: JSON.stringify({
   396→        jsonrpc: "2.0",
   397→        id,
   398→        method: "tools/call",
   399→        params: { name: toolName, arguments: { toolRequest } },
   400→      }),
   401→      signal: controller.signal,
   402→      cache: "no-store",
   403→    });
   404→    httpStatus = res.status;
   405→    raw = await res.text();
   406→  } catch (err) {
   407→    if (err instanceof Error && err.name === "AbortError") {
   408→      throw new QrisGatewayError("MCP DOKU tidak merespons (timeout).");
   409→    }
   410→    throw new QrisGatewayError("Tidak dapat menghubungi MCP DOKU. Periksa koneksi internet server.");
   411→  } finally {
   412→    clearTimeout(timer);
   413→  }
   414→
   415→  if (httpStatus >= 400) {
   416→    throw new QrisGatewayError(
   417→      `MCP DOKU menolak panggilan (HTTP ${httpStatus}).`,
   418→      httpStatus === 401 ? 401 : undefined
   419→    );
   420→  }
   421→
   422→  // Respons MCP bisa JSON murni ATAU SSE ("data: {...}") — urai toleran
   423→  let envelope: Record<string, unknown> | null = null;
   424→  const direct = parseMcpText(raw);
   425→  if (direct && typeof direct.result === "object") {
   426→    envelope = direct;
   427→  } else {
   428→    for (const line of raw.split("\n")) {
   429→      const trimmedLine = line.trim();
   430→      if (!trimmedLine.startsWith("data:")) continue;
   431→      const parsed = parseMcpText(trimmedLine.slice(5));
   432→      if (parsed && parsed.id === id && typeof parsed.result === "object") {
   433→        envelope = parsed;
   434→        break;
   435→      }
   436→    }
   437→  }
   438→  if (!envelope) {
   439→    throw new QrisGatewayError("Respons MCP DOKU tidak dapat diurai.");
   440→  }
   441→  const result = envelope.result as Record<string, unknown> | undefined;
   442→  if (!result) return null;
   443→  if (result.isError === true) {
   444→    throw new QrisGatewayError(`Tool DOKU ${toolName} gagal. ${messageOf(null, raw.slice(0, 300))}`);
   445→  }
   446→  const content = result.content;
   447→  if (!Array.isArray(content) || content.length === 0) return null;
   448→  const first = content[0] as { type?: unknown; text?: unknown };
   449→  if (typeof first?.text !== "string") return null;
   450→  const parsed = parseMcpText(first.text);
   451→  if (parsed && (typeof parsed.error === "string" || typeof parsed.errorMessage === "string")) {
   452→    throw new QrisGatewayError(
   453→      `DOKU menolak ${toolName}: ${messageOf(parsed, "Service temporarily unavailable")}`
   454→    );
   455→  }
   456→  return parsed;
   457→}

   479→  }
   480→  return qr;
   481→}

   483→/** Cari entri transaksi sesuai invoice di dalam objek respons (rekursif) */
   484→function findTransactionEntry(
   485→  value: unknown,
   486→  invoice: string,
   487→  depth = 0
   488→): Record<string, unknown> | null {
   489→  if (depth > 6 || value == null) return null;
   490→  if (Array.isArray(value)) {
   491→    for (const item of value) {
   492→      const hit = findTransactionEntry(item, invoice, depth + 1);
   493→      if (hit) return hit;
   494→    }
   495→    return null;
   496→  }
   497→  if (typeof value === "object") {
   498→    const obj = value as Record<string, unknown>;
   499→    for (const key of [
   500→      "invoiceNumber",
   501→      "invoice_number",
   502→      "partnerReferenceNo",
   503→      "partner_reference_no",
   504→      "originalPartnerReferenceNo",
   505→      "originalReferenceNo",
   506→      "referenceNo",
   507→    ]) {
   508→      const v = obj[key];
   509→      if (typeof v === "string" && v.trim() === invoice) return obj;
   510→    }
   511→    for (const v of Object.values(obj)) {
   512→      const hit = findTransactionEntry(v, invoice, depth + 1);
   513→      if (hit) return hit;
   514→    }
   515→  }
   516→  return null;
   517→}

   519→/** Status via MCP get_transaction_by_invoice_number */
   520→async function mcpQueryStatus(partnerReferenceNo: string): Promise<GatewayStatus> {
   521→  const now = new Date();
   522→  const ddmm = (d: Date) =>
   523→    `${String(d.getDate()).padStart(2, "0")}-${String(d.getMonth() + 1).padStart(2, "0")}-${d.getFullYear()}`;
   524→  const start = new Date(now.getTime() - 3 * 24 * 60 * 60 * 1000);
   525→  const end = new Date(now.getTime() + 24 * 60 * 60 * 1000);
   526→
   527→  let parsed: Record<string, unknown> | null;
   528→  try {
   529→    parsed = await mcpCallTool("get_transaction_by_invoice_number", {
   530→      invoiceNumber: partnerReferenceNo,
   531→      startDate: ddmm(start),
   532→      endDate: ddmm(end),
   533→      pageNumber: 1,
   534→      pageSize: 10,
   535→    });
   536→  } catch (err) {
   537→    const msg = err instanceof Error ? err.message.toLowerCase() : "";
   538→    // DOKU: laporan transaksi kosong / belum ada catatan = transaksi belum
   539→    // ada (belum dibayar) — bukan galat; QR tetap menunggu pembayaran.
   540→    if (msg.includes("searchfailed") || msg.includes("unable to complete search")) {
   541→      return "PENDING";
   542→    }
   543→    throw err;
   544→  }
   545→  if (!parsed) return "PENDING";
   546→
   547→  // DOKU: transaksi belum tercatat (belum dibayar) -> tetap menunggu.
   548→  const entry = findTransactionEntry(parsed, partnerReferenceNo);
   549→  if (!entry) return "PENDING";
   550→
   551→  // Kode SNAP (00/03/...) diutamakan; kalau tidak ada, pakai kata status.
   552→  const codeKeys = [
   553→    "latestTransactionStatus",
   554→    "transactionStatus",
   555→    "trxStatus",
   556→    "status_code",
   557→    "statusCode",
   558→  ];
   559→  for (const k of codeKeys) {
   560→    if (entry[k] != null) {
   561→      const mapped = snapStatusToGateway(entry[k]);
   562→      if (mapped !== "UNKNOWN") return mapped;
   563→    }
   564→  }
   565→  const wordKeys = ["transactionStatusDesc", "status", "transaction_status", "paymentStatus"];
   566→  for (const k of wordKeys) {
   567→    const v = entry[k];
   568→    if (typeof v === "string" && v.trim()) {
   569→      const s = v.trim().toLowerCase();
   570→      if (s === "success" || s === "success payment" || s === "paid") return "PAID";
   571→      if (s.includes("success")) return "PAID";
   572→      if (s.includes("pending") || s.includes("waiting") || s.includes("unpaid")) return "PENDING";
   573→      if (s.includes("expired") || s.includes("cancel") || s.includes("failed") || s.includes("refund"))
   574→        return "EXPIRED";
   575→    }
   576→  }
   577→  return "PENDING";
   578→}

   588→//     DOKU berisi kanal aktif merchant — QRIS tampil di sana begitu kanal

   590→//   GET  /orders/v1/status/{invoice} -> transaction.status

   592→// Timestamp\nRequest-Target\nDigest" (HMACSHA256= prefix).

   594→/** Header autentikasi legacy Non-SNAP — nilai TERKIRIM persis (HMACSHA256=) */
   595→function legacyHeaders(
   596→  target: string,
   597→  bodyRaw: string | null
   598→): Record<string, string> {
   599→  const requestId = externalId();
   600→  const ts = new Date().toISOString().replace(/\.\d{3}Z$/, "Z");
   601→  const parts = [
   602→    `Client-Id:${DOKU_CLIENT_ID}`,
   603→    `Request-Id:${requestId}`,
   604→    `Request-Timestamp:${ts}`,
   605→    `Request-Target:${target}`,
   606→  ];
   607→  if (bodyRaw !== null) {
   608→    parts.push(
   609→      `Digest:${createHash("sha256").update(bodyRaw).digest("base64")}`
   610→    );
   611→  }
   612→  return {
   613→    "Client-Id": DOKU_CLIENT_ID,
   614→    "Request-Id": requestId,
   615→    "Request-Timestamp": ts,
   616→    "Request-Target": target,
   617→    Signature: `HMACSHA256=${createHmac("sha256", DOKU_CLIENT_SECRET)
   618→      .update(parts.join("\n"))
   619→      .digest("base64")}`,
   620→    ...(bodyRaw !== null ? { "Content-Type": "application/json" } : {}),
   621→  };
   622→}

   624→/** Pesan galat terbaca dari respons legacy (message bisa string/array) */
   625→function legacyErrorMessage(
   626→  body: Record<string, unknown>,
   627→  fallback: string
   628→): string {
   629→  const err = body.error;
   630→  if (err && typeof err === "object" && !Array.isArray(err)) {
   631→    const m = (err as Record<string, unknown>).message;
   632→    if (typeof m === "string" && m.trim()) return m;
   633→  }
   634→  const msg = body.message;
   635→  if (typeof msg === "string" && msg.trim()) return msg;
   636→  if (Array.isArray(msg)) {
   637→    const joined = msg.filter((v) => typeof v === "string").join(", ");
   638→    if (joined) return joined;
   639→  }
   640→  if (typeof body.responseMessage === "string" && body.responseMessage.trim()) {
   641→    return body.responseMessage;
   642→  }
   643→  return fallback;
   644→}

   646→/** POST legacy dengan timeout + galat terurai */
   647→async function legacyPost(
   648→  path: string,
   649→  payload: Record<string, unknown>,
   650→  timeoutMs = 20_000
   651→): Promise<Record<string, unknown>> {
   652→  const raw = JSON.stringify(payload);
   653→  const controller = new AbortController();
   654→  const timer = setTimeout(() => controller.abort(), timeoutMs);
   655→  let httpStatus = 0;
   656→  let body: Record<string, unknown>;
   657→  try {
   658→    const res = await fetch(`${DOKU_API_BASE}${path}`, {
   659→      method: "POST",
   660→      headers: legacyHeaders(path, raw),
   661→      body: raw,
   662→      signal: controller.signal,
   663→      cache: "no-store",
   664→    });
   665→    httpStatus = res.status;
   666→    body = (await res.json().catch(() => ({}))) as Record<string, unknown>;
   667→  } catch (err) {
   668→    if (err instanceof Error && err.name === "AbortError") {
   669→      throw new QrisGatewayError("DOKU Checkout tidak merespons (timeout).");
   670→    }
   671→    throw new QrisGatewayError(
   672→      "Tidak dapat menghubungi DOKU Checkout. Periksa koneksi internet server."
   673→    );
   674→  } finally {
   675→    clearTimeout(timer);
   676→  }
   677→  if (httpStatus >= 300) {
   678→    throw new QrisGatewayError(
   679→      `DOKU Checkout menolak ${path} (HTTP ${httpStatus}). ${legacyErrorMessage(body, "")}`.trim(),
   680→      httpStatus === 404 ? 404 : httpStatus === 401 ? 401 : undefined
   681→    );
   682→  }
   683→  return body;
   684→}

   727→/** Cari URL halaman bayar DOKU di kedalaman mana pun (toleran struktur) */
   728→function findPaymentUrl(value: unknown, depth = 0): string | undefined {
   729→  if (depth > 6 || value == null) return undefined;
   730→  if (typeof value === "string") {
   731→    const v = value.trim();
   732→    if (
   733→      v.startsWith("https://") &&
   734→      (v.includes("checkout.doku.com") || v.includes("/checkout-link-v2/"))
   735→    ) {
   736→      return v;
   737→    }
   738→    return undefined;
   739→  }
   740→  if (Array.isArray(value)) {
   741→    for (const item of value) {
   742→      const hit = findPaymentUrl(item, depth + 1);
   743→      if (hit) return hit;
   744→    }
   745→    return undefined;
   746→  }
   747→  if (typeof value === "object") {
   748→    for (const v of Object.values(value as Record<string, unknown>)) {
   749→      const hit = findPaymentUrl(v, depth + 1);
   750→      if (hit) return hit;
   751→    }
   752→  }
   753→  return undefined;
   754→}

   833→    | Record<string, unknown>

   841→        `${digits.slice(0, 4)}-${digits.slice(4, 6)}-${digits.slice(6, 8)}T${digits.slice(8, 10)}:${digits.slice(10, 12)}:${digits.slice(12, 14)}+07:00`
   842→      );
   843→      if (!Number.isNaN(d.getTime())) expiresAt = d.toISOString();
   844→    } else {
   845→      const d = new Date(exp);
   846→      if (!Number.isNaN(d.getTime())) expiresAt = d.toISOString();
   847→    }
   848→  }
   849→  return {
   850→    trxId: partnerReferenceNo,
   851→    paymentUrl,
   852→    expiresAt: expiresAt ?? undefined,
   853→  };

  1019→/**
  1020→ * Verifikasi signature HTTP NOTIFICATION DOKU (HMAC-SHA512 simetris).
  1021→ * Rumus SNAP: HMAC_SHA512(clientSecret, "POST:<path>:<accessToken>:
  1022→ * <hex(sha256(body))>:<timestamp>"). Notifikasi tidak membawa access token —
  1023→ * dicoba juga varian tanpa komponen token (fail-closed: hanya cocok yang benar).
  1024→ */
  1025→export function verifyDokuWebhookSignature(opts: {
  1026→  signature: string;
  1027→  timestamp: string;
  1028→  rawBody: string;
  1029→  paths: string[];
  1030→}): boolean {
  1031→  if (!DOKU_CLIENT_SECRET) return false;
  1032→  const got = opts.signature.trim();
  1033→  if (!got) return false;
  1034→  const bodyHash = createHash("sha256")
  1035→    .update(opts.rawBody)
  1036→    .digest("hex")
  1037→    .toLowerCase();
  1038→  const candidates: string[] = [];
  1039→  for (const p of opts.paths) {
  1040→    candidates.push(`POST:${p}::${bodyHash}:${opts.timestamp}`);
  1041→    candidates.push(`POST:${p}:::${opts.timestamp}`);
  1042→  }
  1043→  for (const s of candidates) {
  1044→    const expected = hmac512(DOKU_CLIENT_SECRET, s);
  1045→    if (expected.length === got.length && expected === got) return true;
  1046→  }
  1047→  return false;
  1048→}
  1049→
  1050→/**
  1051→ * Verifikasi signature NOTIFIKASI legacy (Non-SNAP / DOKU Checkout).
  1052→ * Header: Client-Id, Request-Id, Request-Timestamp, Request-Target, Signature.
  1053→ * Rumus: "HMACSHA256=" + Base64(HMAC-SHA256(secret,
  1054→ *   "Client-Id:..\nRequest-Id:..\nRequest-Timestamp:..\nRequest-Target:..\nDigest:<base64(sha256(body))>"))
  1055→ * Fail-closed: hanya signature yang benar diterima.
  1056→ */
  1057→export function verifyDokuLegacyWebhookSignature(opts: {
  1058→  signature: string;
  1059→  clientId?: string;
  1060→  requestId: string;
  1061→  timestamp: string;
  1062→  target: string;
  1063→  rawBody: string;
  1064→}): boolean {
  1065→  if (!DOKU_CLIENT_SECRET) return false;
  1066→  const got = opts.signature.trim();
  1067→  if (!got) return false;
  1068→  const digest = createHash("sha256").update(opts.rawBody).digest("base64");
  1069→  const parts = [
  1070→    `Client-Id:${opts.clientId?.trim() || DOKU_CLIENT_ID}`,
  1071→    `Request-Id:${opts.requestId}`,
  1072→    `Request-Timestamp:${opts.timestamp}`,
  1073→    `Request-Target:${opts.target.startsWith("/") ? opts.target : `/${opts.target}`}`,
  1074→    `Digest:${digest}`,
  1075→  ];
  1076→  const expected = `HMACSHA256=${createHmac("sha256", DOKU_CLIENT_SECRET)
  1077→    .update(parts.join("\n"))
  1078→    .digest("base64")}`;
  1079→  return expected.length === got.length && expected === got;
  1080→}
