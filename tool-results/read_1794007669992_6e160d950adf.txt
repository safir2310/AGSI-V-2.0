    48→    48→/**
    49→    49→ * Layar pembayaran QRIS — pelanggan maupun kasir memakai layar yang sama.
    50→    50→ *
    51→    51→ * VERIFIKASI OTOMATIS: setelah QR tampil, layar ini memantau status ke server
    52→    52→ * tiap beberapa detik; server bertanya ke penyedia QRIS "sudah masuk belum
    53→    53→ * dananya?" dan melunaskan pesanan begitu dana masuk. TIDAK ADA tombol
    54→    54→ * "Saya Sudah Bayar" / "Uang Sudah Masuk" — kasir memang tidak boleh
    55→    55→ * (dan tidak bisa) melunaskan pesanan QRIS secara manual.
    56→    56→ */
    57→    57→export function QrisPaySheet({
    58→    58→  order,
    59→    59→  variant,
    60→    60→  onClose,
    61→    61→  onPaid,
    62→    62→  onPrintReceipt,
    63→    63→}: {
    64→    64→  order: QrisPayOrder;
    65→    65→  variant: "customer" | "pos";
    66→    66→  onClose: () => void;
    67→    67→  /** dipanggil saat status pembayaran berubah menjadi lunas (mis. refresh daftar) */
    68→    68→  onPaid?: () => void | Promise<void>;
    69→    69→  /** POS: cetak struk setelah lunas (bila disediakan halaman pemanggil) */
    70→    70→  onPrintReceipt?: () => void | Promise<void>;
    71→    71→}) {
    72→    72→  const prefersReduced = useReducedMotion();
    73→    73→  const isPos = variant === "pos";
    74→    74→
    75→    75→  const [info, setInfo] = useState<QrisInfo | null>(null);
    76→    76→  const [loadError, setLoadError] = useState<string | null>(null);
    77→    77→  const [phase, setPhase] = useState<Phase>("scan");
    78→    78→  const [secondsLeft, setSecondsLeft] = useState(DURATION_SEC);
    79→    79→  const [checkingNow, setCheckingNow] = useState(false);
    80→    80→  const [regenerating, setRegenerating] = useState(false);
    81→    81→  /** true saat simulasi dana masuk (sandbox) sedang diproses */
    82→    82→  const [simulating, setSimulating] = useState(false);
    83→    83→  /** pesan error simulasi (ditampilkan di bawah tombolnya) */
    84→    84→  const [simulateError, setSimulateError] = useState<string | null>(null);
    85→    85→  /** status terakhir dari penyedia (mis. "EXPIRED") */
    86→    86→  const [gatewayStatus, setGatewayStatus] = useState<string | null>(null);
    87→    87→  /** gangguan sementara saat memantau (diciutkan agar tidak bikin cemas) */
    88→    88→  const [monitorNote, setMonitorNote] = useState<string | null>(null);
    89→    89→  /** batas waktu pesanan (createdAt + 30 menit) dalam ms epoch */
    90→    90→  const [orderDeadline, setOrderDeadline] = useState<number | null>(null);
    91→    91→  const paidNotified = useRef(false);
    92→    92→
    93→    93→  const notifyPaid = useCallback(() => {
    94→    94→    if (paidNotified.current) return;
    95→    95→    paidNotified.current = true;
    96→    96→    void onPaid?.();
    97→    97→  }, [onPaid]);
    98→    98→
    99→    99→  /** Ambil kode QRIS dari server (gateway bila aktif, fallback lokal) */
   100→   100→  const fetchPayload = useCallback(async () => {
   101→   101→    setLoadError(null);
   102→   102→    setInfo(null);
   103→   103→    setGatewayStatus(null);
   104→   104→    try {
   105→   105→      const res = await apiFetch(`/api/orders/${order.id}/qris`, { cache: "no-store" });
   106→   106→      const data = await res.json();
   107→   107→      if (!res.ok) {
   108→   108→        // Pesanan ternyata sudah lunas → langsung tampilkan sukses
   109→   109→        if (typeof data.error === "string" && data.error.toLowerCase().includes("lunas")) {
   110→   110→          setPhase("paid");
   111→   111→          notifyPaid();
   112→   112→          return;
   113→   113→        }
   114→   114→        // Pesanan dibatalkan otomatis (lewat batas 30 menit) → tampilan batal
   115→   115→        if (data.cancelled === true) {
   116→   116→          setPhase("cancelled");
   117→   117→          return;
   118→   118→        }
   119→   119→        setLoadError(data.error ?? "Gagal memuat kode QRIS.");
   120→   120→        return;
   121→   121→      }
   122→   122→      if (data.paid === true) {
   123→   123→        setPhase("paid");
   124→   124→        notifyPaid();
   125→   125→        return;
   126→   126→      }
   127→   127→      const expiresAt = data.expiresAt ? new Date(data.expiresAt as string) : null;
   128→   128→      const cancelMs = data.cancelAt ? new Date(data.cancelAt as string).getTime() : null;
   129→   129→      if (cancelMs) setOrderDeadline(cancelMs);
   130→   130→      setInfo({
   131→   131→        payload: (data.payload as string) ?? null,
   132→   132→        imageUrl: (data.imageUrl as string) ?? null,
   133→   133→        mode: data.mode === "GATEWAY" ? "GATEWAY" : "LOCAL",
   134→   134→        autoVerify: !!data.autoVerify,
   135→   135→        sandbox: data.sandbox === true,
   136→   136→      });
   137→   137→      // Hitung mundur pakai batas TERKETAT: umur kode gateway vs batas pesanan
   138→   138→      const nowMs = Date.now();
   139→   139→      const gatewaySec = expiresAt
   140→   140→        ? Math.max(0, Math.floor((expiresAt.getTime() - nowMs) / 1000))
   141→   141→        : DURATION_SEC;
   142→   142→      const orderSec = cancelMs
   143→   143→        ? Math.max(0, Math.floor((cancelMs - nowMs) / 1000))
   144→   144→        : DURATION_SEC;
   145→   145→      setSecondsLeft(Math.max(0, Math.min(DURATION_SEC, gatewaySec, orderSec)));
   146→   146→    } catch {
   147→   147→      setLoadError("Tidak dapat menghubungi server. Periksa koneksi lalu coba lagi.");
   148→   148→    }
   149→   149→  }, [order.id, notifyPaid]);
   150→   150→
   151→   151→  useEffect(() => {
   152→   152→    void fetchPayload();
   153→   153→  }, [fetchPayload]);
   154→   154→
   155→   155→  // Hitung waktu mundur (hanya saat fase scan)
   156→   156→  useEffect(() => {
   157→   157→    if (phase !== "scan") return;
   158→   158→    const t = setInterval(() => {
   159→   159→      setSecondsLeft((s) => (s > 0 ? s - 1 : 0));
   160→   160→    }, 1000);
   161→   161→    return () => clearInterval(t);
   162→   162→  }, [phase]);
   163→   163→
   164→   164→  // Tutup dengan tombol Escape
   165→   165→  useEffect(() => {
   166→   166→    const onKey = (e: KeyboardEvent) => {
   167→   167→      if (e.key === "Escape") onClose();
   168→   168→    };
   169→   169→    window.addEventListener("keydown", onKey);
   170→   170→    return () => window.removeEventListener("keydown", onKey);
   171→   171→  }, [onClose]);
   172→   172→
   173→   173→  // ===== Pemantauan verifikasi otomatis =====
   174→   174→  useEffect(() => {
   175→   175→    if (phase !== "scan" || !info) return;
   176→   176→    let alive = true;
   177→   177→    const poll = async (force = false) => {
   178→   178→      try {
   179→   179→        const res = await apiFetch(
   180→   180→          `/api/orders/${order.id}/qris/status${force ? "?force=1" : ""}`,
   181→   181→          { cache: "no-store" }
   182→   182→        );
   183→   183→        if (!res.ok || !alive) return;
   184→   184→        const data = await res.json();
   185→   185→        if (data.paymentStatus === "PAID") {
   186→   186→          setPhase("paid");
   187→   187→          notifyPaid();
   188→   188→          return;
   189→   189→        }
   190→   190→        // Pesanan batal otomatis (lewat batas 30 menit / dana telat)
   191→   191→        if (data.cancelled === true || data.status === "CANCELLED") {
   192→   192→          setMonitorNote((data.notice as string) ?? null);
   193→   193→          setPhase("cancelled");
   194→   194→          return;
   195→   195→        }
   196→   196→        if (data.cancelAt) setOrderDeadline(new Date(data.cancelAt as string).getTime());
   197→   197→        setGatewayStatus((data.gatewayStatus as string) ?? null);
   198→   198→        setMonitorNote((data.gatewayError as string) ?? null);
   199→   199→      } catch {
   200→   200→        // koneksi tersendat — coba lagi pada tick berikutnya
   201→   201→      }
   202→   202→    };
   203→   203→    void poll();
   204→   204→    const t = setInterval(() => void poll(), POLL_MS);
   205→   205→    return () => {
   206→   206→      alive = false;
   207→   207→      clearInterval(t);
   208→   208→    };
   209→   209→  }, [phase, info, order.id, notifyPaid]);
   210→   210→
   211→   211→  /** POS: paksa sistem memeriksa penyedia sekarang (tetap otomatis, bukan menandai lunas) */
   212→   212→  const checkNow = async () => {
   213→   213→    if (checkingNow) return;
   214→   214→    setCheckingNow(true);
   215→   215→    try {
   216→   216→      const res = await apiFetch(`/api/orders/${order.id}/qris/status?force=1`, {
   217→   217→        cache: "no-store",
   218→   218→      });
   219→   219→      const data = await res.json();
   220→   220→      if (res.ok && data.paymentStatus === "PAID") {
   221→   221→        setPhase("paid");
   222→   222→        notifyPaid();
   223→   223→      }
   224→   224→    } catch {
   225→   225→      /* abaikan — polling reguler akan menyusul */
   226→   226→    } finally {
   227→   227→      setCheckingNow(false);
   228→   228→    }
   229→   229→  };
   230→   230→
   231→   231→  /** Buat ulang kode bayar (transaksi lama kedaluwarsa) */
   232→   232→  const regenerate = async () => {
   233→   233→    if (regenerating) return;
   234→   234→    setRegenerating(true);
   235→   235→    await fetchPayload();
   236→   236→    setRegenerating(false);
   237→   237→  };
   238→   238→
   239→   239→  /** SANDBOX: tiru notifikasi "dana masuk" dari dompet digital ke gateway
   240→   240→   *  bawaan. Pesanan tetap dilunaskan OLEH SISTEM di server (jalur yang sama
   241→   241→   *  dengan verifikasi otomatis) — ini bukan tombol "tandai lunas" manual. */
   242→   242→  const simulatePay = async () => {
   243→   243→    if (simulating) return;
   244→   244→    setSimulating(true);
   245→   245→    setSimulateError(null);
   246→   246→    try {
   247→   247→      const res = await apiFetch(`/api/orders/${order.id}/qris/simulate-pay`, {
   248→   248→        method: "POST",
   249→   249→      });
   250→   250→      const data = await res.json();
   251→   251→      if (res.ok && data.paymentStatus === "PAID") {
   252→   252→        setPhase("paid");
   253→   253→        notifyPaid();
   254→   254→        return;
   255→   255→      }
   256→   256→      if (data.cancelled === true) {
   257→   257→        setMonitorNote(
   258→   258→          typeof data.error === "string" ? data.error : null
   259→   259→        );
   260→   260→        setPhase("cancelled");
   261→   261→        return;
   262→   262→      }
   263→   263→      setSimulateError(
   264→   264→        typeof data.error === "string" ? data.error : "Simulasi gagal — coba lagi."
   265→   265→      );
   266→   266→    } catch {
   267→   267→      setSimulateError("Tidak dapat menghubungi server. Periksa koneksi lalu coba lagi.");
   268→   268→    } finally {
   269→   269→      setSimulating(false);
   270→   270→    }
   271→   271→  };
   272→   272→
   273→   273→  const expired = secondsLeft <= 0;
   274→   274→  const codeStale = expired || gatewayStatus === "EXPIRED";
   275→   275→  const autoVerify = info?.autoVerify ?? false;
   276→   276→  /** Batas pesanan (30 menit) sudah terlewat — pesanan mati, jangan tawarkan buat ulang kode */
   277→   277→  const pastDeadline = orderDeadline !== null && Date.now() >= orderDeadline;
   278→   278→
   279→   279→  return (
   280→   280→    <AnimatePresence>
   281→   281→      <motion.div
   282→   282→        key="qris-overlay"
   283→   283→        initial={prefersReduced ? false : { opacity: 0 }}
   284→   284→        animate={{ opacity: 1 }}
   285→   285→        exit={prefersReduced ? undefined : { opacity: 0 }}
   286→   286→        className="fixed inset-0 z-[95] flex items-end justify-center bg-neutral-900/70 backdrop-blur-sm sm:items-center sm:p-4"
   287→   287→        role="dialog"
   288→   288→        aria-modal="true"
   289→   289→        aria-label="Pembayaran QRIS"
   290→   290→      >
   291→   291→        <motion.div
   292→   292→          initial={prefersReduced ? false : { y: 48, scale: 0.97 }}
   293→   293→          animate={{ y: 0, scale: 1 }}
   294→   294→          exit={prefersReduced ? undefined : { y: 48, opacity: 0 }}
   295→   295→          transition={{ type: "spring", stiffness: 380, damping: 34 }}
   296→   296→          className="flex max-h-[94dvh] w-full max-w-md flex-col overflow-hidden rounded-t-3xl bg-white shadow-2xl sm:rounded-3xl"
   297→   297→        >
   298→   298→          {/* ===== KEPALA ===== */}
   299→   299→          <header className="relative shrink-0 bg-gradient-to-r from-orange-700 via-orange-600 to-orange-500 px-5 py-4 text-white">
   300→   300→            <div className="flex items-center gap-3">
   301→   301→              <span className="flex h-10 w-10 items-center justify-center rounded-2xl bg-white/20 ring-1 ring-white/30">
   302→   302→                <ScanLine className="h-5 w-5" />
   303→   303→              </span>
   304→   304→              <div className="min-w-0 flex-1">
   305→   305→                <p className="text-sm font-black tracking-wide">Pembayaran QRIS</p>
   306→   306→                <p className="truncate text-[11px] font-semibold text-orange-100">
   307→   307→                  {order.code} • {isPos ? "Kasir" : "Aplikasi"}
   308→   308→                </p>
   309→   309→              </div>
   310→   310→              <button
   311→   311→                onClick={onClose}
   312→   312→                aria-label="Tutup"
   313→   313→                className="press-effect flex h-9 w-9 items-center justify-center rounded-full bg-white/15 hover:bg-white/30"
   314→   314→              >
   315→   315→                <X className="h-4.5 w-4.5" />
   316→   316→              </button>
   317→   317→            </div>
   318→   318→          </header>
   319→   319→
   320→   320→          {/* ===== ISI ===== */}
   321→   321→          <div className="scrollbar-slim overflow-y-auto px-5 pb-[max(1.25rem,env(safe-area-inset-bottom))] pt-4">
   322→   322→            {phase === "paid" ? (
   323→   323→              /* --------- BERHASIL (verifikasi otomatis) --------- */
   324→   324→              <div className="flex flex-col items-center gap-3 py-8 text-center">
   325→   325→                <motion.span
   326→   326→                  initial={prefersReduced ? false : { scale: 0.5, opacity: 0 }}
   327→   327→                  animate={{ scale: 1, opacity: 1 }}
   328→   328→                  transition={{ type: "spring", stiffness: 300, damping: 18 }}
   329→   329→                  className="flex h-20 w-20 items-center justify-center rounded-full bg-emerald-50 ring-4 ring-emerald-100"
   330→   330→                >
   331→   331→                  <BadgeCheck className="h-11 w-11 text-emerald-600" />
   332→   332→                </motion.span>
   333→   333→                <h2 className="text-lg font-black text-neutral-900">Pembayaran Diterima! 🎉</h2>
   334→   334→                <p className="max-w-[300px] text-xs leading-relaxed text-neutral-500">
   335→   335→                  {formatIDR(order.total)} untuk pesanan {order.code} sudah lunas —
   336→   336→                  terverifikasi <strong className="font-bold text-emerald-600">otomatis oleh sistem</strong>{" "}
   337→   337→                  saat dana masuk. Geprekmu segera dimasak ya!
   338→   338→                </p>
   339→   339→                {isPos && onPrintReceipt ? (
   340→   340→                  <div className="mt-2 flex w-full items-center gap-2">
   341→   341→                    <button
   342→   342→                      onClick={onClose}
   343→   343→                      className="press-effect h-11 flex-1 rounded-xl border border-neutral-200 bg-white text-sm font-extrabold text-neutral-600 hover:bg-neutral-50"
   344→   344→                    >
   345→   345→                      Selesai
   346→   346→                    </button>
   347→   347→                    <button
   348→   348→                      onClick={() => void onPrintReceipt()}
   349→   349→                      className="press-effect flex h-11 flex-[1.6] items-center justify-center gap-2 rounded-xl bg-orange-600 text-sm font-extrabold text-white shadow-lg shadow-orange-500/30 hover:bg-orange-700"
   350→   350→                    >
   351→   351→                      <Printer className="h-4 w-4" /> Cetak Struk
   352→   352→                    </button>
   353→   353→                  </div>
   354→   354→                ) : (
   355→   355→                  <button
   356→   356→                    onClick={onClose}
   357→   357→                    className="press-effect mt-2 w-full rounded-xl bg-orange-600 py-3 text-sm font-extrabold text-white shadow-lg shadow-orange-500/30 hover:bg-orange-700"
   358→   358→                  >
   359→   359→                    Selesai
   360→   360→                  </button>
   361→   361→                )}
   362→   362→              </div>
   363→   363→            ) : phase === "cancelled" ? (
   364→   364→              /* --------- DIBATALKAN OTOMATIS (belum bayar 30 menit) --------- */
   365→   365→              <div className="flex flex-col items-center gap-3 py-8 text-center">
   366→   366→                <motion.span
   367→   367→                  initial={prefersReduced ? false : { scale: 0.5, opacity: 0 }}
   368→   368→                  animate={{ scale: 1, opacity: 1 }}
   369→   369→                  transition={{ type: "spring", stiffness: 300, damping: 18 }}
   370→   370→                  className="flex h-20 w-20 items-center justify-center rounded-full bg-red-50 ring-4 ring-red-100"
   371→   371→                >
   372→   372→                  <XCircle className="h-11 w-11 text-red-500" />
   373→   373→                </motion.span>
   374→   374→                <h2 className="text-lg font-black text-neutral-900">Pesanan Dibatalkan Otomatis</h2>
   375→   375→                <p className="max-w-[320px] text-xs leading-relaxed text-neutral-500">
   376→   376→                  Pembayaran QRIS untuk pesanan <strong className="font-bold text-neutral-700">{order.code}</strong>{" "}
   377→   377→                  tidak diterima dalam <strong className="font-bold text-red-500">30 menit</strong>, jadi
   378→   378→                  pesanan dibatalkan. Voucher yang terpakai sudah dikembalikan ke akunmu.
   379→   379→                </p>
   380→   380→                {monitorNote && (
   381→   381→                  <p className="max-w-[320px] rounded-xl bg-amber-50 p-2.5 text-[11px] font-semibold leading-snug text-amber-700 ring-1 ring-amber-200">
   382→   382→                    {monitorNote}
   383→   383→                  </p>
   384→   384→                )}
   385→   385→                <p className="text-[11px] text-neutral-400">
   386→   386→                  Masih pengin? Buat pesanan baru — dapur selalu siap menggeprek 😊
   387→   387→                </p>
   388→   388→                <button
   389→   389→                  onClick={onClose}
   390→   390→                  className="press-effect mt-2 w-full rounded-xl bg-orange-600 py-3 text-sm font-extrabold text-white shadow-lg shadow-orange-500/30 hover:bg-orange-700"
   391→   391→                >
   392→   392→                  Tutup
   393→   393→                </button>
   394→   394→              </div>
   395→   395→            ) : (
   396→   396→              /* --------- PINDAI QR --------- */
   397→   397→              <>
   398→   398→                <div className="flex items-end justify-between gap-3">
   399→   399→                  <div>
   400→   400→                    <p className="text-[11px] font-bold uppercase tracking-wide text-neutral-400">
   401→   401→                      Total Bayar
   402→   402→                    </p>
   403→   403→                    <p className="text-2xl font-black text-orange-600">{formatIDR(order.total)}</p>
   404→   404→                  </div>
   405→   405→                  <span
   406→   406→                    className={cn(
   407→   407→                      "flex items-center gap-1.5 rounded-full px-2.5 py-1 text-[11px] font-black tabular-nums ring-1",
   408→   408→                      codeStale
   409→   409→                        ? "bg-red-50 text-red-600 ring-red-200"
   410→   410→                        : secondsLeft < 60
   411→   411→                          ? "bg-amber-50 text-amber-700 ring-amber-200"
   412→   412→                          : "bg-orange-50 text-orange-700 ring-orange-200"
   413→   413→                    )}
   414→   414→                  >
   415→   415→                    <Timer className="h-3.5 w-3.5" />
   416→   416→                    {codeStale && autoVerify ? "Kode Hangus" : expired ? "Lewat waktu" : mmss(secondsLeft)}
   417→   417→                  </span>
   418→   418→                </div>
   419→   419→
   420→   420→                {/* Kartu QR */}
   421→   421→                <div className="mt-3.5 flex flex-col items-center rounded-2xl border-2 border-orange-200 bg-white p-4 shadow-inner">
   422→   422→                  {loadError ? (
   423→   423→                    <div className="flex min-h-[216px] w-[216px] flex-col items-center justify-center gap-2 text-center">
   424→   424→                      <p className="text-xs font-semibold leading-relaxed text-red-500">{loadError}</p>
   425→   425→                      <button
   426→   426→                        onClick={() => void fetchPayload()}
   427→   427→                        className="press-effect rounded-full bg-orange-600 px-4 py-2 text-[11px] font-extrabold text-white hover:bg-orange-700"
   428→   428→                      >
   429→   429→                        Coba Lagi
   430→   430→                      </button>
   431→   431→                    </div>
   432→   432→                  ) : info?.imageUrl ? (
   433→   433→                    <img
   434→   434→                      src={info.imageUrl}
   435→   435→                      alt={`Kode QRIS pesanan ${order.code}`}
   436→   436→                      className="h-[208px] w-[208px] object-contain"
   437→   437→                    />
   438→   438→                  ) : info?.payload ? (
   439→   439→                    <motion.div
   440→   440→                      initial={prefersReduced ? false : { opacity: 0, scale: 0.94 }}
   441→   441→                      animate={{ opacity: 1, scale: 1 }}
   442→   442→                      className="flex flex-col items-center"
   443→   443→                    >
   444→   444→                      <QRCodeSVG
   445→   445→                        value={info.payload}
   446→   446→                        size={208}
   447→   447→                        level="M"
   448→   448→                        marginSize={1}
   449→   449→                        bgColor="#FFFFFF"
   450→   450→                        fgColor="#1C1917"
