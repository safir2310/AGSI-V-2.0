     1→"use client";
     2→
     3→import { useCallback, useEffect, useRef, useState } from "react";
     4→import { AnimatePresence, motion, useReducedMotion } from "framer-motion";
     5→import { QRCodeSVG } from "qrcode.react";
     6→import {
     7→  BadgeCheck,
     8→  FlaskConical,
     9→  Loader2,
    10→  Printer,
    11→  RefreshCw,
    12→  ScanLine,
    13→  ShieldCheck,
    14→  Timer,
    15→  X,
    16→  XCircle,
    17→} from "lucide-react";
    18→import { apiFetch } from "@/lib/api";
    19→import { formatIDR } from "@/lib/types";
    20→import { cn } from "@/lib/utils";
    21→
    22→/** Pesanan yang akan dibayar via QRIS */
    23→export type QrisPayOrder = { id: string; code: string; total: number };
    24→
    25→type Phase = "scan" | "paid" | "cancelled";
    26→
    27→type QrisInfo = {
    28→  payload: string | null;
    29→  imageUrl: string | null;
    30→  mode: "GATEWAY" | "LOCAL";
    31→  autoVerify: boolean;
    32→  /** gateway QRIS bawaan (sandbox) aktif — tombol simulasi dana ditampilkan */
    33→  sandbox: boolean;
    34→};
    35→
    36→/** Interval polling verifikasi otomatis */
    37→const POLL_MS = 4000;
    38→/** Batas waktu pembayaran — 30 menit sejak pesanan dibuat.
    39→ *  Lewat dari itu pesanan batal otomatis (server membatalkan saat sweep/poll). */
    40→const DURATION_SEC = 30 * 60; // 30 menit
    41→
    42→function mmss(sec: number): string {
    43→  const m = Math.floor(sec / 60);
    44→  const s = sec % 60;
    45→  return `${String(m).padStart(2, "0")}:${String(s).padStart(2, "0")}`;
    46→}
    47→
    48→/**
    49→ * Layar pembayaran QRIS — pelanggan maupun kasir memakai layar yang sama.
    50→ *
    51→ * VERIFIKASI OTOMATIS: setelah QR tampil, layar ini memantau status ke server
    52→ * tiap beberapa detik; server bertanya ke penyedia QRIS "sudah masuk belum
    53→ * dananya?" dan melunaskan pesanan begitu dana masuk. TIDAK ADA tombol
    54→ * "Saya Sudah Bayar" / "Uang Sudah Masuk" — kasir memang tidak boleh
    55→ * (dan tidak bisa) melunaskan pesanan QRIS secara manual.
    56→ */
    57→export function QrisPaySheet({
    58→  order,
    59→  variant,
    60→  onClose,
    61→  onPaid,
    62→  onPrintReceipt,
    63→}: {
    64→  order: QrisPayOrder;
    65→  variant: "customer" | "pos";
    66→  onClose: () => void;
    67→  /** dipanggil saat status pembayaran berubah menjadi lunas (mis. refresh daftar) */
    68→  onPaid?: () => void | Promise<void>;
    69→  /** POS: cetak struk setelah lunas (bila disediakan halaman pemanggil) */
    70→  onPrintReceipt?: () => void | Promise<void>;
    71→}) {
    72→  const prefersReduced = useReducedMotion();
    73→  const isPos = variant === "pos";
    74→
    75→  const [info, setInfo] = useState<QrisInfo | null>(null);
    76→  const [loadError, setLoadError] = useState<string | null>(null);
    77→  const [phase, setPhase] = useState<Phase>("scan");
    78→  const [secondsLeft, setSecondsLeft] = useState(DURATION_SEC);
    79→  const [checkingNow, setCheckingNow] = useState(false);
    80→  const [regenerating, setRegenerating] = useState(false);
    81→  /** true saat simulasi dana masuk (sandbox) sedang diproses */
    82→  const [simulating, setSimulating] = useState(false);
    83→  /** pesan error simulasi (ditampilkan di bawah tombolnya) */
    84→  const [simulateError, setSimulateError] = useState<string | null>(null);
    85→  /** status terakhir dari penyedia (mis. "EXPIRED") */
    86→  const [gatewayStatus, setGatewayStatus] = useState<string | null>(null);
    87→  /** gangguan sementara saat memantau (diciutkan agar tidak bikin cemas) */
    88→  const [monitorNote, setMonitorNote] = useState<string | null>(null);
    89→  /** batas waktu pesanan (createdAt + 30 menit) dalam ms epoch */
    90→  const [orderDeadline, setOrderDeadline] = useState<number | null>(null);
    91→  const paidNotified = useRef(false);
    92→
    93→  const notifyPaid = useCallback(() => {
    94→    if (paidNotified.current) return;
    95→    paidNotified.current = true;
    96→    void onPaid?.();
    97→  }, [onPaid]);
    98→
    99→  /** Ambil kode QRIS dari server (gateway bila aktif, fallback lokal) */
   100→  const fetchPayload = useCallback(async () => {
   101→    setLoadError(null);
   102→    setInfo(null);
   103→    setGatewayStatus(null);
   104→    try {
   105→      const res = await apiFetch(`/api/orders/${order.id}/qris`, { cache: "no-store" });
   106→      const data = await res.json();
   107→      if (!res.ok) {
   108→        // Pesanan ternyata sudah lunas → langsung tampilkan sukses
   109→        if (typeof data.error === "string" && data.error.toLowerCase().includes("lunas")) {
   110→          setPhase("paid");
   111→          notifyPaid();
   112→          return;
   113→        }
   114→        // Pesanan dibatalkan otomatis (lewat batas 30 menit) → tampilan batal
   115→        if (data.cancelled === true) {
   116→          setPhase("cancelled");
   117→          return;
   118→        }
   119→        setLoadError(data.error ?? "Gagal memuat kode QRIS.");
   120→        return;
   121→      }
   122→      if (data.paid === true) {
   123→        setPhase("paid");
   124→        notifyPaid();
   125→        return;
   126→      }
   127→      const expiresAt = data.expiresAt ? new Date(data.expiresAt as string) : null;
   128→      const cancelMs = data.cancelAt ? new Date(data.cancelAt as string).getTime() : null;
   129→      if (cancelMs) setOrderDeadline(cancelMs);
   130→      setInfo({
   131→        payload: (data.payload as string) ?? null,
   132→        imageUrl: (data.imageUrl as string) ?? null,
   133→        mode: data.mode === "GATEWAY" ? "GATEWAY" : "LOCAL",
   134→        autoVerify: !!data.autoVerify,
   135→        sandbox: data.sandbox === true,
   136→      });
   137→      // Hitung mundur pakai batas TERKETAT: umur kode gateway vs batas pesanan
   138→      const nowMs = Date.now();
   139→      const gatewaySec = expiresAt
   140→        ? Math.max(0, Math.floor((expiresAt.getTime() - nowMs) / 1000))
   141→        : DURATION_SEC;
   142→      const orderSec = cancelMs
   143→        ? Math.max(0, Math.floor((cancelMs - nowMs) / 1000))
   144→        : DURATION_SEC;
   145→      setSecondsLeft(Math.max(0, Math.min(DURATION_SEC, gatewaySec, orderSec)));
   146→    } catch {
   147→      setLoadError("Tidak dapat menghubungi server. Periksa koneksi lalu coba lagi.");
   148→    }
   149→  }, [order.id, notifyPaid]);
   150→
   151→  useEffect(() => {
   152→    void fetchPayload();
   153→  }, [fetchPayload]);
   154→
   155→  // Hitung waktu mundur (hanya saat fase scan)
   156→  useEffect(() => {
   157→    if (phase !== "scan") return;
   158→    const t = setInterval(() => {
   159→      setSecondsLeft((s) => (s > 0 ? s - 1 : 0));
   160→    }, 1000);
   161→    return () => clearInterval(t);
   162→  }, [phase]);
   163→
   164→  // Tutup dengan tombol Escape
   165→  useEffect(() => {
   166→    const onKey = (e: KeyboardEvent) => {
   167→      if (e.key === "Escape") onClose();
   168→    };
   169→    window.addEventListener("keydown", onKey);
   170→    return () => window.removeEventListener("keydown", onKey);
   171→  }, [onClose]);
   172→
   173→  // ===== Pemantauan verifikasi otomatis =====
   174→  useEffect(() => {
   175→    if (phase !== "scan" || !info) return;
   176→    let alive = true;
   177→    const poll = async (force = false) => {
   178→      try {
   179→        const res = await apiFetch(
   180→          `/api/orders/${order.id}/qris/status${force ? "?force=1" : ""}`,
   181→          { cache: "no-store" }
   182→        );
   183→        if (!res.ok || !alive) return;
   184→        const data = await res.json();
   185→        if (data.paymentStatus === "PAID") {
   186→          setPhase("paid");
   187→          notifyPaid();
   188→          return;
   189→        }
   190→        // Pesanan batal otomatis (lewat batas 30 menit / dana telat)
   191→        if (data.cancelled === true || data.status === "CANCELLED") {
   192→          setMonitorNote((data.notice as string) ?? null);
   193→          setPhase("cancelled");
   194→          return;
   195→        }
   196→        if (data.cancelAt) setOrderDeadline(new Date(data.cancelAt as string).getTime());
   197→        setGatewayStatus((data.gatewayStatus as string) ?? null);
   198→        setMonitorNote((data.gatewayError as string) ?? null);
   199→      } catch {
   200→        // koneksi tersendat — coba lagi pada tick berikutnya
   201→      }
   202→    };
   203→    void poll();
   204→    const t = setInterval(() => void poll(), POLL_MS);
   205→    return () => {
   206→      alive = false;
   207→      clearInterval(t);
   208→    };
   209→  }, [phase, info, order.id, notifyPaid]);
   210→
   211→  /** POS: paksa sistem memeriksa penyedia sekarang (tetap otomatis, bukan menandai lunas) */
   212→  const checkNow = async () => {
   213→    if (checkingNow) return;
   214→    setCheckingNow(true);
   215→    try {
   216→      const res = await apiFetch(`/api/orders/${order.id}/qris/status?force=1`, {
   217→        cache: "no-store",
   218→      });
   219→      const data = await res.json();
   220→      if (res.ok && data.paymentStatus === "PAID") {
   221→        setPhase("paid");
   222→        notifyPaid();
   223→      }
   224→    } catch {
   225→      /* abaikan — polling reguler akan menyusul */
   226→    } finally {
   227→      setCheckingNow(false);
   228→    }
   229→  };
   230→
   231→  /** Buat ulang kode bayar (transaksi lama kedaluwarsa) */
   232→  const regenerate = async () => {
   233→    if (regenerating) return;
   234→    setRegenerating(true);
   235→    await fetchPayload();
   236→    setRegenerating(false);
   237→  };
   238→
   239→  /** SANDBOX: tiru notifikasi "dana masuk" dari dompet digital ke gateway
   240→   *  bawaan. Pesanan tetap dilunaskan OLEH SISTEM di server (jalur yang sama
   241→   *  dengan verifikasi otomatis) — ini bukan tombol "tandai lunas" manual. */
   242→  const simulatePay = async () => {
   243→    if (simulating) return;
   244→    setSimulating(true);
   245→    setSimulateError(null);
   246→    try {
   247→      const res = await apiFetch(`/api/orders/${order.id}/qris/simulate-pay`, {
   248→        method: "POST",
   249→      });
   250→      const data = await res.json();
   251→      if (res.ok && data.paymentStatus === "PAID") {
   252→        setPhase("paid");
   253→        notifyPaid();
   254→        return;
   255→      }
   256→      if (data.cancelled === true) {
   257→        setMonitorNote(
   258→          typeof data.error === "string" ? data.error : null
   259→        );
   260→        setPhase("cancelled");
   261→        return;
   262→      }
   263→      setSimulateError(
   264→        typeof data.error === "string" ? data.error : "Simulasi gagal — coba lagi."
   265→      );
   266→    } catch {
   267→      setSimulateError("Tidak dapat menghubungi server. Periksa koneksi lalu coba lagi.");
   268→    } finally {
   269→      setSimulating(false);
   270→    }
   271→  };
   272→
   273→  const expired = secondsLeft <= 0;
   274→  const codeStale = expired || gatewayStatus === "EXPIRED";
   275→  const autoVerify = info?.autoVerify ?? false;
   276→  /** Batas pesanan (30 menit) sudah terlewat — pesanan mati, jangan tawarkan buat ulang kode */
   277→  const pastDeadline = orderDeadline !== null && Date.now() >= orderDeadline;
   278→
   279→  return (
   280→    <AnimatePresence>
   281→      <motion.div
   282→        key="qris-overlay"
   283→        initial={prefersReduced ? false : { opacity: 0 }}
   284→        animate={{ opacity: 1 }}
   285→        exit={prefersReduced ? undefined : { opacity: 0 }}
   286→        className="fixed inset-0 z-[95] flex items-end justify-center bg-neutral-900/70 backdrop-blur-sm sm:items-center sm:p-4"
   287→        role="dialog"
   288→        aria-modal="true"
   289→        aria-label="Pembayaran QRIS"
   290→      >
   291→        <motion.div
   292→          initial={prefersReduced ? false : { y: 48, scale: 0.97 }}
   293→          animate={{ y: 0, scale: 1 }}
   294→          exit={prefersReduced ? undefined : { y: 48, opacity: 0 }}
   295→          transition={{ type: "spring", stiffness: 380, damping: 34 }}
   296→          className="flex max-h-[94dvh] w-full max-w-md flex-col overflow-hidden rounded-t-3xl bg-white shadow-2xl sm:rounded-3xl"
   297→        >
   298→          {/* ===== KEPALA ===== */}
   299→          <header className="relative shrink-0 bg-gradient-to-r from-orange-700 via-orange-600 to-orange-500 px-5 py-4 text-white">
   300→            <div className="flex items-center gap-3">
   301→              <span className="flex h-10 w-10 items-center justify-center rounded-2xl bg-white/20 ring-1 ring-white/30">
   302→                <ScanLine className="h-5 w-5" />
   303→              </span>
   304→              <div className="min-w-0 flex-1">
   305→                <p className="text-sm font-black tracking-wide">Pembayaran QRIS</p>
   306→                <p className="truncate text-[11px] font-semibold text-orange-100">
   307→                  {order.code} • {isPos ? "Kasir" : "Aplikasi"}
   308→                </p>
   309→              </div>
   310→              <button
   311→                onClick={onClose}
   312→                aria-label="Tutup"
   313→                className="press-effect flex h-9 w-9 items-center justify-center rounded-full bg-white/15 hover:bg-white/30"
   314→              >
   315→                <X className="h-4.5 w-4.5" />
   316→              </button>
   317→            </div>
   318→          </header>
   319→
   320→          {/* ===== ISI ===== */}
   321→          <div className="scrollbar-slim overflow-y-auto px-5 pb-[max(1.25rem,env(safe-area-inset-bottom))] pt-4">
   322→            {phase === "paid" ? (
   323→              /* --------- BERHASIL (verifikasi otomatis) --------- */
   324→              <div className="flex flex-col items-center gap-3 py-8 text-center">
   325→                <motion.span
   326→                  initial={prefersReduced ? false : { scale: 0.5, opacity: 0 }}
   327→                  animate={{ scale: 1, opacity: 1 }}
   328→                  transition={{ type: "spring", stiffness: 300, damping: 18 }}
   329→                  className="flex h-20 w-20 items-center justify-center rounded-full bg-emerald-50 ring-4 ring-emerald-100"
   330→                >
   331→                  <BadgeCheck className="h-11 w-11 text-emerald-600" />
   332→                </motion.span>
   333→                <h2 className="text-lg font-black text-neutral-900">Pembayaran Diterima! 🎉</h2>
   334→                <p className="max-w-[300px] text-xs leading-relaxed text-neutral-500">
   335→                  {formatIDR(order.total)} untuk pesanan {order.code} sudah lunas —
   336→                  terverifikasi <strong className="font-bold text-emerald-600">otomatis oleh sistem</strong>{" "}
   337→                  saat dana masuk. Geprekmu segera dimasak ya!
   338→                </p>
   339→                {isPos && onPrintReceipt ? (
   340→                  <div className="mt-2 flex w-full items-center gap-2">
   341→                    <button
   342→                      onClick={onClose}
   343→                      className="press-effect h-11 flex-1 rounded-xl border border-neutral-200 bg-white text-sm font-extrabold text-neutral-600 hover:bg-neutral-50"
   344→                    >
   345→                      Selesai
   346→                    </button>
   347→                    <button
   348→                      onClick={() => void onPrintReceipt()}
   349→                      className="press-effect flex h-11 flex-[1.6] items-center justify-center gap-2 rounded-xl bg-orange-600 text-sm font-extrabold text-white shadow-lg shadow-orange-500/30 hover:bg-orange-700"
   350→                    >
   351→                      <Printer className="h-4 w-4" /> Cetak Struk
   352→                    </button>
   353→                  </div>
   354→                ) : (
   355→                  <button
   356→                    onClick={onClose}
   357→                    className="press-effect mt-2 w-full rounded-xl bg-orange-600 py-3 text-sm font-extrabold text-white shadow-lg shadow-orange-500/30 hover:bg-orange-700"
   358→                  >
   359→                    Selesai
   360→                  </button>
   361→                )}
   362→              </div>
   363→            ) : phase === "cancelled" ? (
   364→              /* --------- DIBATALKAN OTOMATIS (belum bayar 30 menit) --------- */
   365→              <div className="flex flex-col items-center gap-3 py-8 text-center">
   366→                <motion.span
   367→                  initial={prefersReduced ? false : { scale: 0.5, opacity: 0 }}
   368→                  animate={{ scale: 1, opacity: 1 }}
   369→                  transition={{ type: "spring", stiffness: 300, damping: 18 }}
   370→                  className="flex h-20 w-20 items-center justify-center rounded-full bg-red-50 ring-4 ring-red-100"
   371→                >
   372→                  <XCircle className="h-11 w-11 text-red-500" />
   373→                </motion.span>
   374→                <h2 className="text-lg font-black text-neutral-900">Pesanan Dibatalkan Otomatis</h2>
   375→                <p className="max-w-[320px] text-xs leading-relaxed text-neutral-500">
   376→                  Pembayaran QRIS untuk pesanan <strong className="font-bold text-neutral-700">{order.code}</strong>{" "}
   377→                  tidak diterima dalam <strong className="font-bold text-red-500">30 menit</strong>, jadi
   378→                  pesanan dibatalkan. Voucher yang terpakai sudah dikembalikan ke akunmu.
   379→                </p>
   380→                {monitorNote && (
   381→                  <p className="max-w-[320px] rounded-xl bg-amber-50 p-2.5 text-[11px] font-semibold leading-snug text-amber-700 ring-1 ring-amber-200">
   382→                    {monitorNote}
   383→                  </p>
   384→                )}
   385→                <p className="text-[11px] text-neutral-400">
   386→                  Masih pengin? Buat pesanan baru — dapur selalu siap menggeprek 😊
   387→                </p>
   388→                <button
   389→                  onClick={onClose}
   390→                  className="press-effect mt-2 w-full rounded-xl bg-orange-600 py-3 text-sm font-extrabold text-white shadow-lg shadow-orange-500/30 hover:bg-orange-700"
   391→                >
   392→                  Tutup
   393→                </button>
   394→              </div>
   395→            ) : (
   396→              /* --------- PINDAI QR --------- */
   397→              <>
   398→                <div className="flex items-end justify-between gap-3">
   399→                  <div>
   400→                    <p className="text-[11px] font-bold uppercase tracking-wide text-neutral-400">
   401→                      Total Bayar
   402→                    </p>
   403→                    <p className="text-2xl font-black text-orange-600">{formatIDR(order.total)}</p>
   404→                  </div>
   405→                  <span
   406→                    className={cn(
   407→                      "flex items-center gap-1.5 rounded-full px-2.5 py-1 text-[11px] font-black tabular-nums ring-1",
   408→                      codeStale
   409→                        ? "bg-red-50 text-red-600 ring-red-200"
   410→                        : secondsLeft < 60
   411→                          ? "bg-amber-50 text-amber-700 ring-amber-200"
   412→                          : "bg-orange-50 text-orange-700 ring-orange-200"
   413→                    )}
   414→                  >
   415→                    <Timer className="h-3.5 w-3.5" />
   416→                    {codeStale && autoVerify ? "Kode Hangus" : expired ? "Lewat waktu" : mmss(secondsLeft)}
   417→                  </span>
   418→                </div>
   419→
   420→                {/* Kartu QR */}
   421→                <div className="mt-3.5 flex flex-col items-center rounded-2xl border-2 border-orange-200 bg-white p-4 shadow-inner">
   422→                  {loadError ? (
   423→                    <div className="flex min-h-[216px] w-[216px] flex-col items-center justify-center gap-2 text-center">
   424→                      <p className="text-xs font-semibold leading-relaxed text-red-500">{loadError}</p>
   425→                      <button
   426→                        onClick={() => void fetchPayload()}
   427→                        className="press-effect rounded-full bg-orange-600 px-4 py-2 text-[11px] font-extrabold text-white hover:bg-orange-700"
   428→                      >
   429→                        Coba Lagi
   430→                      </button>
   431→                    </div>
   432→                  ) : info?.imageUrl ? (
   433→                    <img
   434→                      src={info.imageUrl}
   435→                      alt={`Kode QRIS pesanan ${order.code}`}
   436→                      className="h-[208px] w-[208px] object-contain"
   437→                    />
   438→                  ) : info?.payload ? (
   439→                    <motion.div
   440→                      initial={prefersReduced ? false : { opacity: 0, scale: 0.94 }}
   441→                      animate={{ opacity: 1, scale: 1 }}
   442→                      className="flex flex-col items-center"
   443→                    >
   444→                      <QRCodeSVG
   445→                        value={info.payload}
   446→                        size={208}
   447→                        level="M"
   448→                        marginSize={1}
   449→                        bgColor="#FFFFFF"
   450→                        fgColor="#1C1917"
   451→                        title={`QRIS ${order.code}`}
   452→                      />
   453→                      <span className="mt-2.5 flex items-center gap-1.5 rounded-full bg-neutral-900 px-3 py-1 text-[10px] font-black tracking-[0.2em] text-white">
   454→                        QRIS
   455→                      </span>
   456→                    </motion.div>
   457→                  ) : (
   458→                    <div className="flex min-h-[216px] w-[216px] items-center justify-center">
   459→                      <Loader2 className="h-8 w-8 animate-spin text-orange-400" />
   460→                    </div>
   461→                  )}
   462→                </div>
   463→
   464→                <p className="mt-3 text-center text-[11px] font-bold uppercase tracking-wide text-neutral-400">
   465→                  AYAM GEPREK SAMBAL IJO
   466→                </p>
   467→                <p className="mt-1 text-center text-[11px] leading-relaxed text-neutral-500">
   468→                  Buka aplikasi{" "}
   469→                  <strong className="font-bold text-neutral-700">
   470→                    GoPay, OVO, DANA, ShopeePay, atau m-Banking
   471→                  </strong>{" "}
   472→                  apa pun, lalu pindai kode di atas. Nominal sudah terisi otomatis.
   473→                </p>
   474→                <p className="mt-2 rounded-xl bg-orange-50 px-3 py-2 text-center text-[10px] font-bold leading-snug text-orange-700 ring-1 ring-orange-100">
   475→                  ⏳ Selesaikan sebelum waktu habis — pesanan yang belum dibayar
   476→                  30 menit sejak dibuat batal otomatis & voucher kembali.
   477→                </p>
   478→
   479→                {/* ===== KARTU STATUS VERIFIKASI OTOMATIS ===== */}
   480→                {autoVerify && (
   481→                  <div className="mt-3.5 flex items-start gap-2.5 rounded-xl bg-emerald-50 p-3 ring-1 ring-emerald-200">
