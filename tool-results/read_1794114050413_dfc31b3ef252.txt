     1→"use client";
     2→
     3→import { useCallback, useEffect, useRef, useState } from "react";
     4→import { createPortal } from "react-dom";
     5→import { AnimatePresence, motion, useReducedMotion } from "framer-motion";
     6→import { QRCodeSVG } from "qrcode.react";
     7→import {
     8→  BadgeCheck,
     9→  FlaskConical,
    10→  Loader2,
    11→  Printer,
    12→  RefreshCw,
    13→  ScanLine,
    14→  ShieldCheck,
    15→  Timer,
    16→  TriangleAlert,
    17→  X,
    18→  XCircle,
    19→} from "lucide-react";
    20→import { apiFetch } from "@/lib/api";
    21→import { toast } from "@/hooks/use-toast";
    22→import { formatIDR } from "@/lib/types";
    23→import { cn } from "@/lib/utils";
    24→
    25→/** Pesanan yang akan dibayar via QRIS */
    26→export type QrisPayOrder = { id: string; code: string; total: number };
    27→
    28→type Phase = "scan" | "paid" | "cancelled";
    29→
    30→type QrisInfo = {
    31→  payload: string | null;
    32→  imageUrl: string | null;
    33→  mode: "GATEWAY" | "LOCAL";
    34→  autoVerify: boolean;
    35→  /** gateway QRIS bawaan (sandbox) aktif — tombol simulasi dana ditampilkan */
    36→  sandbox: boolean;
    37→  /** penyedia verifikasi: "midtrans" | "midtrans-sandbox" | "internal" | null */
    38→  provider: string | null;
    39→  /** Alasan penyedia utama (Midtrans) ditolak sehingga QR berasal dari fallback uji lokal */
    40→  fallbackReason: string | null;
    41→  /** Galat gateway saat mode LOCAL — ditampilkan jujur (kanal belum aktif, dll.) */
    42→  gatewayError: string | null;
    43→  /** Sandbox Midtrans: tautan simulator uji pembayaran (QR berisi URL uji) */
    44→  sandboxSimulatorUrl: string | null;
    45→};
    46→
    47→/** Interval polling verifikasi otomatis */
    48→const POLL_MS = 4000;
    49→/** Batas waktu pembayaran QRIS — 5 MENIT sejak pesanan dibuat (permintaan
    50→ *  pemilik). Lewat dari itu pesanan batal otomatis (server membatalkan
    51→ *  saat sweep/poll) dan kode bayar hangus. */
    52→const DURATION_SEC = 5 * 60; // 5 menit
    53→
    54→function mmss(sec: number): string {
    55→  const m = Math.floor(sec / 60);
    56→  const s = sec % 60;
    57→  return `${String(m).padStart(2, "0")}:${String(s).padStart(2, "0")}`;
    58→}
    59→
    60→/**
    61→ * Layar pembayaran QRIS — pelanggan maupun kasir memakai layar yang sama.
    62→ *
    63→ * VERIFIKASI OTOMATIS: setelah QR tampil, layar ini memantau status ke server
    64→ * tiap beberapa detik; server bertanya ke penyedia QRIS "sudah masuk belum
    65→ * dananya?" dan melunaskan pesanan begitu dana masuk. TIDAK ADA tombol
    66→ * "Saya Sudah Bayar" / "Uang Sudah Masuk" — kasir memang tidak boleh
    67→ * (dan tidak bisa) melunaskan pesanan QRIS secara manual.
    68→ */
    69→export function QrisPaySheet({
    70→  order,
    71→  variant,
    72→  onClose,
    73→  onPaid,
    74→  onPrintReceipt,
    75→}: {
    76→  order: QrisPayOrder;
    77→  variant: "customer" | "pos";
    78→  onClose: () => void;
    79→  /** dipanggil saat status pembayaran berubah menjadi lunas (mis. refresh daftar) */
    80→  onPaid?: () => void | Promise<void>;
    81→  /** POS: cetak struk setelah lunas (bila disediakan halaman pemanggil) */
    82→  onPrintReceipt?: () => void | Promise<void>;
    83→}) {
    84→  const prefersReduced = useReducedMotion();
    85→  const isPos = variant === "pos";
    86→
    87→  const [info, setInfo] = useState<QrisInfo | null>(null);
    88→  const [loadError, setLoadError] = useState<string | null>(null);
    89→  const [phase, setPhase] = useState<Phase>("scan");
    90→  const [secondsLeft, setSecondsLeft] = useState(DURATION_SEC);
    91→  const [checkingNow, setCheckingNow] = useState(false);
    92→  const [regenerating, setRegenerating] = useState(false);
    93→  /** status terakhir dari penyedia (mis. "EXPIRED") */
    94→  const [gatewayStatus, setGatewayStatus] = useState<string | null>(null);
    95→  /** gangguan sementara saat memantau (diciutkan agar tidak bikin cemas) */
    96→  const [monitorNote, setMonitorNote] = useState<string | null>(null);
    97→  /** batas waktu pesanan (createdAt + 5 menit) dalam ms epoch */
    98→  const [orderDeadline, setOrderDeadline] = useState<number | null>(null);
    99→  const paidNotified = useRef(false);
   100→  /** onPaid disimpan di ref: identitas prop baru dari re-render parent
   101→   *  (mis. jam kasir yang berdetik tiap detik) TIDAK boleh memicu ulang
   102→   *  efek fetch/polling di bawah — itulah penyebab layar QR berkedip. */
   103→  const onPaidRef = useRef(onPaid);
   104→  useEffect(() => {
   105→    onPaidRef.current = onPaid;
   106→  }, [onPaid]);
   107→
   108→  /** Stabil antar-render — aman dipakai sebagai dependensi efek */
   109→  const notifyPaid = useCallback(() => {
   110→    if (paidNotified.current) return;
   111→    paidNotified.current = true;
   112→    void onPaidRef.current?.();
   113→  }, []);
   114→
   115→  /** Penomoran permintaan: respons yang datang terlambat dari fetch lama
   116→   *  tidak boleh menimpa data fetch terbaru (cegah QR "maju-mundur"). */
   117→  const fetchSeq = useRef(0);
   118→  /** Ambil kode QRIS dari server (gateway bila aktif, fallback lokal).
   119→   *  soft=true → QR lama TETAP tampil selagi data baru diambil (kode
   120→   *  langsung tertukar saat respons tiba) sehingga layar tak berkedip. */
   121→  const fetchPayload = useCallback(async (soft = false) => {
   122→    const seq = ++fetchSeq.current;
   123→    setLoadError(null);
   124→    setGatewayStatus(null);
   125→    if (!soft) setInfo(null);
   126→    try {
   127→      const res = await apiFetch(`/api/orders/${order.id}/qris`, { cache: "no-store" });
   128→      const data = await res.json();
   129→      if (seq !== fetchSeq.current) return; // respons kedaluwarsa — abaikan
   130→      if (!res.ok) {
   131→        // Pesanan ternyata sudah lunas → langsung tampilkan sukses
   132→        if (typeof data.error === "string" && data.error.toLowerCase().includes("lunas")) {
   133→          setPhase("paid");
   134→          notifyPaid();
   135→          return;
   136→        }
   137→        // Pesanan dibatalkan otomatis (lewat batas 5 menit) → tampilan batal
   138→        if (data.cancelled === true) {
   139→          setPhase("cancelled");
   140→          return;
   141→        }
   142→        setLoadError(data.error ?? "Gagal memuat kode QRIS.");
   143→        return;
   144→      }
   145→      if (data.paid === true) {
   146→        setPhase("paid");
   147→        notifyPaid();
   148→        return;
   149→      }
   150→      const expiresAt = data.expiresAt ? new Date(data.expiresAt as string) : null;
   151→      const cancelMs = data.cancelAt ? new Date(data.cancelAt as string).getTime() : null;
   152→      if (cancelMs) setOrderDeadline(cancelMs);
   153→      setInfo({
   154→        payload: (data.payload as string) ?? null,
   155→        imageUrl: (data.imageUrl as string) ?? null,
   156→        mode: data.mode === "GATEWAY" ? "GATEWAY" : "LOCAL",
   157→        autoVerify: !!data.autoVerify,
   158→        sandbox: data.sandbox === true,
   159→        provider: typeof data.provider === "string" ? data.provider : null,
   160→        fallbackReason:
   161→          typeof data.fallbackReason === "string" ? data.fallbackReason : null,
   162→        gatewayError:
   163→          typeof data.gatewayError === "string" ? data.gatewayError : null,
   164→        sandboxSimulatorUrl:
   165→          typeof data.sandboxSimulatorUrl === "string" ? data.sandboxSimulatorUrl : null,
   166→      });
   167→      // Hitung mundur pakai batas TERKETAT: umur kode gateway vs batas pesanan
   168→      const nowMs = Date.now();
   169→      const gatewaySec = expiresAt
   170→        ? Math.max(0, Math.floor((expiresAt.getTime() - nowMs) / 1000))
   171→        : DURATION_SEC;
   172→      const orderSec = cancelMs
   173→        ? Math.max(0, Math.floor((cancelMs - nowMs) / 1000))
   174→        : DURATION_SEC;
   175→      setSecondsLeft(Math.max(0, Math.min(DURATION_SEC, gatewaySec, orderSec)));
   176→    } catch {
   177→      if (seq !== fetchSeq.current) return; // respons kedaluwarsa — abaikan
   178→      setLoadError("Tidak dapat menghubungi server. Periksa koneksi lalu coba lagi.");
   179→    }
   180→    // notifyPaid stabil ([]) — dependensi ini tidak berubah antar-render
   181→    // parent, sehingga QR hanya diambil SEKALI per pesanan (bukan tiap detik).
   182→  }, [order.id, notifyPaid]);
   183→
   184→  useEffect(() => {
   185→    void fetchPayload();
   186→  }, [fetchPayload]);
   187→
   188→  // Hitung waktu mundur (hanya saat fase scan)
   189→  useEffect(() => {
   190→    if (phase !== "scan") return;
   191→    const t = setInterval(() => {
   192→      setSecondsLeft((s) => (s > 0 ? s - 1 : 0));
   193→    }, 1000);
   194→    return () => clearInterval(t);
   195→  }, [phase]);
   196→
   197→  /** Tutup layar pembayaran + notifikasi sesuai hasil pembayaran terakhir
   198→   *  (permintaan pemilik):
   199→   *  - belum lunas  -> "Pembayaran belum berhasil — silakan lanjutkan
   200→   *                    pembayaran" (merah, agar kasir/pelanggan tak bingung)
   201→   *  - sudah lunas  -> "Pembayaran berhasil — pesanan Anda sedang diproses"
   202→   *  - sudah batal  -> cukup tutup (layar batal sudah menjelaskan) */
   203→  const requestClose = useCallback(() => {
   204→    if (phase === "paid") {
   205→      toast({
   206→        title: "Pembayaran berhasil",
   207→        description: `Pesanan ${order.code} ${formatIDR(order.total)} sudah terkonfirmasi — pesanan Anda sedang diproses.`,
   208→        duration: 7000,
   209→      });
   210→    } else if (phase === "scan") {
   211→      toast({
   212→        variant: "destructive",
   213→        title: "Pembayaran belum berhasil",
   214→        description: "Silakan lanjutkan pembayaran QRIS sebelum waktu habis — pesanan belum terlunasi.",
   215→        duration: 7000,
   216→      });
   217→    }
   218→    onClose();
   219→  }, [phase, order.code, order.total, onClose]);
   220→
   221→  // Tutup dengan tombol Escape (lewat requestClose: notifikasi status ikut tampil)
   222→  useEffect(() => {
   223→    const onKey = (e: KeyboardEvent) => {
   224→      if (e.key === "Escape") requestClose();
   225→    };
   226→    window.addEventListener("keydown", onKey);
   227→    return () => window.removeEventListener("keydown", onKey);
   228→  }, [requestClose]);
   229→
   230→  // ===== Pemantauan verifikasi otomatis =====
   231→  // hasInfo (boolean) sebagai dependensi — bukan objek info — agar interval
   232→  // polling TIDAK dibongkar-pasang setiap data baru masuk.
   233→  const hasInfo = info !== null;
   234→  useEffect(() => {
   235→    if (phase !== "scan" || !hasInfo) return;
   236→    let alive = true;
   237→    const poll = async (force = false) => {
   238→      try {
   239→        const res = await apiFetch(
   240→          `/api/orders/${order.id}/qris/status${force ? "?force=1" : ""}`,
   241→          { cache: "no-store" }
   242→        );
   243→        if (!res.ok || !alive) return;
   244→        const data = await res.json();
   245→        if (data.paymentStatus === "PAID") {
   246→          setPhase("paid");
   247→          notifyPaid();
   248→          return;
   249→        }
   250→        // Pesanan batal otomatis (lewat batas 5 menit / dana telat)
   251→        if (data.cancelled === true || data.status === "CANCELLED") {
   252→          setMonitorNote((data.notice as string) ?? null);
   253→          setPhase("cancelled");
   254→          return;
   255→        }
   256→        if (data.cancelAt) setOrderDeadline(new Date(data.cancelAt as string).getTime());
   257→        setGatewayStatus((data.gatewayStatus as string) ?? null);
   258→        setMonitorNote((data.gatewayError as string) ?? null);
   259→      } catch {
   260→        // koneksi tersendat — coba lagi pada tick berikutnya
   261→      }
   262→    };
   263→    void poll();
   264→    const t = setInterval(() => void poll(), POLL_MS);
   265→    return () => {
   266→      alive = false;
   267→      clearInterval(t);
   268→    };
   269→  }, [phase, hasInfo, order.id, notifyPaid]);
   270→
   271→  /** POS: paksa sistem memeriksa penyedia sekarang (tetap otomatis, bukan menandai lunas) */
   272→  const checkNow = async () => {
   273→    if (checkingNow) return;
   274→    setCheckingNow(true);
   275→    try {
   276→      const res = await apiFetch(`/api/orders/${order.id}/qris/status?force=1`, {
   277→        cache: "no-store",
   278→      });
   279→      const data = await res.json();
   280→      if (res.ok && data.paymentStatus === "PAID") {
   281→        setPhase("paid");
   282→        notifyPaid();
   283→      }
   284→    } catch {
   285→      /* abaikan — polling reguler akan menyusul */
   286→    } finally {
   287→      setCheckingNow(false);
   288→    }
   289→  };
   290→
   291→  /** Buat ulang kode bayar (transaksi lama kedaluwarsa) — soft: QR lama
   292→   *  tetap tampil sampai kode baru siap, tanpa kedipan spinner. */
   293→  const regenerate = async () => {
   294→    if (regenerating) return;
   295→    setRegenerating(true);
   296→    await fetchPayload(true);
   297→    setRegenerating(false);
   298→  };
   299→
   300→  const expired = secondsLeft <= 0;
   301→  const codeStale = expired || gatewayStatus === "EXPIRED";
   302→  const autoVerify = info?.autoVerify ?? false;
   303→  /** Batas pesanan (5 menit) sudah terlewat — pesanan mati, jangan tawarkan buat ulang kode */
   304→  const pastDeadline = orderDeadline !== null && Date.now() >= orderDeadline;
   305→
   306→  // Portal ke <body> agar modal tidak terjebak stacking context `main` (z-10)
   307→  // — selalu tampil DI ATAS navigasi bawah liquid glass (z-50) & header (z-40).
   308→  return createPortal(
   309→    <AnimatePresence>
   310→      <motion.div
   311→        key="qris-overlay"
   312→        initial={prefersReduced ? false : { opacity: 0 }}
   313→        animate={{ opacity: 1 }}
   314→        exit={prefersReduced ? undefined : { opacity: 0 }}
   315→        className="fixed inset-0 z-[95] flex items-end justify-center bg-neutral-900/70 backdrop-blur-sm sm:items-center sm:p-4"
   316→        role="dialog"
   317→        aria-modal="true"
   318→        aria-label="Pembayaran QRIS"
   319→      >
   320→        <motion.div
   321→          initial={prefersReduced ? false : { y: 48, scale: 0.97 }}
   322→          animate={{ y: 0, scale: 1 }}
   323→          exit={prefersReduced ? undefined : { y: 48, opacity: 0 }}
   324→          transition={{ type: "spring", stiffness: 380, damping: 34 }}
   325→          className="flex max-h-[94dvh] w-full max-w-md flex-col overflow-hidden rounded-t-3xl bg-white shadow-2xl sm:rounded-3xl"
   326→        >
   327→          {/* ===== KEPALA ===== */}
   328→          <header className="relative shrink-0 bg-gradient-to-r from-orange-700 via-orange-600 to-orange-500 px-5 py-4 text-white">
   329→            <div className="flex items-center gap-3">
   330→              <span className="flex h-10 w-10 items-center justify-center rounded-2xl bg-white/20 ring-1 ring-white/30">
   331→                <ScanLine className="h-5 w-5" />
   332→              </span>
   333→              <div className="min-w-0 flex-1">
   334→                <p className="text-sm font-black tracking-wide">Pembayaran QRIS</p>
   335→                <p className="truncate text-[11px] font-semibold text-orange-100">
   336→                  {order.code} • {isPos ? "Kasir" : "Aplikasi"}
   337→                </p>
   338→              </div>
   339→              <button
   340→                onClick={requestClose}
   341→                aria-label="Tutup"
   342→                className="press-effect flex h-9 w-9 items-center justify-center rounded-full bg-white/15 hover:bg-white/30"
   343→              >
   344→                <X className="h-4.5 w-4.5" />
   345→              </button>
   346→            </div>
   347→          </header>
   348→
   349→          {/* ===== ISI ===== */}
   350→          <div className="scrollbar-slim overflow-y-auto px-5 pb-[max(1.25rem,env(safe-area-inset-bottom))] pt-4">
   351→            {phase === "paid" ? (
   352→              /* --------- BERHASIL (verifikasi otomatis) --------- */
   353→              <div className="flex flex-col items-center gap-3 py-8 text-center">
   354→                <motion.span
   355→                  initial={prefersReduced ? false : { scale: 0.5, opacity: 0 }}
   356→                  animate={{ scale: 1, opacity: 1 }}
   357→                  transition={{ type: "spring", stiffness: 300, damping: 18 }}
   358→                  className="flex h-20 w-20 items-center justify-center rounded-full bg-emerald-50 ring-4 ring-emerald-100"
   359→                >
   360→                  <BadgeCheck className="h-11 w-11 text-emerald-600" />
   361→                </motion.span>
   362→                <h2 className="text-lg font-black text-neutral-900">Pembayaran Diterima! 🎉</h2>
   363→                <p className="max-w-[300px] text-xs leading-relaxed text-neutral-500">
   364→                  {formatIDR(order.total)} untuk pesanan {order.code} sudah lunas —
   365→                  terverifikasi <strong className="font-bold text-emerald-600">otomatis oleh sistem</strong>{" "}
   366→                  saat dana masuk. Geprekmu segera dimasak ya!
   367→                </p>
   368→                {isPos && onPrintReceipt ? (
   369→                  <div className="mt-2 flex w-full items-center gap-2">
   370→                    <button
   371→                      onClick={requestClose}
   372→                      className="press-effect h-11 flex-1 rounded-xl border border-neutral-200 bg-white text-sm font-extrabold text-neutral-600 hover:bg-neutral-50"
   373→                    >
   374→                      Selesai
   375→                    </button>
   376→                    <button
   377→                      onClick={() => void onPrintReceipt()}
   378→                      className="press-effect flex h-11 flex-[1.6] items-center justify-center gap-2 rounded-xl bg-orange-600 text-sm font-extrabold text-white shadow-lg shadow-orange-500/30 hover:bg-orange-700"
   379→                    >
   380→                      <Printer className="h-4 w-4" /> Cetak Struk
   381→                    </button>
   382→                  </div>
   383→                ) : (
   384→                  <button
   385→                    onClick={requestClose}
   386→                    className="press-effect mt-2 w-full rounded-xl bg-orange-600 py-3 text-sm font-extrabold text-white shadow-lg shadow-orange-500/30 hover:bg-orange-700"
   387→                  >
   388→                    Selesai
   389→                  </button>
   390→                )}
   391→              </div>
   392→            ) : phase === "cancelled" ? (
   393→              /* --------- DIBATALKAN OTOMATIS (belum bayar 5 menit) --------- */
   394→              <div className="flex flex-col items-center gap-3 py-8 text-center">
   395→                <motion.span
   396→                  initial={prefersReduced ? false : { scale: 0.5, opacity: 0 }}
   397→                  animate={{ scale: 1, opacity: 1 }}
   398→                  transition={{ type: "spring", stiffness: 300, damping: 18 }}
   399→                  className="flex h-20 w-20 items-center justify-center rounded-full bg-red-50 ring-4 ring-red-100"
   400→                >
   401→                  <XCircle className="h-11 w-11 text-red-500" />
   402→                </motion.span>
   403→                <h2 className="text-lg font-black text-neutral-900">Pesanan Dibatalkan Otomatis</h2>
   404→                <p className="max-w-[320px] text-xs leading-relaxed text-neutral-500">
   405→                  Pembayaran QRIS untuk pesanan <strong className="font-bold text-neutral-700">{order.code}</strong>{" "}
   406→                  tidak diterima dalam <strong className="font-bold text-red-500">5 menit</strong>, jadi
   407→                  pesanan dibatalkan. Voucher yang terpakai sudah dikembalikan ke akunmu.
   408→                </p>
   409→                {monitorNote && (
   410→                  <p className="max-w-[320px] rounded-xl bg-amber-50 p-2.5 text-[11px] font-semibold leading-snug text-amber-700 ring-1 ring-amber-200">
   411→                    {monitorNote}
   412→                  </p>
   413→                )}
   414→                <p className="text-[11px] text-neutral-400">
   415→                  Masih pengin? Buat pesanan baru — dapur selalu siap menggeprek 😊
   416→                </p>
   417→                <button
   418→                  onClick={onClose}
   419→                  className="press-effect mt-2 w-full rounded-xl bg-orange-600 py-3 text-sm font-extrabold text-white shadow-lg shadow-orange-500/30 hover:bg-orange-700"
   420→                >
   421→                  Tutup
   422→                </button>
   423→              </div>
   424→            ) : (
   425→              /* --------- PINDAI QR --------- */
   426→              <>
   427→                <div className="flex items-end justify-between gap-3">
   428→                  <div>
   429→                    <p className="text-[11px] font-bold uppercase tracking-wide text-neutral-400">
   430→                      Total Bayar
   431→                    </p>
   432→                    <p className="text-2xl font-black text-orange-600">{formatIDR(order.total)}</p>
   433→                  </div>
   434→                  <span
   435→                    className={cn(
   436→                      "flex items-center gap-1.5 rounded-full px-2.5 py-1 text-[11px] font-black tabular-nums ring-1",
   437→                      codeStale
   438→                        ? "bg-red-50 text-red-600 ring-red-200"
   439→                        : secondsLeft < 60
   440→                          ? "bg-amber-50 text-amber-700 ring-amber-200"
   441→                          : "bg-orange-50 text-orange-700 ring-orange-200"
   442→                    )}
   443→                  >
   444→                    <Timer className="h-3.5 w-3.5" />
   445→                    {codeStale && autoVerify ? "Kode Hangus" : expired ? "Lewat waktu" : mmss(secondsLeft)}
   446→                  </span>
   447→                </div>
   448→
   449→                {/* Kartu QR */}
   450→                <div className="mt-3.5 flex flex-col items-center rounded-2xl border-2 border-orange-200 bg-white p-4 shadow-inner">
   451→                  {loadError ? (
   452→                    <div className="flex min-h-[216px] w-[216px] flex-col items-center justify-center gap-2 text-center">
   453→                      <p className="text-xs font-semibold leading-relaxed text-red-500">{loadError}</p>
   454→                      <button
   455→                        onClick={() => void fetchPayload()}
   456→                        className="press-effect rounded-full bg-orange-600 px-4 py-2 text-[11px] font-extrabold text-white hover:bg-orange-700"
   457→                      >
   458→                        Coba Lagi
   459→                      </button>

   493→                <p className="mt-3 text-center text-[11px] font-bold uppercase tracking-wide text-neutral-400">

   525→                      {info?.provider?.startsWith("midtrans") && (

   584→                    <div className="flex items-center gap-2">
   585→                      <span className="flex h-6 w-6 items-center justify-center rounded-lg bg-amber-100 ring-1 ring-amber-300">

   602→                    >
   603→                      <FlaskConical className="h-4 w-4" /> Buka Simulator Bayar
   604→                    </a>
   605→                  </div>
   606→                )}
   607→
   608→                {/* Kode kedaluwarsa di gateway (pesanan masih hidup) -> buat ulang */}
   609→                {codeStale && autoVerify && !pastDeadline && (

   611→                    onClick={() => void regenerate()}
   612→                    disabled={regenerating}

   621→                        <RefreshCw className="h-4 w-4" /> Buat Ulang Kode Bayar

   623→                    )}
   624→                  </button>
   625→                )}
   626→                {expired && !autoVerify && !pastDeadline && (
   627→                  <p className="mt-2 rounded-xl bg-amber-50 p-2.5 text-center text-[11px] font-semibold leading-snug text-amber-700 ring-1 ring-amber-200">

   634→                {isPos ? (

   638→                      disabled={checkingNow || !info}
   639→                      className="press-effect flex h-11 w-full items-center justify-center gap-2 rounded-xl border-2 border-orange-200 bg-white text-sm font-extrabold text-orange-700 hover:bg-orange-50 disabled:opacity-60"
   640→                    >
   641→                      {checkingNow ? (
   642→                        <>
   643→                          <Loader2 className="h-4 w-4 animate-spin" /> Memeriksa…
   644→                        </>
   645→                      ) : (
   646→                        <>
   647→                          <ShieldCheck className="h-4 w-4" /> Periksa Pembayaran Sekarang
   648→                        </>
   649→                      )}
   650→                    </button>
   651→                    <p className="text-center text-[10px] leading-snug text-neutral-400">
   652→                      Minta pelanggan memindai QR di layar ini. Kasir{" "}

   675→              </>
