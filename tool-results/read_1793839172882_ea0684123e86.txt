     1→"use client";
     2→
     3→import { useEffect, useMemo, useState } from "react";
     4→import { AnimatePresence, motion, useReducedMotion } from "framer-motion";
     5→import {
     6→  Bike,
     7→  BadgeCheck,
     8→  ClipboardList,
     9→  Clock,
    10→  Coins,
    11→  Loader2,
    12→  LogOut,
    13→  Minus,
    14→  MonitorSmartphone,
    15→  Plus,
    16→  Receipt,
    17→  RefreshCw,
    18→  ScanLine,
    19→  Search,
    20→  ShoppingBag,
    21→  Trash2,
    22→  Utensils,
    23→  Wifi,
    24→  X,
    25→} from "lucide-react";
    26→import { Input } from "@/components/ui/input";
    27→import { ChickenBadge } from "@/components/logo/ChickenLogo";
    28→import { ProductThumb } from "@/components/shared/ProductThumb";
    29→import { QrisPaySheet, type QrisPayOrder } from "@/components/shared/QrisPaySheet";
    30→import { refreshOrders, useProductsStore } from "@/lib/hooks";
    31→import { useNoticeStore, type AppView } from "@/lib/store";
    32→import { apiFetch } from "@/lib/api";
    33→import { formatIDR, type AppUser, type Product } from "@/lib/types";
    34→import { cn } from "@/lib/utils";
    35→
    36→type PosItem = { product: Product; qty: number };
    37→
    38→export function PosView({ user, go, onLogout }: { user: AppUser; go: (v: AppView) => void; onLogout: () => void }) {
    39→  const prefersReduced = useReducedMotion();
    40→  const showNotice = useNoticeStore((s) => s.show);
    41→  const { products, load } = useProductsStore();
    42→  const [query, setQuery] = useState("");
    43→  const [category, setCategory] = useState("Semua");
    44→  /** Mode bar pencarian di bawah header: scan barcode member ATAU cari nama produk */
    45→  const [scanMode, setScanMode] = useState<"member" | "product">("product");
    46→  const [items, setItems] = useState<PosItem[]>([]);
    47→  const [customerName, setCustomerName] = useState("");
    48→  const [tableNo, setTableNo] = useState("");
    49→  /** Kode member (hasil scan barcode kartu member / ketikan kasir) */
    50→  const [memberCode, setMemberCode] = useState("");
    51→  const [memberInfo, setMemberInfo] = useState<{ name: string; memberCode: string; points: number } | null>(null);
    52→  const [memberChecking, setMemberChecking] = useState(false);
    53→  const [orderType, setOrderType] = useState<"DINE_IN" | "TAKEAWAY" | "DELIVERY">("DINE_IN");
    54→  const [payment, setPayment] = useState<"CASH" | "QRIS">("CASH");
    55→  const [submitting, setSubmitting] = useState(false);
    56→  const [clock, setClock] = useState("");
    57→  const [lastOrder, setLastOrder] = useState<{
    58→    id: string;
    59→    code: string;
    60→    total: number;
    61→    name: string;
    62→    paymentMethod: string;
    63→    paymentStatus: string;
    64→  } | null>(null);
    65→  /** Transaksi QRIS yang sedang dibayar (QR tampil di layar kasir) */
    66→  const [payOrder, setPayOrder] = useState<QrisPayOrder | null>(null);
    67→
    68→  // Jam operasional live
    69→  useEffect(() => {
    70→    const tick = () =>
    71→      setClock(
    72→        new Date().toLocaleTimeString("id-ID", { hour: "2-digit", minute: "2-digit", second: "2-digit" })
    73→      );
    74→    tick();
    75→    const t = setInterval(tick, 1000);
    76→    return () => clearInterval(t);
    77→  }, []);
    78→
    79→  useEffect(() => {
    80→    load(true);
    81→  }, [load]);
    82→
    83→  const categories = useMemo(
    84→    () => ["Semua", ...Array.from(new Set(products.map((p) => p.category)))],
    85→    [products]
    86→  );
    87→
    88→  const filtered = products.filter(
    89→    (p) =>
    90→      p.available &&
    91→      (category === "Semua" || p.category === category) &&
    92→      p.name.toLowerCase().includes(query.toLowerCase().trim())
    93→  );
    94→
    95→  const addItem = (p: Product) =>
    96→    setItems((cur) => {
    97→      const found = cur.find((i) => i.product.id === p.id);
    98→      if (found) return cur.map((i) => (i.product.id === p.id ? { ...i, qty: i.qty + 1 } : i));
    99→      return [...cur, { product: p, qty: 1 }];
   100→    });
   101→
   102→  const changeQty = (id: string, delta: number) =>
   103→    setItems((cur) =>
   104→      cur
   105→        .map((i) => (i.product.id === id ? { ...i, qty: i.qty + delta } : i))
   106→        .filter((i) => i.qty > 0)
   107→    );
   108→
   109→  const subtotal = items.reduce((s, i) => s + i.product.price * i.qty, 0);
   110→  const totalItems = items.reduce((s, i) => s + i.qty, 0);
   111→
   112→  /** Cari member dari kode (dipakai saat scan barcode / ketik lalu Enter) */
   113→  const lookupMember = async (raw?: string) => {
   114→    const code = (raw ?? memberCode).replace(/\D/g, "");
   115→    if (!code) {
   116→      setMemberInfo(null);
   117→      return;
   118→    }
   119→    setMemberChecking(true);
   120→    try {
   121→      const res = await apiFetch(`/api/member/lookup?code=${encodeURIComponent(code)}`, {
   122→        cache: "no-store",
   123→      });
   124→      const data = await res.json();
   125→      if (!res.ok) {
   126→        setMemberInfo(null);
   127→        showNotice({ type: "error", title: "Member Tidak Ditemukan", message: data.error ?? "Periksa kode member." });
   128→        return;
   129→      }
   130→      setMemberInfo(data.member);
   131→    } catch {
   132→      setMemberInfo(null);
   133→      showNotice({ type: "error", title: "Gagal Memeriksa", message: "Tidak dapat menghubungi server." });
   134→    } finally {
   135→      setMemberChecking(false);
   136→    }
   137→  };
   138→
   139→  const clearMember = () => {
   140→    setMemberCode("");
   141→    setMemberInfo(null);
   142→  };
   143→
   144→  const submit = async () => {
   145→    if (items.length === 0) return;
   146→    setSubmitting(true);
   147→    try {
   148→      const res = await apiFetch("/api/orders", {
   149→        method: "POST",
   150→        headers: { "Content-Type": "application/json" },
   151→        body: JSON.stringify({
   152→          items: items.map((i) => ({ productId: i.product.id, qty: i.qty })),
   153→          customerName: customerName.trim() || `Walk-in${tableNo ? ` Meja ${tableNo}` : ""}`,
   154→          memberCode: memberInfo ? memberInfo.memberCode : undefined,
   155→          type: orderType,
   156→          tableNo: orderType === "DINE_IN" ? tableNo : undefined,
   157→          paymentMethod: payment,
   158→          source: "POS",
   159→        }),
   160→      });
   161→      const data = await res.json();
   162→      if (!res.ok) {
   163→        showNotice({ type: "error", title: "Transaksi Gagal", message: data.error ?? "Coba lagi." });
   164→        return;
   165→      }
   166→      setLastOrder({
   167→        id: data.order.id,
   168→        code: data.order.code,
   169→        total: data.order.total,
   170→        name: data.order.customerName,
   171→        paymentMethod: data.order.paymentMethod,
   172→        paymentStatus: data.order.paymentStatus,
   173→      });
   174→      setItems([]);
   175→      setCustomerName("");
   176→      setTableNo("");
   177→      clearMember();
   178→      refreshOrders("all");
   179→
   180→      // QRIS: tampilkan QR di layar kasir agar pelanggan bisa memindai.
   181→      // Sistem yang melunaskan otomatis saat dana masuk (webhook/polling) —
   182→      // kasir TIDAK menandai lunas.
   183→      if (
   184→        data.order.paymentMethod === "QRIS" &&
   185→        data.order.paymentStatus !== "PAID" &&
   186→        data.order.total > 0
   187→      ) {
   188→        setPayOrder({ id: data.order.id, code: data.order.code, total: data.order.total });
   189→        return;
   190→      }
   191→
   192→      showNotice({
   193→        type: "success",
   194→        title: "Transaksi Berhasil ✓",
   195→        message: `Pesanan ${data.order.code} (${formatIDR(data.order.total)}) telah dicatat dan masuk antrean dapur.`,
   196→      });
   197→    } catch {
   198→      showNotice({ type: "error", title: "Koneksi Bermasalah", message: "Periksa jaringan lalu ulangi." });
   199→    } finally {
   200→      setSubmitting(false);
   201→    }
   202→  };
   203→
   204→  return (
   205→    <div className="flex h-dvh flex-col bg-orange-50/40">
   206→      {/* ===== TOOLBAR OPERASIONAL POS ===== */}
   207→      <header className="z-30 shrink-0 bg-gradient-to-r from-orange-700 via-orange-600 to-orange-500 text-white shadow-xl">
   208→        <div className="flex flex-wrap items-center gap-x-4 gap-y-2 px-4 py-2.5 lg:px-6">
   209→          <div className="flex items-center gap-2.5">
   210→            <ChickenBadge size={38} />
   211→            <div className="leading-tight">
   212→              <p className="text-sm font-black uppercase tracking-wide">POS Kasir</p>
   213→              <p className="text-[10px] font-semibold text-orange-100">Ayam Geprek Sambal Ijo</p>
   214→            </div>
   215→          </div>
   216→
   217→          <div className="hidden items-center gap-1.5 rounded-full bg-white/15 px-3 py-1 text-[11px] font-bold md:flex">
   218→            <Clock className="h-3.5 w-3.5" /> <span className="tabular-nums">{clock}</span>
   219→          </div>
   220→          <div className="hidden items-center gap-1.5 rounded-full bg-white/15 px-3 py-1 text-[11px] font-bold md:flex">
   221→            <Wifi className="h-3.5 w-3.5" /> Online
   222→          </div>
   223→          <div className="flex items-center gap-1.5 rounded-full bg-white/15 px-3 py-1 text-[11px] font-bold">
   224→            👤 {user.name}
   225→          </div>
   226→
   227→          <div className="ml-auto flex items-center gap-2">
   228→            {lastOrder && (
   229→              <button
   230→                onClick={() => {
   231→                  // Pesanan QRIS belum lunas -> buka lagi layar QR (pantau otomatis)
   232→                  if (
   233→                    lastOrder.paymentMethod === "QRIS" &&
   234→                    lastOrder.paymentStatus !== "PAID" &&
   235→                    lastOrder.total > 0
   236→                  ) {
   237→                    setPayOrder({
   238→                      id: lastOrder.id,
   239→                      code: lastOrder.code,
   240→                      total: lastOrder.total,
   241→                    });
   242→                    return;
   243→                  }
   244→                  showNotice({
   245→                    type: "info",
   246→                    title: `Struk Terakhir — ${lastOrder.code}`,
   247→                    message: `${lastOrder.name} • Total ${formatIDR(lastOrder.total)}. Terima kasih!`,
   248→                  });
   249→                }}
   250→                className="press-effect flex items-center gap-1.5 rounded-full bg-white/15 px-3.5 py-1.5 text-xs font-bold hover:bg-white/25"
   251→                title={
   252→                  lastOrder.paymentMethod === "QRIS" && lastOrder.paymentStatus !== "PAID"
   253→                    ? "Buka lagi layar QRIS (memantau otomatis)"
   254→                    : "Lihat struk terakhir"
   255→                }
   256→              >
   257→                <Receipt className="h-4 w-4" /> {lastOrder.code}
   258→              </button>
   259→            )}
   260→            {(user.role === "ADMIN" || user.role === "OWNER") && (
   261→              <button
   262→                onClick={() => go("dashboard")}
   263→                className="press-effect flex items-center gap-1.5 rounded-full bg-white/15 px-3.5 py-1.5 text-xs font-bold hover:bg-white/25"
   264→              >
   265→                <MonitorSmartphone className="h-4 w-4" /> Mode Admin
   266→              </button>
   267→            )}
   268→            <button
   269→              onClick={() => load(true)}
   270→              aria-label="Muat ulang produk"
   271→              className="press-effect flex h-8 w-8 items-center justify-center rounded-full bg-white/15 hover:bg-white/25"
   272→            >
   273→              <RefreshCw className="h-4 w-4" />
   274→            </button>
   275→            <button
   276→              onClick={onLogout}
   277→              className="press-effect flex items-center gap-1.5 rounded-full bg-white/15 px-3.5 py-1.5 text-xs font-bold hover:bg-white/25"
   278→            >
   279→              <LogOut className="h-4 w-4" /> Keluar
   280→            </button>
   281→          </div>
   282→        </div>
   283→      </header>
   284→
   285→      {/* ===== BAR SCAN BARCODE & PENCARIAN (di bawah header) =====
   286→          Mode 1: Scan Member — scan/ketik kode member (8 digit) → member terpasang
   287→          Mode 2: Nama Produk — saring katalog menu berdasarkan nama produk */}
   288→      <div className="z-20 shrink-0 border-b border-orange-100 bg-white px-4 py-2.5 shadow-sm lg:px-5">
   289→        <div className="flex flex-wrap items-center gap-2">
   290→          {/* Pemilih mode */}
   291→          <div
   292→            role="tablist"
   293→            aria-label="Mode pencarian"
   294→            className="flex shrink-0 items-center gap-0.5 rounded-xl bg-orange-50 p-1 ring-1 ring-orange-200"
   295→          >
   296→            <button
   297→              role="tab"
   298→              aria-selected={scanMode === "member"}
   299→              onClick={() => setScanMode("member")}
   300→              className={cn(
   301→                "press-effect flex items-center gap-1.5 rounded-lg px-3 py-1.5 text-[11px] font-extrabold transition",
   302→                scanMode === "member"
   303→                  ? "bg-orange-600 text-white shadow-md"
   304→                  : "text-orange-700 hover:bg-orange-100"
   305→              )}
   306→            >
   307→              <ScanLine className="h-3.5 w-3.5" /> Scan Member
   308→            </button>
   309→            <button
   310→              role="tab"
   311→              aria-selected={scanMode === "product"}
   312→              onClick={() => setScanMode("product")}
   313→              className={cn(
   314→                "press-effect flex items-center gap-1.5 rounded-lg px-3 py-1.5 text-[11px] font-extrabold transition",
   315→                scanMode === "product"
   316→                  ? "bg-orange-600 text-white shadow-md"
   317→                  : "text-orange-700 hover:bg-orange-100"
   318→              )}
   319→            >
   320→              <Search className="h-3.5 w-3.5" /> Nama Produk
   321→            </button>
   322→          </div>
   323→
   324→          {/* Input sesuai mode */}
   325→          {scanMode === "member" ? (
   326→            <form
   327→              className="relative min-w-[220px] flex-1"
   328→              onSubmit={(e) => {
   329→                e.preventDefault();
   330→                void lookupMember();
   331→              }}
   332→            >
   333→              <ScanLine className="pointer-events-none absolute left-3 top-1/2 h-4 w-4 -translate-y-1/2 text-orange-400" />
   334→              <Input
   335→                inputMode="numeric"
   336→                autoFocus
   337→                placeholder="Scan / ketik kode member (8 digit), Enter…"
   338→                className="border-orange-200 pl-9 pr-24 focus-visible:ring-orange-400"
   339→                value={memberCode}
   340→                disabled={memberChecking}
   341→                onChange={(e) => {
   342→                  const v = e.target.value.replace(/\D/g, "").slice(0, 12);
   343→                  setMemberCode(v);
   344→                  // Kode member selalu 8 digit → cari otomatis begitu genap 8 angka
   345→                  if (v.length === 8 && !memberInfo) void lookupMember(v);
   346→                }}
   347→              />
   348→              <div className="absolute right-2 top-1/2 flex -translate-y-1/2 items-center gap-1.5">
   349→                {memberChecking && <Loader2 className="h-3.5 w-3.5 animate-spin text-orange-500" />}
   350→                {memberCode && !memberInfo && (
   351→                  <button
   352→                    type="button"
   353→                    onClick={() => setMemberCode("")}
   354→                    aria-label="Bersihkan kode"
   355→                    className="press-effect flex h-6 w-6 items-center justify-center rounded-full text-neutral-400 hover:bg-orange-50 hover:text-neutral-600"
   356→                  >
   357→                    <X className="h-3.5 w-3.5" />
   358→                  </button>
   359→                )}
   360→              </div>
   361→            </form>
   362→          ) : (
   363→            <div className="relative min-w-[220px] flex-1">
   364→              <Search className="pointer-events-none absolute left-3 top-1/2 h-4 w-4 -translate-y-1/2 text-orange-400" />
   365→              <Input
   366→                autoFocus
   367→                placeholder="Cari menu dengan nama produk… (cth: geprek, mozarella)"
   368→                className="border-orange-200 pl-9 pr-9 focus-visible:ring-orange-400"
   369→                value={query}
   370→                onChange={(e) => setQuery(e.target.value)}
   371→              />
   372→              {query && (
   373→                <button
   374→                  onClick={() => setQuery("")}
   375→                  aria-label="Bersihkan pencarian"
   376→                  className="press-effect absolute right-2 top-1/2 flex h-6 w-6 -translate-y-1/2 items-center justify-center rounded-full text-neutral-400 hover:bg-orange-50 hover:text-neutral-600"
   377→                >
   378→                  <X className="h-3.5 w-3.5" />
   379→                </button>
   380→              )}
   381→            </div>
   382→          )}
   383→        </div>
   384→
   385→        {/* Chip member yang sedang terpasang (hasil scan) */}
   386→        {memberInfo && (
   387→          <div className="mt-2 flex items-center gap-2 rounded-xl border border-emerald-200 bg-emerald-50 px-3 py-2">
   388→            <BadgeCheck className="h-4 w-4 shrink-0 text-emerald-600" />
   389→            <div className="min-w-0 flex-1">
   390→              <p className="truncate text-xs font-black text-emerald-800">
   391→                {memberInfo.name} <span className="font-mono">({memberInfo.memberCode})</span>
   392→              </p>
   393→              <p className="flex items-center gap-1 text-[10px] font-bold text-emerald-700">
   394→                <Coins className="h-3 w-3" /> Saldo {memberInfo.points.toLocaleString("id-ID")} poin — masuk
   395→                saat pesanan selesai
   396→              </p>
   397→            </div>
   398→            <button
   399→              onClick={clearMember}
   400→              aria-label="Lepas member"
   401→              className="press-effect flex h-6 w-6 shrink-0 items-center justify-center rounded-full text-emerald-600 hover:bg-emerald-100"
   402→            >
   403→              <X className="h-3.5 w-3.5" />
   404→            </button>
   405→          </div>
   406→        )}
   407→      </div>
   408→
   409→      <div className="flex min-h-0 flex-1 flex-col gap-4 overflow-hidden p-4 lg:flex-row lg:p-5">
   410→        {/* ===== KATALOG PRODUK ===== */}
   411→        <main className="flex min-h-0 flex-1 flex-col gap-3 overflow-hidden">
   412→          <div className="scrollbar-slim flex gap-1.5 overflow-x-auto pb-0.5">
   413→              {categories.map((c) => (
   414→                <button
   415→                  key={c}
   416→                  onClick={() => setCategory(c)}
   417→                  className={cn(
   418→                    "shrink-0 rounded-full px-3.5 py-2 text-xs font-bold transition",
   419→                    category === c
   420→                      ? "bg-orange-600 text-white shadow-md"
   421→                      : "bg-white text-orange-700 ring-1 ring-orange-200 hover:bg-orange-50"
   422→                  )}
   423→                >
   424→                  {c}
   425→                </button>
   426→              ))}
   427→          </div>
   428→
   429→          <div className="scrollbar-slim grid min-h-0 flex-1 auto-rows-min grid-cols-2 content-start gap-3 overflow-y-auto pb-2 sm:grid-cols-3 xl:grid-cols-4 2xl:grid-cols-5">
   430→            <AnimatePresence mode="popLayout">
   431→              {filtered.map((p) => {
   432→                const inCart = items.find((i) => i.product.id === p.id)?.qty ?? 0;
   433→                return (
   434→                  <motion.button
   435→                    layout={!prefersReduced}
   436→                    key={p.id}
   437→                    initial={prefersReduced ? false : { opacity: 0, scale: 0.94 }}
   438→                    animate={{ opacity: 1, scale: 1 }}
   439→                    exit={prefersReduced ? undefined : { opacity: 0, scale: 0.94 }}
   440→                    whileTap={{ scale: 0.96 }}
   441→                    onClick={() => addItem(p)}
   442→                    className={cn(
   443→                      "card-lift group relative flex flex-col rounded-2xl border bg-white p-3 text-left shadow-sm",
   444→                      inCart > 0 ? "border-orange-400 ring-2 ring-orange-200" : "border-orange-100"
   445→                    )}
   446→                  >
   447→                    {inCart > 0 && (
   448→                      <span className="absolute -right-1.5 -top-1.5 flex h-6 min-w-6 items-center justify-center rounded-full bg-orange-600 px-1.5 text-xs font-black text-white shadow-lg">
   449→                        {inCart}
   450→                      </span>
   451→                    )}
   452→                    <ProductThumb
   453→                      product={p}
   454→                      className="mb-2 aspect-square w-full text-4xl"
   455→                      imgClassName="aspect-square transition-transform group-hover:scale-110"
   456→                      emojiClassName="transition-transform group-hover:scale-110"
   457→                    />
   458→                    <p className="line-clamp-1 text-sm font-bold text-neutral-800">{p.name}</p>
   459→                    <p className="text-[10px] text-neutral-400">{p.category}</p>
   460→                    <p className="mt-1 text-sm font-black text-orange-600">{formatIDR(p.price)}</p>
   461→                  </motion.button>
   462→                );
   463→              })}
   464→            </AnimatePresence>
   465→            {filtered.length === 0 && (
   466→              <div className="col-span-full flex items-center justify-center rounded-2xl border border-dashed border-orange-200 bg-white/60 p-10 text-sm text-neutral-400">
   467→                Menu tidak ditemukan
   468→              </div>
   469→            )}
   470→          </div>
   471→        </main>
   472→
   473→        {/* ===== PANEL KERANJANG / PEMBAYARAN ===== */}
   474→        <aside className="flex min-h-0 w-full shrink-0 flex-col overflow-hidden rounded-2xl border border-orange-100 bg-white shadow-xl lg:w-[360px]">
   475→          <div className="flex items-center justify-between border-b border-orange-100 bg-gradient-to-r from-orange-600 to-orange-500 px-4 py-3 text-white">
   476→            <div className="flex items-center gap-2">
   477→              <ClipboardList className="h-5 w-5" />
   478→              <div>
   479→                <p className="text-sm font-black">Pesanan Saat Ini</p>
   480→                <p className="text-[10px] text-orange-100">{totalItems} item di keranjang</p>
   481→              </div>
   482→            </div>
   483→            {items.length > 0 && (
   484→              <button
   485→                onClick={() => setItems([])}
   486→                className="press-effect flex items-center gap-1 rounded-full bg-white/15 px-2.5 py-1 text-[10px] font-bold hover:bg-white/25"
   487→              >

   628→                  {payment === "QRIS" ? "Bayar QRIS" : "Bayar & Cetak"} — {formatIDR(subtotal)}
