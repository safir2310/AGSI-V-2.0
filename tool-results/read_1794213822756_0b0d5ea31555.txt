    25→    25→/** Pesanan yang akan dibayar via QRIS */
    26→    26→export type QrisPayOrder = { id: string; code: string; total: number };
    27→    27→
    28→    28→type Phase = "scan" | "paid" | "cancelled";
    29→    29→
    30→    30→type QrisInfo = {
    31→    31→  payload: string | null;
    32→    32→  imageUrl: string | null;
    33→    33→  mode: "GATEWAY" | "LOCAL";
    34→    34→  autoVerify: boolean;
    35→    35→  /** gateway QRIS bawaan (sandbox) aktif — tombol simulasi dana ditampilkan */
    36→    36→  sandbox: boolean;
    37→    37→  /** penyedia verifikasi: "casaku" | "midtrans" | "midtrans-sandbox" | "internal" | null */
    38→    38→  provider: string | null;
    39→    39→  /** Alasan penyedia utama (mis. Casaku/Midtrans) ditolak sehingga QR berasal dari fallback uji lokal */
    40→    40→  fallbackReason: string | null;
    41→    41→  /** Galat gateway saat mode LOCAL — ditampilkan jujur (kanal belum aktif, dll.) */
    42→    42→  gatewayError: string | null;
    43→    43→  /** Sandbox Midtrans: tautan simulator uji pembayaran (QR berisi URL uji) */
    44→    44→  sandboxSimulatorUrl: string | null;
    45→    45→  /** Nominal FINAL yang harus dibayar (total + kode unik Casaku bila aktif) */
    46→    46→  payAmount: number | null;
    47→    47→  /** Kode unik yang disematkan Casaku ke nominal (bila aktif) */
    48→    48→  uniqueCode: number | null;
    49→    49→};
    50→    50→
    51→    51→/** Interval polling verifikasi otomatis */
    52→    52→const POLL_MS = 4000;
    53→    53→/** Batas waktu pembayaran QRIS — 5 MENIT sejak pesanan dibuat (permintaan
    54→    54→ *  pemilik). Lewat dari itu pesanan batal otomatis (server membatalkan
    55→    55→ *  saat sweep/poll) dan kode bayar hangus. */
    56→    56→const DURATION_SEC = 5 * 60; // 5 menit
    57→    57→
    58→    58→function mmss(sec: number): string {
    59→    59→  const m = Math.floor(sec / 60);
    60→    60→  const s = sec % 60;
    61→    61→  return `${String(m).padStart(2, "0")}:${String(s).padStart(2, "0")}`;
    62→    62→}
    63→    63→
    64→    64→/**
    65→    65→ * Layar pembayaran QRIS — pelanggan maupun kasir memakai layar yang sama.
    66→    66→ *
    67→    67→ * VERIFIKASI OTOMATIS: setelah QR tampil, layar ini memantau status ke server
    68→    68→ * tiap beberapa detik; server bertanya ke penyedia QRIS "sudah masuk belum
    69→    69→ * dananya?" dan melunaskan pesanan begitu dana masuk. TIDAK ADA tombol
    70→    70→ * "Saya Sudah Bayar" / "Uang Sudah Masuk" — kasir memang tidak boleh
    71→    71→ * (dan tidak bisa) melunaskan pesanan QRIS secara manual.
    72→    72→ */
    73→    73→export function QrisPaySheet({
    74→    74→  order,
    75→    75→  variant,
    76→    76→  onClose,
    77→    77→  onPaid,
    78→    78→  onPrintReceipt,
    79→    79→}: {
    80→    80→  order: QrisPayOrder;
    81→    81→  variant: "customer" | "pos";
    82→    82→  /** Dipanggil saat layar ditutup. result.paid memberi tahu apakah pembayaran
    83→    83→   *  SUDAH lunas saat ditutup (false = belum dibayar / sudah batal) —
    84→    84→   *  pemanggil TIDAK boleh menampilkan pesan sukses bila paid=false. */
    85→    85→  onClose: (result: { paid: boolean }) => void;
    86→    86→  /** dipanggil saat status pembayaran berubah menjadi lunas (mis. refresh daftar) */
    87→    87→  onPaid?: () => void | Promise<void>;
    88→    88→  /** POS: cetak struk setelah lunas (bila disediakan halaman pemanggil) */
    89→    89→  onPrintReceipt?: () => void | Promise<void>;
    90→    90→}) {
    91→    91→  const prefersReduced = useReducedMotion();
    92→    92→  const isPos = variant === "pos";
    93→    93→
    94→    94→  const [info, setInfo] = useState<QrisInfo | null>(null);
    95→    95→  const [loadError, setLoadError] = useState<string | null>(null);
    96→    96→  const [phase, setPhase] = useState<Phase>("scan");
    97→    97→  const [secondsLeft, setSecondsLeft] = useState(DURATION_SEC);
    98→    98→  const [checkingNow, setCheckingNow] = useState(false);
    99→    99→  const [regenerating, setRegenerating] = useState(false);
   100→   100→  /** status terakhir dari penyedia (mis. "EXPIRED") */
   101→   101→  const [gatewayStatus, setGatewayStatus] = useState<string | null>(null);
   102→   102→  /** gangguan sementara saat memantau (diciutkan agar tidak bikin cemas) */
   103→   103→  const [monitorNote, setMonitorNote] = useState<string | null>(null);
   104→   104→  /** batas waktu pesanan (createdAt + 5 menit) dalam ms epoch */
   105→   105→  const [orderDeadline, setOrderDeadline] = useState<number | null>(null);
   106→   106→  const paidNotified = useRef(false);
   107→   107→  /** onPaid disimpan di ref: identitas prop baru dari re-render parent
   108→   108→   *  (mis. jam kasir yang berdetik tiap detik) TIDAK boleh memicu ulang
   109→   109→   *  efek fetch/polling di bawah — itulah penyebab layar QR berkedip. */
   110→   110→  const onPaidRef = useRef(onPaid);
   111→   111→  useEffect(() => {
   112→   112→    onPaidRef.current = onPaid;
   113→   113→  }, [onPaid]);
   114→   114→
   115→   115→  /** Stabil antar-render — aman dipakai sebagai dependensi efek */
   116→   116→  const notifyPaid = useCallback(() => {
   117→   117→    if (paidNotified.current) return;
   118→   118→    paidNotified.current = true;
   119→   119→    void onPaidRef.current?.();
   120→   120→  }, []);
   121→   121→
   122→   122→  /** Penomoran permintaan: respons yang datang terlambat dari fetch lama
   123→   123→   *  tidak boleh menimpa data fetch terbaru (cegah QR "maju-mundur"). */
   124→   124→  const fetchSeq = useRef(0);
   125→   125→  /** Ambil kode QRIS dari server (gateway bila aktif, fallback lokal).
   126→   126→   *  soft=true → QR lama TETAP tampil selagi data baru diambil (kode
   127→   127→   *  langsung tertukar saat respons tiba) sehingga layar tak berkedip. */
   128→   128→  const fetchPayload = useCallback(async (soft = false) => {
   129→   129→    const seq = ++fetchSeq.current;
   130→   130→    setLoadError(null);
   131→   131→    setGatewayStatus(null);
   132→   132→    if (!soft) setInfo(null);
   133→   133→    try {
   134→   134→      const res = await apiFetch(`/api/orders/${order.id}/qris`, { cache: "no-store" });
   135→   135→      const data = await res.json();
   136→   136→      if (seq !== fetchSeq.current) return; // respons kedaluwarsa — abaikan
   137→   137→      if (!res.ok) {
   138→   138→        // Pesanan ternyata sudah lunas → langsung tampilkan sukses
   139→   139→        if (typeof data.error === "string" && data.error.toLowerCase().includes("lunas")) {
   140→   140→          setPhase("paid");
   141→   141→          notifyPaid();
   142→   142→          return;
   143→   143→        }
   144→   144→        // Pesanan dibatalkan otomatis (lewat batas 5 menit) → tampilan batal
   145→   145→        if (data.cancelled === true) {
   146→   146→          setPhase("cancelled");
   147→   147→          return;
   148→   148→        }
   149→   149→        setLoadError(data.error ?? "Gagal memuat kode QRIS.");
   150→   150→        return;
   151→   151→      }
   152→   152→      if (data.paid === true) {
   153→   153→        setPhase("paid");
   154→   154→        notifyPaid();
   155→   155→        return;
   156→   156→      }
   157→   157→      const expiresAt = data.expiresAt ? new Date(data.expiresAt as string) : null;
   158→   158→      const cancelMs = data.cancelAt ? new Date(data.cancelAt as string).getTime() : null;
   159→   159→      if (cancelMs) setOrderDeadline(cancelMs);
   160→   160→      setInfo({
   161→   161→        payload: (data.payload as string) ?? null,
   162→   162→        imageUrl: (data.imageUrl as string) ?? null,
   163→   163→        mode: data.mode === "GATEWAY" ? "GATEWAY" : "LOCAL",
   164→   164→        autoVerify: !!data.autoVerify,
   165→   165→        sandbox: data.sandbox === true,
   166→   166→        provider: typeof data.provider === "string" ? data.provider : null,
   167→   167→        fallbackReason:
   168→   168→          typeof data.fallbackReason === "string" ? data.fallbackReason : null,
   169→   169→        gatewayError:
   170→   170→          typeof data.gatewayError === "string" ? data.gatewayError : null,
   171→   171→        sandboxSimulatorUrl:
   172→   172→          typeof data.sandboxSimulatorUrl === "string" ? data.sandboxSimulatorUrl : null,
   173→   173→        payAmount: typeof data.amount === "number" ? data.amount : null,
   174→   174→        uniqueCode: typeof data.uniqueCode === "number" ? data.uniqueCode : null,
   175→   175→      });
   176→   176→      // Hitung mundur pakai batas TERKETAT: umur kode gateway vs batas pesanan
   177→   177→      const nowMs = Date.now();
   178→   178→      const gatewaySec = expiresAt
   179→   179→        ? Math.max(0, Math.floor((expiresAt.getTime() - nowMs) / 1000))
   180→   180→        : DURATION_SEC;
   181→   181→      const orderSec = cancelMs
   182→   182→        ? Math.max(0, Math.floor((cancelMs - nowMs) / 1000))
   183→   183→        : DURATION_SEC;
   184→   184→      setSecondsLeft(Math.max(0, Math.min(DURATION_SEC, gatewaySec, orderSec)));
   185→   185→    } catch {
   186→   186→      if (seq !== fetchSeq.current) return; // respons kedaluwarsa — abaikan
   187→   187→      setLoadError("Tidak dapat menghubungi server. Periksa koneksi lalu coba lagi.");
   188→   188→    }
   189→   189→    // notifyPaid stabil ([]) — dependensi ini tidak berubah antar-render
   190→   190→    // parent, sehingga QR hanya diambil SEKALI per pesanan (bukan tiap detik).
   191→   191→  }, [order.id, notifyPaid]);
   192→   192→
   193→   193→  useEffect(() => {
   194→   194→    void fetchPayload();
   195→   195→  }, [fetchPayload]);
   196→   196→
   197→   197→  // Hitung waktu mundur (hanya saat fase scan)
   198→   198→  useEffect(() => {
   199→   199→    if (phase !== "scan") return;
   200→   200→    const t = setInterval(() => {
   201→   201→      setSecondsLeft((s) => (s > 0 ? s - 1 : 0));
   202→   202→    }, 1000);
   203→   203→    return () => clearInterval(t);
   204→   204→  }, [phase]);
   205→   205→
   206→   206→  /** Tutup layar pembayaran + notifikasi sesuai hasil pembayaran terakhir
   207→   207→   *  (permintaan pemilik):
   208→   208→   *  - belum lunas  -> "Pembayaran belum berhasil — silakan lanjutkan
   209→   209→   *                    pembayaran" (merah, agar kasir/pelanggan tak bingung)
   210→   210→   *  - sudah lunas  -> "Pembayaran berhasil — pesanan Anda sedang diproses"
   211→   211→   *  - sudah batal  -> cukup tutup (layar batal sudah menjelaskan) */
   212→   212→  const requestClose = useCallback(() => {
   213→   213→    if (phase === "paid") {
   214→   214→      toast({
   215→   215→        title: "Pembayaran berhasil",
   216→   216→        description: `Pesanan ${order.code} ${formatIDR(order.total)} sudah terkonfirmasi — pesanan Anda sedang diproses.`,
   217→   217→        duration: 7000,
   218→   218→      });
   219→   219→    } else if (phase === "scan") {
   220→   220→      toast({
   221→   221→        variant: "destructive",
   222→   222→        title: "Pembayaran belum berhasil",
   223→   223→        description: "Silakan lanjutkan pembayaran QRIS sebelum waktu habis — pesanan belum terlunasi.",
   224→   224→        duration: 7000,
   225→   225→      });
   226→   226→    }
   227→   227→    onClose({ paid: phase === "paid" });
   228→   228→  }, [phase, order.code, order.total, onClose]);
   229→   229→
   230→   230→  // Tutup dengan tombol Escape (lewat requestClose: notifikasi status ikut tampil)
   231→   231→  useEffect(() => {
   232→   232→    const onKey = (e: KeyboardEvent) => {
   233→   233→      if (e.key === "Escape") requestClose();
   234→   234→    };
   235→   235→    window.addEventListener("keydown", onKey);
   236→   236→    return () => window.removeEventListener("keydown", onKey);
   237→   237→  }, [requestClose]);
   238→   238→
   239→   239→  // ===== Pemantauan verifikasi otomatis =====
   240→   240→  // hasInfo (boolean) sebagai dependensi — bukan objek info — agar interval
   241→   241→  // polling TIDAK dibongkar-pasang setiap data baru masuk.
   242→   242→  const hasInfo = info !== null;
   243→   243→  useEffect(() => {
   244→   244→    if (phase !== "scan" || !hasInfo) return;
   245→   245→    let alive = true;
   246→   246→    const poll = async (force = false) => {
   247→   247→      try {
   248→   248→        const res = await apiFetch(
   249→   249→          `/api/orders/${order.id}/qris/status${force ? "?force=1" : ""}`,
   250→   250→          { cache: "no-store" }
   251→   251→        );
   252→   252→        if (!res.ok || !alive) return;
   253→   253→        const data = await res.json();
   254→   254→        if (data.paymentStatus === "PAID") {
   255→   255→          setPhase("paid");
   256→   256→          notifyPaid();
   257→   257→          return;
   258→   258→        }
   259→   259→        // Pesanan batal otomatis (lewat batas 5 menit / dana telat)
   260→   260→        if (data.cancelled === true || data.status === "CANCELLED") {
   261→   261→          setMonitorNote((data.notice as string) ?? null);
   262→   262→          setPhase("cancelled");
   263→   263→          return;
   264→   264→        }
   265→   265→        if (data.cancelAt) setOrderDeadline(new Date(data.cancelAt as string).getTime());
   266→   266→        setGatewayStatus((data.gatewayStatus as string) ?? null);
   267→   267→        setMonitorNote((data.gatewayError as string) ?? null);
   268→   268→      } catch {
   269→   269→        // koneksi tersendat — coba lagi pada tick berikutnya
   270→   270→      }
   271→   271→    };
   272→   272→    void poll();
   273→   273→    const t = setInterval(() => void poll(), POLL_MS);
   274→   274→    return () => {
   275→   275→      alive = false;
   276→   276→      clearInterval(t);
   277→   277→    };
   278→   278→  }, [phase, hasInfo, order.id, notifyPaid]);
   279→   279→
   280→   280→  /** POS: paksa sistem memeriksa penyedia sekarang (tetap otomatis, bukan menandai lunas) */
   281→   281→  const checkNow = async () => {
   282→   282→    if (checkingNow) return;
   283→   283→    setCheckingNow(true);
   284→   284→    try {
   285→   285→      const res = await apiFetch(`/api/orders/${order.id}/qris/status?force=1`, {
   286→   286→        cache: "no-store",
   287→   287→      });
   288→   288→      const data = await res.json();
   289→   289→      if (res.ok && data.paymentStatus === "PAID") {
   290→   290→        setPhase("paid");
   291→   291→        notifyPaid();
   292→   292→      }
   293→   293→    } catch {
   294→   294→      /* abaikan — polling reguler akan menyusul */
   295→   295→    } finally {
   296→   296→      setCheckingNow(false);
   297→   297→    }
   298→   298→  };
   299→   299→
   300→   300→  /** Buat ulang kode bayar (transaksi lama kedaluwarsa) — soft: QR lama
   301→   301→   *  tetap tampil sampai kode baru siap, tanpa kedipan spinner. */
   302→   302→  const regenerate = async () => {
   303→   303→    if (regenerating) return;
   304→   304→    setRegenerating(true);
   305→   305→    await fetchPayload(true);
   306→   306→    setRegenerating(false);
   307→   307→  };
   308→   308→
   309→   309→  const expired = secondsLeft <= 0;
   310→   310→  const codeStale = expired || gatewayStatus === "EXPIRED";
   311→   311→  const autoVerify = info?.autoVerify ?? false;
   312→   312→  /** Batas pesanan (5 menit) sudah terlewat — pesanan mati, jangan tawarkan buat ulang kode */
   313→   313→  const pastDeadline = orderDeadline !== null && Date.now() >= orderDeadline;
   314→   314→
   315→   315→  // Portal ke <body> agar modal tidak terjebak stacking context `main` (z-10)
   316→   316→  // — selalu tampil DI ATAS navigasi bawah liquid glass (z-50) & header (z-40).
   317→   317→  return createPortal(
   318→   318→    <AnimatePresence>
   319→   319→      <motion.div
   320→   320→        key="qris-overlay"
   321→   321→        initial={prefersReduced ? false : { opacity: 0 }}
   322→   322→        animate={{ opacity: 1 }}
   323→   323→        exit={prefersReduced ? undefined : { opacity: 0 }}
   324→   324→        className="fixed inset-0 z-[95] flex items-end justify-center bg-neutral-900/70 backdrop-blur-sm sm:items-center sm:p-4"
   325→   325→        role="dialog"
   326→   326→        aria-modal="true"
   327→   327→        aria-label="Pembayaran QRIS"
   328→   328→      >
   329→   329→        <motion.div
   330→   330→          initial={prefersReduced ? false : { y: 48, scale: 0.97 }}
   331→   331→          animate={{ y: 0, scale: 1 }}
   332→   332→          exit={prefersReduced ? undefined : { y: 48, opacity: 0 }}
   333→   333→          transition={{ type: "spring", stiffness: 380, damping: 34 }}
   334→   334→          className="flex max-h-[94dvh] w-full max-w-md flex-col overflow-hidden rounded-t-3xl bg-white shadow-2xl sm:rounded-3xl"
   335→   335→        >
   336→   336→          {/* ===== KEPALA ===== */}
   337→   337→          <header className="relative shrink-0 bg-gradient-to-r from-orange-700 via-orange-600 to-orange-500 px-5 py-4 text-white">
   338→   338→            <div className="flex items-center gap-3">
   339→   339→              <span className="flex h-10 w-10 items-center justify-center rounded-2xl bg-white/20 ring-1 ring-white/30">
   340→   340→                <ScanLine className="h-5 w-5" />
   341→   341→              </span>
   342→   342→              <div className="min-w-0 flex-1">
   343→   343→                <p className="text-sm font-black tracking-wide">Pembayaran QRIS</p>
   344→   344→                <p className="truncate text-[11px] font-semibold text-orange-100">
   345→   345→                  {order.code} • {isPos ? "Kasir" : "Aplikasi"}
   346→   346→                </p>
   347→   347→              </div>
   348→   348→              <button
   349→   349→                onClick={requestClose}
   350→   350→                aria-label="Tutup"
   351→   351→                className="press-effect flex h-9 w-9 items-center justify-center rounded-full bg-white/15 hover:bg-white/30"
   352→   352→              >
   353→   353→                <X className="h-4.5 w-4.5" />
   354→   354→              </button>
   355→   355→            </div>
   356→   356→          </header>
   357→   357→
   358→   358→          {/* ===== ISI ===== */}
   359→   359→          <div className="scrollbar-slim overflow-y-auto px-5 pb-[max(1.25rem,env(safe-area-inset-bottom))] pt-4">
   360→   360→            {phase === "paid" ? (
   361→   361→              /* --------- BERHASIL (verifikasi otomatis) --------- */
   362→   362→              <div className="flex flex-col items-center gap-3 py-8 text-center">
   363→   363→                <motion.span
   364→   364→                  initial={prefersReduced ? false : { scale: 0.5, opacity: 0 }}
   365→   365→                  animate={{ scale: 1, opacity: 1 }}
   366→   366→                  transition={{ type: "spring", stiffness: 300, damping: 18 }}
   367→   367→                  className="flex h-20 w-20 items-center justify-center rounded-full bg-emerald-50 ring-4 ring-emerald-100"
   368→   368→                >
   369→   369→                  <BadgeCheck className="h-11 w-11 text-emerald-600" />
   370→   370→                </motion.span>
   371→   371→                <h2 className="text-lg font-black text-neutral-900">Pembayaran Diterima! 🎉</h2>
   372→   372→                <p className="max-w-[300px] text-xs leading-relaxed text-neutral-500">
   373→   373→                  {formatIDR(order.total)} untuk pesanan {order.code} sudah lunas —
   374→   374→                  terverifikasi <strong className="font-bold text-emerald-600">otomatis oleh sistem</strong>{" "}
   375→   375→                  saat dana masuk. Geprekmu segera dimasak ya!
   376→   376→                </p>
   377→   377→                {isPos && onPrintReceipt ? (
   378→   378→                  <div className="mt-2 flex w-full items-center gap-2">
   379→   379→                    <button
   380→   380→                      onClick={requestClose}
   381→   381→                      className="press-effect h-11 flex-1 rounded-xl border border-neutral-200 bg-white text-sm font-extrabold text-neutral-600 hover:bg-neutral-50"
   382→   382→                    >
   383→   383→                      Selesai
   384→   384→                    </button>
   385→   385→                    <button
   386→   386→                      onClick={() => void onPrintReceipt()}
   387→   387→                      className="press-effect flex h-11 flex-[1.6] items-center justify-center gap-2 rounded-xl bg-orange-600 text-sm font-extrabold text-white shadow-lg shadow-orange-500/30 hover:bg-orange-700"
   388→   388→                    >
   389→   389→                      <Printer className="h-4 w-4" /> Cetak Struk
   390→   390→                    </button>
