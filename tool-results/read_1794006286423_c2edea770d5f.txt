     1→"use client";
     2→
     3→import { useCallback, useEffect, useMemo, useRef, useState } from "react";
     4→import { AnimatePresence, motion, useReducedMotion } from "framer-motion";
     5→import {
     6→  Bike,
     7→  BadgeCheck,
     8→  ClipboardList,
     9→  Clock,
    10→  Coins,

    12→  LogOut,

    21→  Trash2,

    23→  Wifi,
    24→  X,
    25→} from "lucide-react";
    26→import { Input } from "@/components/ui/input";
    27→import { ChickenBadge } from "@/components/logo/ChickenLogo";
    28→import { ProductThumb } from "@/components/shared/ProductThumb";
    29→import { QrisPaySheet, type QrisPayOrder } from "@/components/shared/QrisPaySheet";
    30→import { refreshOrders, useProductsStore, useSettingsStore } from "@/lib/hooks";
    31→import { useNoticeStore, type AppView } from "@/lib/store";
    32→import { apiFetch } from "@/lib/api";
    33→import { formatIDR, type AppUser, type Product } from "@/lib/types";
    34→import { cn } from "@/lib/utils";
    35→
    36→type PosItem = { product: Product; qty: number };
    37→
    38→export function PosView({ user, go, onLogout }: { user: AppUser; go: (v: AppView) => void; onLogout: () => void }) {
    39→  const prefersReduced = useReducedMotion();
    40→  const showNotice = useNoticeStore((s) => s.show);
    41→  const { products, load } = useProductsStore();
    42→  const [query, setQuery] = useState("");
    43→  /** Popup hasil pencarian menu di bawah input cari (autocompletion) */
    44→  const [searchOpen, setSearchOpen] = useState(false);
    45→  const searchBoxRef = useRef<HTMLDivElement | null>(null);
    46→  const [category, setCategory] = useState("Semua");
    47→  /** Mode bar pencarian di bawah header: scan barcode member ATAU cari nama produk */
    48→  const [scanMode, setScanMode] = useState<"member" | "product">("product");
    49→  const [items, setItems] = useState<PosItem[]>([]);
    50→  const [customerName, setCustomerName] = useState("");
    51→  const [tableNo, setTableNo] = useState("");
    52→  /** Kode member (hasil scan barcode kartu member / ketikan kasir) */
    53→  const [memberCode, setMemberCode] = useState("");
    54→  const [memberInfo, setMemberInfo] = useState<{ name: string; memberCode: string; points: number } | null>(null);
    55→  const [memberChecking, setMemberChecking] = useState(false);
    56→  const [orderType, setOrderType] = useState<"DINE_IN" | "TAKEAWAY" | "DELIVERY">("DINE_IN");
    57→  const [payment, setPayment] = useState<"CASH" | "QRIS">("CASH");
    58→  // Saklar layanan toko (Profil Admin → Pengaturan Toko):
    59→  // nonaktif → tombol "Antar" / "QRIS" disembunyikan & pilihan lama di-reset
    60→  const deliveryEnabled = useSettingsStore((s) => s.deliveryEnabled);
    61→  const qrisEnabled = useSettingsStore((s) => s.qrisEnabled);
    62→
    63→  // Layanan dimatikan admin → pilihan lama otomatis dikembalikan
    64→  useEffect(() => {
    65→    if (!deliveryEnabled) {
    66→      setOrderType((t) => (t === "DELIVERY" ? "DINE_IN" : t));
    67→    }
    68→    if (!qrisEnabled) {
    69→      setPayment((p) => (p === "QRIS" ? "CASH" : p));
    70→    }
    71→  }, [deliveryEnabled, qrisEnabled]);
    72→  const [submitting, setSubmitting] = useState(false);
    73→  const [clock, setClock] = useState("");
    74→  const [lastOrder, setLastOrder] = useState<{
    75→    id: string;
    76→    code: string;
    77→    total: number;
    78→    name: string;
    79→    paymentMethod: string;
    80→    paymentStatus: string;
    81→  } | null>(null);
    82→  /** Transaksi QRIS yang sedang dibayar (QR tampil di layar kasir) */
    83→  const [payOrder, setPayOrder] = useState<QrisPayOrder | null>(null);
    84→
    85→  // Jam operasional live
    86→  useEffect(() => {
    87→    const tick = () =>
    88→      setClock(
    89→        new Date().toLocaleTimeString("id-ID", { hour: "2-digit", minute: "2-digit", second: "2-digit" })
    90→      );
    91→    tick();
    92→    const t = setInterval(tick, 1000);
    93→    return () => clearInterval(t);
    94→  }, []);
    95→
    96→  useEffect(() => {
    97→    load(true);
    98→  }, [load]);
    99→
   100→  const categories = useMemo(
   101→    () => ["Semua", ...Array.from(new Set(products.map((p) => p.category)))],
   102→    [products]
   103→  );
   104→
   105→  const q = query.trim().toLowerCase();
   106→  const filtered = products.filter((p) => {
   107→    if (!p.available) return false;
   108→    // Sedang mencari → hasil dari SEMUA kategori (saringan kategori dilewati),
   109→    // cocokkan nama produk ATAU nama kategorinya
   110→    if (q) {
   111→      return (
   112→        p.name.toLowerCase().includes(q) || p.category.toLowerCase().includes(q)
   113→      );
   114→    }
   115→    return category === "Semua" || p.category === category;
   116→  });
   117→
   118→  /** Menu yang tampil di popup pencarian (maks. 6 supaya ringkas) */
   119→  const POPUP_MAX = 6;
   120→  const popupMatches = q ? filtered.slice(0, POPUP_MAX) : [];
   121→
   122→  /** Tutup popup saat klik di luar kotak pencarian atau tekan Escape */
   123→  useEffect(() => {
   124→    if (!searchOpen) return;
   125→    const onPointerDown = (e: PointerEvent) => {
   126→      if (searchBoxRef.current && !searchBoxRef.current.contains(e.target as Node)) {
   127→        setSearchOpen(false);
   128→      }
   129→    };
   130→    const onKey = (e: KeyboardEvent) => {
   131→      if (e.key === "Escape") setSearchOpen(false);
   132→    };
   133→    window.addEventListener("pointerdown", onPointerDown);
   134→    window.addEventListener("keydown", onKey);
   135→    return () => {
   136→      window.removeEventListener("pointerdown", onPointerDown);
   137→      window.removeEventListener("keydown", onKey);
   138→    };
   139→  }, [searchOpen]);
   140→
   141→  const addItem = (p: Product) =>
   142→    setItems((cur) => {
   143→      const found = cur.find((i) => i.product.id === p.id);
   144→      if (found) return cur.map((i) => (i.product.id === p.id ? { ...i, qty: i.qty + 1 } : i));
   145→      return [...cur, { product: p, qty: 1 }];
   146→    });
   147→
   148→  /** Pilih menu dari popup: masuk keranjang, bersihkan pencarian, tutup popup */
   149→  const pickFromPopup = (p: Product) => {
   150→    addItem(p);
   151→    setQuery("");
   152→    setSearchOpen(false);
   153→  };
   154→
   155→  const changeQty = (id: string, delta: number) =>
   156→    setItems((cur) =>
   157→      cur
   158→        .map((i) => (i.product.id === id ? { ...i, qty: i.qty + delta } : i))
   159→        .filter((i) => i.qty > 0)
   160→    );
   161→
   162→  const subtotal = items.reduce((s, i) => s + i.product.price * i.qty, 0);
   163→  const totalItems = items.reduce((s, i) => s + i.qty, 0);
   164→
   165→  /** Banyak jenis menu di keranjang (≥7) → baris keranjang otomatis jadi
   166→   *  RINGKAS & TRANSPARAN (kaca) supaya daftar tetap ringan digulir dan
   167→   *  kasir tetap fokus ke total & tombol bayar saat transaksi ramai */
   168→  const compactCart = items.length > 6;
   169→
   170→  /** Mode scroll daftar keranjang: gradasi atas/bawah menandai daftar bisa
   171→   *  digulir + item yang BARU ditambahkan langsung digulirkan ke layar */
   172→  const listRef = useRef<HTMLDivElement | null>(null);
   173→  const [fade, setFade] = useState({ top: false, bottom: false });
   174→  const updateFades = useCallback(() => {
   175→    const el = listRef.current;
   176→    if (!el) return;
   177→    setFade({
   178→      top: el.scrollTop > 4,
   179→      bottom: el.scrollTop + el.clientHeight < el.scrollHeight - 4,
   180→    });
   181→  }, []);
   182→  const prevRows = useRef(0);
   183→  useEffect(() => {
   184→    const el = listRef.current;
   185→    // Jenis menu BARU masuk keranjang → gulir ke bawah agar langsung terlihat
   186→    if (items.length > prevRows.current && el) {
   187→      el.scrollTo({
   188→        top: el.scrollHeight,
   189→        behavior: prefersReduced ? "auto" : "smooth",
   190→      });
   191→    }
   192→    prevRows.current = items.length;
   193→    updateFades();
   194→  }, [items, prefersReduced, updateFades]);
   195→
   196→  /** Cari member dari kode (dipakai saat scan barcode / ketik lalu Enter) */
   197→  const lookupMember = async (raw?: string) => {
   198→    const code = (raw ?? memberCode).replace(/\D/g, "");
   199→    if (!code) {
   200→      setMemberInfo(null);
   201→      return;
   202→    }
   203→    setMemberChecking(true);
   204→    try {
   205→      const res = await apiFetch(`/api/member/lookup?code=${encodeURIComponent(code)}`, {
   206→        cache: "no-store",
   207→      });
   208→      const data = await res.json();
   209→      if (!res.ok) {
   210→        setMemberInfo(null);
   211→        showNotice({ type: "error", title: "Member Tidak Ditemukan", message: data.error ?? "Periksa kode member." });
   212→        return;
   213→      }
   214→      setMemberInfo(data.member);
   215→    } catch {
   216→      setMemberInfo(null);
   217→      showNotice({ type: "error", title: "Gagal Memeriksa", message: "Tidak dapat menghubungi server." });
   218→    } finally {
   219→      setMemberChecking(false);
   220→    }
   221→  };
   222→
   223→  const clearMember = () => {
   224→    setMemberCode("");
   225→    setMemberInfo(null);
   226→  };
   227→
   228→  const submit = async () => {
   229→    if (items.length === 0) return;
   230→    setSubmitting(true);
   231→    try {
   232→      const res = await apiFetch("/api/orders", {
   233→        method: "POST",
   234→        headers: { "Content-Type": "application/json" },
   235→        body: JSON.stringify({
   236→          items: items.map((i) => ({ productId: i.product.id, qty: i.qty })),
   237→          customerName: customerName.trim() || `Walk-in${tableNo ? ` Meja ${tableNo}` : ""}`,
   238→          memberCode: memberInfo ? memberInfo.memberCode : undefined,
   239→          type: orderType,
   240→          tableNo: orderType === "DINE_IN" ? tableNo : undefined,
   241→          paymentMethod: payment,
   242→          source: "POS",
   243→        }),
   244→      });
   245→      const data = await res.json();
   246→      if (!res.ok) {
   247→        showNotice({ type: "error", title: "Transaksi Gagal", message: data.error ?? "Coba lagi." });
   248→        return;
   249→      }
   250→      setLastOrder({
   251→        id: data.order.id,
   252→        code: data.order.code,
   253→        total: data.order.total,
   254→        name: data.order.customerName,
   255→        paymentMethod: data.order.paymentMethod,
   256→        paymentStatus: data.order.paymentStatus,
   257→      });
   258→      setItems([]);
   259→      setCustomerName("");
   260→      setTableNo("");
   261→      clearMember();
   262→      refreshOrders("all");
   263→
   264→      // QRIS: tampilkan QR di layar kasir agar pelanggan bisa memindai.
   265→      // Sistem yang melunaskan otomatis saat dana masuk (webhook/polling) —
   266→      // kasir TIDAK menandai lunas.
   267→      if (
   268→        data.order.paymentMethod === "QRIS" &&
   269→        data.order.paymentStatus !== "PAID" &&
   270→        data.order.total > 0
   271→      ) {
   272→        setPayOrder({ id: data.order.id, code: data.order.code, total: data.order.total });
   273→        return;
   274→      }
   275→
   276→      showNotice({
   277→        type: "success",
   278→        title: "Transaksi Berhasil ✓",
   279→        message: `Pesanan ${data.order.code} (${formatIDR(data.order.total)}) telah dicatat dan masuk antrean dapur.`,
   280→      });
   281→    } catch {
   282→      showNotice({ type: "error", title: "Koneksi Bermasalah", message: "Periksa jaringan lalu ulangi." });
   283→    } finally {
   284→      setSubmitting(false);
   285→    }
   286→  };
   287→
   288→  return (
   289→    <div className="flex h-dvh flex-col bg-orange-50/40 pb-[env(safe-area-inset-bottom,0px)] pt-[env(safe-area-inset-top,0px)]">
   290→      {/* ===== TOOLBAR OPERASIONAL POS ===== */}
   291→      <header className="z-30 shrink-0 bg-gradient-to-r from-orange-700 via-orange-600 to-orange-500 text-white shadow-xl">
   292→        <div className="flex flex-wrap items-center gap-x-4 gap-y-2 px-4 py-2.5 lg:px-6">
   293→          <div className="flex items-center gap-2.5">
   294→            <ChickenBadge size={38} />
   295→            <div className="leading-tight">
   296→              <p className="text-sm font-black uppercase tracking-wide">POS Kasir</p>
   297→              <p className="text-[10px] font-semibold text-orange-100">Ayam Geprek Sambal Ijo</p>
   298→            </div>
   299→          </div>
   300→
   301→          <div className="hidden items-center gap-1.5 rounded-full bg-white/15 px-3 py-1 text-[11px] font-bold md:flex">
   302→            <Clock className="h-3.5 w-3.5" /> <span className="tabular-nums">{clock}</span>
   303→          </div>
   304→          <div className="hidden items-center gap-1.5 rounded-full bg-white/15 px-3 py-1 text-[11px] font-bold md:flex">
   305→            <Wifi className="h-3.5 w-3.5" /> Online
   306→          </div>
   307→          <div className="flex items-center gap-1.5 rounded-full bg-white/15 px-3 py-1 text-[11px] font-bold">
   308→            👤 {user.name}
   309→          </div>
   310→
   311→          <div className="ml-auto flex items-center gap-2">
   312→            {lastOrder && (
   313→              <button
   314→                onClick={() => {
   315→                  // Pesanan QRIS belum lunas -> buka lagi layar QR (pantau otomatis)
   316→                  if (
   317→                    lastOrder.paymentMethod === "QRIS" &&
   318→                    lastOrder.paymentStatus !== "PAID" &&
   319→                    lastOrder.total > 0
   320→                  ) {
   321→                    setPayOrder({
   322→                      id: lastOrder.id,
   323→                      code: lastOrder.code,
   324→                      total: lastOrder.total,
   325→                    });
   326→                    return;
   327→                  }
   328→                  showNotice({
   329→                    type: "info",
   330→                    title: `Struk Terakhir — ${lastOrder.code}`,
   331→                    message: `${lastOrder.name} • Total ${formatIDR(lastOrder.total)}. Terima kasih!`,
   332→                  });
   333→                }}
   334→                className="press-effect flex items-center gap-1.5 rounded-full bg-white/15 px-3.5 py-1.5 text-xs font-bold hover:bg-white/25"
   335→                title={
   336→                  lastOrder.paymentMethod === "QRIS" && lastOrder.paymentStatus !== "PAID"
   337→                    ? "Buka lagi layar QRIS (memantau otomatis)"
   338→                    : "Lihat struk terakhir"
   339→                }
   340→              >
   341→                <Receipt className="h-4 w-4" /> {lastOrder.code}
   342→              </button>
   343→            )}
   344→            {(user.role === "ADMIN" || user.role === "OWNER") && (
   345→              <button
   346→                onClick={() => go("dashboard")}
   347→                className="press-effect flex items-center gap-1.5 rounded-full bg-white/15 px-3.5 py-1.5 text-xs font-bold hover:bg-white/25"
   348→              >
   349→                <MonitorSmartphone className="h-4 w-4" /> Mode Admin
   350→              </button>
   351→            )}
   352→            <button
   353→              onClick={() => load(true)}
   354→              aria-label="Muat ulang produk"
   355→              className="press-effect flex h-8 w-8 items-center justify-center rounded-full bg-white/15 hover:bg-white/25"
   356→            >
   357→              <RefreshCw className="h-4 w-4" />
   358→            </button>
   359→            <button
   360→              onClick={onLogout}
   361→              className="press-effect flex items-center gap-1.5 rounded-full bg-white/15 px-3.5 py-1.5 text-xs font-bold hover:bg-white/25"
   362→            >
   363→              <LogOut className="h-4 w-4" /> Keluar
   364→            </button>
   365→          </div>
   366→        </div>
   367→      </header>
   368→
   369→      {/* ===== BAR SCAN BARCODE & PENCARIAN (di bawah header) =====
   370→          Mode 1: Scan Member — scan/ketik kode member (8 digit) → member terpasang
   371→          Mode 2: Nama Produk — saring katalog menu berdasarkan nama produk */}
   372→      <div className="z-20 shrink-0 border-b border-orange-100 bg-white px-4 py-2.5 shadow-sm lg:px-5">
   373→        <div className="flex flex-wrap items-center gap-2">
   374→          {/* Pemilih mode */}
   375→          <div
   376→            role="tablist"
   377→            aria-label="Mode pencarian"
   378→            className="flex shrink-0 items-center gap-0.5 rounded-xl bg-orange-50 p-1 ring-1 ring-orange-200"
   379→          >
   380→            <button
   381→              role="tab"
   382→              aria-selected={scanMode === "member"}
   383→              onClick={() => setScanMode("member")}
   384→              className={cn(
   385→                "press-effect flex items-center gap-1.5 rounded-lg px-3 py-1.5 text-[11px] font-extrabold transition",
   386→                scanMode === "member"
   387→                  ? "bg-orange-600 text-white shadow-md"
   388→                  : "text-orange-700 hover:bg-orange-100"
   389→              )}
   390→            >
   391→              <ScanLine className="h-3.5 w-3.5" /> Scan Member
   392→            </button>
   393→            <button
   394→              role="tab"
   395→              aria-selected={scanMode === "product"}
   396→              onClick={() => setScanMode("product")}
   397→              className={cn(
   398→                "press-effect flex items-center gap-1.5 rounded-lg px-3 py-1.5 text-[11px] font-extrabold transition",
   399→                scanMode === "product"
   400→                  ? "bg-orange-600 text-white shadow-md"
   401→                  : "text-orange-700 hover:bg-orange-100"
   402→              )}
   403→            >
   404→              <Search className="h-3.5 w-3.5" /> Nama Produk
   405→            </button>
   406→          </div>
   407→
   408→          {/* Input sesuai mode */}
   409→          {scanMode === "member" ? (
   410→            <form
   411→              className="relative min-w-[220px] flex-1"
   412→              onSubmit={(e) => {
   413→                e.preventDefault();
   414→                void lookupMember();
   415→              }}
   416→            >
   417→              <ScanLine className="pointer-events-none absolute left-3 top-1/2 h-4 w-4 -translate-y-1/2 text-orange-400" />
   418→              <Input
   419→                inputMode="numeric"
   420→                autoFocus
   421→                placeholder="Scan / ketik kode member (8 digit), Enter…"
   422→                className="border-orange-200 pl-9 pr-24 focus-visible:ring-orange-400"
   423→                value={memberCode}
   424→                disabled={memberChecking}
   425→                onChange={(e) => {
   426→                  const v = e.target.value.replace(/\D/g, "").slice(0, 12);
   427→                  setMemberCode(v);
   428→                  // Kode member selalu 8 digit → cari otomatis begitu genap 8 angka
   429→                  if (v.length === 8 && !memberInfo) void lookupMember(v);
   430→                }}
   431→              />
   432→              <div className="absolute right-2 top-1/2 flex -translate-y-1/2 items-center gap-1.5">
   433→                {memberChecking && <Loader2 className="h-3.5 w-3.5 animate-spin text-orange-500" />}
   434→                {memberCode && !memberInfo && (
   435→                  <button
   436→                    type="button"
   437→                    onClick={() => setMemberCode("")}
   438→                    aria-label="Bersihkan kode"
   439→                    className="press-effect flex h-6 w-6 items-center justify-center rounded-full text-neutral-400 hover:bg-orange-50 hover:text-neutral-600"
   440→                  >
   441→                    <X className="h-3.5 w-3.5" />
   442→                  </button>
   443→                )}
   444→              </div>
   445→            </form>
   446→          ) : (
   447→            <div ref={searchBoxRef} className="relative min-w-[220px] flex-1">
   448→              <Search className="pointer-events-none absolute left-3 top-1/2 h-4 w-4 -translate-y-1/2 text-orange-400" />
   449→              <Input
   450→                autoFocus
   451→                placeholder="Cari menu dengan nama produk… (cth: geprek, mozarella)"
   452→                className="border-orange-200 pl-9 pr-9 focus-visible:ring-orange-400"
   453→                value={query}
   454→                onChange={(e) => {
   455→                  setQuery(e.target.value);
   456→                  setSearchOpen(true);
   457→                }}
   458→                onFocus={() => {
   459→                  if (query.trim()) setSearchOpen(true);
   460→                }}
   461→                onKeyDown={(e) => {
   462→                  // Enter = langsung masukkan hasil teratas ke keranjang
   463→                  if (e.key === "Enter" && popupMatches.length > 0) {
   464→                    e.preventDefault();
   465→                    pickFromPopup(popupMatches[0]);
   466→                  }
   467→                }}
   468→              />
   469→              {query && (
   470→                <button
   471→                  onClick={() => {
   472→                    setQuery("");
   473→                    setSearchOpen(false);
   474→                  }}
   475→                  aria-label="Bersihkan pencarian"
   476→                  className="press-effect absolute right-2 top-1/2 flex h-6 w-6 -translate-y-1/2 items-center justify-center rounded-full text-neutral-400 hover:bg-orange-50 hover:text-neutral-600"
   477→                >
   478→                  <X className="h-3.5 w-3.5" />
   479→                </button>
   480→              )}
   481→
   482→              {/* ===== POPUP HASIL PENCARIAN ===== */}
   483→              <AnimatePresence>
   484→                {searchOpen && q && (
   485→                  <motion.div
   486→                    initial={prefersReduced ? false : { opacity: 0, y: -6, scale: 0.98 }}
   487→                    animate={{ opacity: 1, y: 0, scale: 1 }}
   488→                    exit={prefersReduced ? undefined : { opacity: 0, y: -6, scale: 0.98 }}
   489→                    transition={{ duration: 0.16, ease: "easeOut" }}
   490→                    role="listbox"
   491→                    aria-label="Hasil pencarian menu"
   492→                    className="absolute left-0 right-0 top-full z-50 mt-2 overflow-hidden rounded-2xl border border-orange-200 bg-white shadow-2xl shadow-orange-900/10"
   493→                  >
   494→                    {popupMatches.length === 0 ? (
